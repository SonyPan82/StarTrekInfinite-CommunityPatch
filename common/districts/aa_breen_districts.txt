@base_cost = 300
@city_cost = 500

# Labour Camp
district_labor_camp_breen = {
	base_buildtime = 360
	is_capped_by_modifier = no

	potential = {
		exists = owner
		owner = { is_country_type = contender }
		owner = { has_country_id = breen }
		not = { has_planet_flag = breen_capital }
	}

	show_on_uncolonized = {
		exists = from
		is_colonizable = yes
		uses_district_set = standard
		from = { is_country_type = contender }
		from = { has_country_id = breen }
		not = { has_planet_flag = breen_capital }
	}

	resources = {
		category = planet_districts_industrial
		cost = {
			minerals = @city_cost
		}
		upkeep = {
			energy = 1
			food = 1
		}
	}

	planet_modifier = {
		planet_housing_add = 2
		job_enforcer_add = 1
		job_administrator_add = 1
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_country_id = breen }
		}
		modifier = {
			planet_housing_add = 2
			job_forced_miner_cardassian_add = 1
			job_forced_farmer_cardassian_add = 1
		}
	}

	# AI Forced Miner x1
	ai_resource_production = {
		trigger = {
			always = yes
		 }
		 minerals = 3
		 food = 3
	}
}
