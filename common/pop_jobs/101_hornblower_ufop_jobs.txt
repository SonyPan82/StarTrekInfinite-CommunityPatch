# Starfleet Officer
starfleet_officer_ufop = {
	category = worker
	condition_string = SOLDIER_JOB_TRIGGER
	building_icon = building_stronghold
	clothes_texture_index = 4
		
	pop_modifier = {
		pop_defense_armies_add = 1
	}
	country_modifier = {
		country_admin_cap_add = 1
	}
		
	resources = {
		category = planet_soldiers
		upkeep = {
			food = 0
		}
		
		produces = {
			officers = 3
		}
				
		produces = {
						
			trigger = {
				owner = {
					is_ufop = yes
					has_quest_tree_node = qt_ufop_control_demilitarized_zone_21
				}
			}
						
			unity = 2
		}
			
		produces = {
			trigger = {
				owner = { has_quest_tree_node = qt_ufop_inevitable }
			}
			officers = 2
		}
			
		produces = { 
			 
			trigger = { 
				owner = { 
					is_ufop = yes 
					has_quest_tree_node = qt_ufop_control_neutral_zone
				} 
			} 
			officers = 2
		} 
			
		produces = {
			trigger = {
				exists = planet
				planet = {
					exists = owner
					planet = { has_designation = col_fortress }
				}
			}
			influence = 0.1
		}
			
		produces = {
			trigger = {
				exists = owner
				owner = { has_tradition = tr_defense_oath_citizenship }
			}
			unity = 1
		}

		# Recruitment Campaign, Decision
		produces = {
			trigger = {
				exists = owner
				exists = planet
				planet = {
					has_modifier = recruitment_campaign
				}
			}
			officers = 1
		}
	}

	triggered_planet_modifier = {
	 	potential = {
	 		exists = planet
	 		planet = {
	 			has_modifier = martial_law_ufop
	 		}
	 	}
	 	modifier = {
	 		planet_stability_add = 5
	 	}
	}

	# War Measures, Decision
	triggered_pop_modifier = {
		potential = {
			exists = owner
			exists = planet
			planet = {
				has_modifier = war_measures
			}
		}
		pop_defense_armies_add = 1
	}

	# Fortress World, Planet Designation
	triggered_pop_modifier = {
		potential = {
			exists = planet
			planet = {
				exists = owner
				planet = { has_designation = col_fortress }
			}
		}
		pop_defense_armies_add = 2
	}

	# Extra Ration, Edict
	triggered_pop_modifier = {
		potential = {
			exists = owner
			owner = { has_edict = extra_rations }
		}
		pop_defense_armies_add = 1
	}

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
		
	possible = {
		soldier_job_check_trigger = yes
	}
		
	weight = {
		weight = @soldier_job_weight
	}
}
