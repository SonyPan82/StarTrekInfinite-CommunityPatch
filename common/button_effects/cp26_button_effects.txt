cp26_open_ferengi_branch_manager = {
	potential = {
		is_scope_type = planet
		is_colony = yes
		owner = { is_same_value = from }
		from = {
			is_ai = no
			is_ferengi_empire = yes
		}
	}

	allow = {
		owner = { is_same_value = from }
	}

effect = {
		from = {
			country_event = { id = cp26_ferengi_branch.3 }
		}
	}
}

cp26_build_ferengi_branch_nagus = {
	potential = { from = { is_ferengi_empire = yes } exists = event_target:cp26_branch_selected_planet exists = event_target:cp26_branch_host_country }
	allow = {
		from = {
			resource_stockpile_compare = { resource = influence value >= 50 }
			resource_stockpile_compare = { resource = minerals value >= 400 }
		}
		event_target:cp26_branch_selected_planet = {
			NOR = {
				has_planet_flag = megastructure
				has_planet_flag = has_megastructure
			}
		}
	}
	effect = {
		from = {
			add_resource = { influence = -50 }
			add_resource = { minerals = -400 }
			set_relation_flag = { who = event_target:cp26_branch_host_country flag = cp26_ferengi_branch_trade_relation }
			save_global_event_target_as = cp26_branch_report_ferengi
		}
		event_target:cp26_branch_selected_planet = {
			add_modifier = { modifier = cp26_ferengi_trade_forum days = -1 }
			if = {
				limit = { NOT = { has_planet_flag = cp26_ferengi_trade_hub_present } }
				solar_system = {
					spawn_megastructure = {
						type = "cp26_ferengi_trade_hub"
						planet = prev
						owner = from
						orbit_distance = 2
						orbit_angle = 20
					}
				}
				set_planet_flag = cp26_ferengi_trade_hub_present
			}
			save_global_event_target_as = cp26_branch_report_planet
		}
		event_target:cp26_branch_host_country = {
			set_relation_flag = { who = from flag = cp26_ferengi_branch_trade_relation }
			save_global_event_target_as = cp26_branch_report_host
			every_owned_planet = {
				limit = { is_colony = yes }
				set_surveyed = { surveyed = yes surveyor = from }
				solar_system = {
					set_surveyed = { surveyed = yes surveyor = from }
					every_system_planet = {
						set_surveyed = { surveyed = yes surveyor = from }
					}
				}
			}
			country_event = { id = cp26_ferengi_branch.6 }
		}
		from = { country_event = { id = cp26_ferengi_branch.4 } }
	}
}
