cp26_open_ferengi_branch_manager = {
	potential = {
		is_scope_type = planet
		is_colony = yes
		from = {
			is_ai = no
			is_ferengi_empire = yes
		}
	}

	allow = {
		OR = {
			owner = { is_same_value = from }
			OR = {
				has_modifier = cp26_ferengi_trade_forum
				has_modifier = cp26_ferengi_mining_consortium
				has_modifier = cp26_ferengi_energy_exchange
			}
			AND = {
				custom_tooltip = {
					fail_text = cp26_ferengi_branch_requires_foreign_planet
					owner = { NOT = { is_same_value = from } }
				}
				custom_tooltip = {
					fail_text = cp26_ferengi_branch_requires_commercial_pact
					owner = { has_commercial_pact = from }
				}
				custom_tooltip = {
					fail_text = requires_recipient_not_megacorp
					owner = { NOT = { has_authority = auth_corporate } }
				}
				custom_tooltip = {
					fail_text = cp26_ferengi_branch_requires_eligible_planet
					has_upgraded_capital = yes
				}
				custom_tooltip = {
					fail_text = cp26_ferengi_branch_requires_unoccupied_planet
					NOR = {
						has_modifier = cp26_ferengi_trade_forum
						has_modifier = cp26_ferengi_mining_consortium
						has_modifier = cp26_ferengi_energy_exchange
					}
				}
				custom_tooltip = {
					fail_text = cp26_ferengi_branch_requires_free_partner
					owner = {
						NOT = { has_relation_flag = { who = from flag = cp26_ferengi_branch_trade_relation } }
						NOT = { has_relation_flag = { who = from flag = cp26_ferengi_branch_mining_relation } }
						NOT = { has_relation_flag = { who = from flag = cp26_ferengi_branch_energy_relation } }
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				owner = { is_same_value = from }
			}
			from = {
				country_event = { id = cp26_ferengi_branch.3 }
			}
		}
		else_if = {
			limit = {
				OR = {
					has_modifier = cp26_ferengi_trade_forum
					has_modifier = cp26_ferengi_mining_consortium
					has_modifier = cp26_ferengi_energy_exchange
				}
			}
			save_global_event_target_as = cp26_branch_report_planet
			owner = {
				save_global_event_target_as = cp26_branch_report_host
			}
			from = {
				country_event = { id = cp26_ferengi_branch.4 }
			}
		}
		else = {
			save_global_event_target_as = cp26_branch_selected_planet
			owner = {
				save_global_event_target_as = cp26_branch_host_country
			}
			from = {
				country_event = { id = cp26_ferengi_branch.2 }
			}
		}
	}
}
