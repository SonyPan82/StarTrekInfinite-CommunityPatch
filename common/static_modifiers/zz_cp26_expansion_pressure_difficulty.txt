# CP26 - Expansion Pressure: AI difficulty compensation (Captain/Commodore)
#
# These two blocks fully redefine vanilla's difficulty_captain and
# difficulty_commodore (common/static_modifiers/00_static_modifiers.txt) -
# Paradox static modifier blocks are replaced wholesale by name, not merged
# field by field, so every vanilla field is repeated here unchanged and only
# country_starbase_influence_cost_mult is new.
#
# This continues, at a smaller scale, the same partial offset CP26 already
# gives Admiral/Grand Admiral AI in zz_cp26_hard_ai_difficulty.txt for this
# cost axis - so the curve is smooth across all 5 difficulties instead of
# jumping straight from 0% at Ensign/Captain to -20%/-35% at Admiral+.
# Ensign is untouched on purpose: at the lowest difficulty the AI is meant
# to get no crutches, same as every other vanilla difficulty axis.
#
# Offset only partially cancels the player-facing malus (25-50% of it,
# growing with difficulty) - it is not meant to remove the pressure for
# the AI, only to keep it from under-performing compared to a human who
# can plan ahead, per project wiki "Expansion Pressure" -> AI risk section.
#
# NOTE: like all difficulty_* blocks, these are only meant to apply to
# AI-controlled countries (has_difficulty_bonuses on country_type `default`
# is shared with the player, but the difficulty system itself is assumed,
# consistent with vanilla Stellaris-derived behaviour, to gate application
# to AI countries only). Verify in-game that the player does not receive
# these values after just picking a difficulty - see wiki test checklist.

difficulty_captain = {
	stations_produces_mult = 0.25
	planet_jobs_produces_mult = 0.25
	ships_upkeep_mult = -0.1
	starbase_shipyard_build_cost_mult = -0.1
	pop_resettlement_cost_mult = -0.25
	planet_stability_add = 5

	country_starbase_influence_cost_mult = -0.05
}

difficulty_commodore = {
	stations_produces_mult = 0.5
	planet_jobs_produces_mult = 0.5
	ships_upkeep_mult = -0.2
	starbase_shipyard_build_cost_mult = -0.2
	pop_resettlement_cost_mult = -0.50
	planet_stability_add = 10

	country_starbase_influence_cost_mult = -0.10
}
