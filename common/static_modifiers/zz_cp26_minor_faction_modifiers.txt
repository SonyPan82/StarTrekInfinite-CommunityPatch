# CP26 - Minor Faction Combat Doctrines
#
# Identité tactique propre pour chaque minor faction. Appliqué une fois sur le
# country lors du setup_*_capital (days = -1 = permanent).
#
# Objectif : donner une raison stratégique de différencier les rencontres avec
# chacune de ces 4 factions, sans dérégler l'équilibre vanilla.
#
# Référence lore :
#   - Gorn       : reptiliens lents et costauds, tactique brutale
#   - Tholian    : cristallins, boucliers d'énergie supérieurs mais fragiles
#   - Orion      : pirates rapides, alpha-strike, blindage faible
#   - Breen      : armes cryo, armure froide épaisse, cadence lente

cp26_minor_gorn_combat_doctrine = {
	icon = "GFX_modifier_ship_hull"
	icon_frame = 1

	ship_hull_mult       = 0.35
	ship_armor_mult      = 0.10
	ship_speed_mult      = -0.10
	ship_evasion_mult    = -0.15
}

cp26_minor_tholian_combat_doctrine = {
	icon = "GFX_modifier_shield"
	icon_frame = 1

	ship_shield_mult           = 0.45
	ship_shield_regen_add_perc = 0.40
	ship_hull_mult             = -0.10
}

	icon = "GFX_modifier_ship_fire_rate"
	icon_frame = 1

}

	icon = "GFX_modifier_ship_armor"

	ship_fire_rate_mult = -0.10
}
