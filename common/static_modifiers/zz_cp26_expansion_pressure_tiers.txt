# CP26 - Expansion Pressure (feature: escalating outpost cost)
#
# Four static country modifiers, one per tier, applied/swapped by
# events/zz_cp26_expansion_pressure_events.txt (cp26_expansion_pressure.1).
#
# They are NOT auto-applied by the engine (unlike the special "empire_size_over_cap"
# block) - they only take effect on a country that currently has them added via
# add_modifier, which the pulse event manages every year based on owned starbase count.
#
# Full design rationale, tuning table and testing notes: see project wiki,
# page "Expansion Pressure".
#
# Tier thresholds (owned starbases of any level, is_ship_class = shipclass_starbase):
#   Tier 0 (no modifier) : 1-3 starbases  - free, protects early border consolidation
#   Tier 1               : 4-7 starbases
#   Tier 2               : 8-12 starbases
#   Tier 3               : 13-18 starbases
#   Tier 4               : 19+ starbases

cp26_expansion_pressure_tier_1 = {
	country_starbase_influence_cost_mult = 0.25
}

cp26_expansion_pressure_tier_2 = {
	country_starbase_influence_cost_mult = 0.60
	starbase_outpost_alloys_cost_mult = 0.15
}

cp26_expansion_pressure_tier_3 = {
	country_starbase_influence_cost_mult = 1.10
	starbase_outpost_alloys_cost_mult = 0.30
}

cp26_expansion_pressure_tier_4 = {
	country_starbase_influence_cost_mult = 1.80
	starbase_outpost_alloys_cost_mult = 0.50
}
