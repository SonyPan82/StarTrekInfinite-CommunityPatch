cp26_daystrom_station = {
	entity = "federation_daystrom_station_entity"
	construction_entity = "ufop_01_starbase_habitat_construction_entity"
	portrait = GFX_megastructure_construction_background
	place_entity_on_planet_plane = no
	entity_offset = { x = -50 y = -15 }
	build_time = 1800
	prerequisites = { "tech_cp26_daystrom_institute" }

	potential = {
		is_ufop = yes
	}

	possible = {
		solar_system = {
			exists = owner
			owner = {
				has_country_flag = ufop
			}
		}
	}

	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = cp26_daystrom_requires_planet_or_moon
				NOT = { is_star = yes }
			}
			custom_tooltip = {
				fail_text = cp26_daystrom_requires_ufop_system
				solar_system = {
					owner = { has_country_flag = ufop }
				}
			}
			custom_tooltip = {
				fail_text = cp26_daystrom_requires_non_capital_system
				solar_system = {
					NOT = { is_same_value = owner.capital_scope.solar_system }
				}
			}
			custom_tooltip = {
				fail_text = cp26_daystrom_requires_empty_system
				solar_system = {
					NOT = {
						any_system_megastructure = {
							NOT = { is_megastructure_type = cp26_daystrom_station }
						}
					}
				}
			}
			custom_tooltip = {
				fail_text = cp26_daystrom_requires_no_megastructure
				NOT = { has_planet_flag = has_megastructure }
			}
		}
	}

	resources = {
		category = megastructures
		cost = {
			alloys = 5000
			influence = 200
		}
		upkeep = {
			energy = 10
		}
	}

	country_modifier = {
		planet_jobs_research_produces_mult = 0.10
	}

	on_build_complete = {
		save_global_event_target_as = cp26_daystrom_station_scope
		from = {
			set_country_flag = cp26_daystrom_station_built
			if = {
				limit = {
					NOT = { has_country_flag = cp26_daystrom_arc_initialized }
					NOT = { has_country_flag = cp26_daystrom_arc_ended }
					NOT = { has_country_flag = cp26_daystrom_spawned_rift_empire }
				}
				set_country_flag = cp26_daystrom_arc_initialized
				country_event = { id = cp26_daystrom.1 days = 120 }
			}
		}
		fromfrom.solar_system = {
			set_star_flag = cp26_daystrom_station_system
			save_global_event_target_as = cp26_daystrom_system
		}
		fromfrom.planet = {
			set_planet_flag = has_megastructure
			save_global_event_target_as = cp26_daystrom_target_world
		}
	}

	ai_weight = { weight = 0 }
}
