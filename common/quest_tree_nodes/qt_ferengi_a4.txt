# Ferengi Quest Tree - A4
# Règle 4 : "Peace is good for business."
# Condition : avoir proposé la paix lors d'une guerre
#          OU avoir envoyé une flotte militaire à Bajor (arc Libération)

qt_ferengi_a4 = {
	position = { x = 0 y = 3 }

	potential = {
		has_country_id = ferengi
	}

	parents = {
	}

	blocked_by = {
	}

	on_before_possible = {
		if = {
			limit = {
				OR = {
					num_diplo_actions_entered = {
						type  = action_offer_peace
						value >= 1
					}
					has_country_flag = cp26_bajor_fleet_arrived
				}
			}
			set_variable = { which = qt_ferengi_a4_peace_achieved value = 1 }
		}
		else = {
			set_variable = { which = qt_ferengi_a4_peace_achieved value = 0 }
		}
	}

	possible = {
		custom_tooltip = {
			success_text = qt_ferengi_a4_peace_tooltip_success
			fail_text    = qt_ferengi_a4_peace_tooltip_fail
			check_variable = {
				which = qt_ferengi_a4_peace_achieved
				value >= 1
			}
		}
	}

	on_started = {
	}

	on_blocked = {
	}

	on_finished = {
		add_resource = {
			energy    = 800
			influence = 150
		}
	}

	modifier = {
	}

	ai_weight = {
		factor = 1000
	}
}
