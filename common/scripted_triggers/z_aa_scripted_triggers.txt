is_ferengi_empire = {
	OR = {
		is_country_type = ferengi_alliance
		has_country_flag = ferengi
		has_country_id = "ferengi"
	}
}

is_regular_empire = {
	or = {
		is_country_type = default
		is_country_type = contender
		AND = {
			is_ferengi_empire = yes
			NOT = { has_authority = auth_corporate }
		}
	}
}

# Backward-compatible gate for restored CP26 diplomatic actions.
# Keep non-occupied-state diplomatic options available regardless of the occupied-state mode.
cp26_diplomacy_mode_allows_actor = {
	always = yes
}

# Global gate for CP26 occupied-state diplomacy.
cp26_vassalization_mode_allows_actor = {
	OR = {
		AND = {
			NOT = { has_global_flag = cp26_diplo_vassalization_disabled }
			NOT = { has_global_flag = cp26_diplo_vassalization_enabled }
			is_regular_empire = yes
		}
		AND = {
			has_global_flag = cp26_diplo_vassalization_enabled
			is_regular_empire = yes
		}
	}
}

is_cp26_major_mission_country = {
	OR = {
		is_ufop = yes
		is_klingon = yes
		is_romulan = yes
		is_cardassian = yes
	}
}
