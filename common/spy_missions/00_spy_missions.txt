#Gather Intel Mission
spy_situation_gather_intel = {
	mission_type = espionage_mission
	situation = spy_mission_situation_01
	scope = starbase
	success_dc = 4
	partial_success_dc = 1

	experience = value:spy_mission_experience|MONTHLY_PROGRESS|@mission_monthly_progress_base|

	tooltip_desc = spy_situation_gather_intel_effect_desc

	possible = {
		exists = from.owner
		from.owner = { NOT = { is_same_value = root.owner } }

		AND = { NOT = { from.owner = {is_neutral_zone = yes } } }
	}

	on_start = { 
		# Saving the event targets for the localization
		from = { save_event_target_as = spy_mission_scope }
		from.solar_system = { save_event_target_as = spy_mission_system }
		from.owner = { save_event_target_as = spy_mission_target_country }
		this.leader = { save_event_target_as = spy_mission_spy }
		from.fleet = { set_fleet_flag = spy_situation_gather_intel_flag }
		owner = { save_event_target_as = spy_owner }
	}

	on_cancel = { from.fleet = { remove_fleet_flag = spy_situation_gather_intel_flag } }

	on_success = { 
		owner = { 
			country_event = { id = spy_popup.20 }
			add_intel_espionage = yes
		}
		spy_mission_misc_effects = yes
	}

	on_partial_success = { 
		owner = { 
			country_event = { id = spy_popup.20 } 
			add_intel_espionage = yes
		}

		event_target:spy_mission_target_country = {
			trigger_silent_alarm = { location = event_target:spy_mission_system }
		}
		spy_mission_misc_effects = yes
	}
        
	on_fail = {
		if = { 
			limit = { owner = { has_civic = civic_cunning_subterfuge } }
			from = {
				solar_system = {
					every_fleet_in_system = {
						limit = { 
							is_ship_class = shipclass_spy_ship
							owner = { is_same_value = root.owner }
						}
						add_modifier = { modifier = spy_mission_civic_cunning_subterfuge }					
					}
				}
			}
		}
        
		if = {
			limit = { owner = { has_civic = civic_twofold_chances } }
			random_list = {
				8 = {}
				2 = {			
					owner = {
					country_event = { id = spy_popup.50 }
					add_intel_espionage = yes
					}
				}
			}

		}
		else = {
			owner = { country_event = { id = spy_popup.30 } }
		}

		from.fleet = { remove_fleet_flag = spy_situation_gather_intel_flag }

		fleet_event = { id = mission.1000 }
	}
}

# CP26 - Romulan Projet Synth
spy_situation_cp26_projet_synth = {
	mission_type = espionage_mission
	situation = spy_mission_situation_02
	scope = planet
	success_dc = 9
	partial_success_dc = 6

	experience = value:spy_mission_experience|MONTHLY_PROGRESS|@mission_monthly_progress_base|

	tooltip_desc = spy_situation_cp26_projet_synth_effect_desc

	possible = {
		owner = {
			is_romulan = yes
			is_major = yes
			has_country_flag = cp26_romulan_projet_synth_unlocked
			NOT = { has_country_flag = cp26_romulan_synth_cooldown }
		}
		exists = from.owner
		from.owner = { NOT = { is_same_value = root.owner } }
		from = {
			NOT = { has_planet_flag = cp26_synth_rebellion_active }
			any_owned_pop = {
				species = { is_species_class = CP26_A500 }
			}
		}
	}

	on_start = {
		from = { save_event_target_as = spy_mission_scope }
		from.solar_system = { save_event_target_as = spy_mission_system }
		from.owner = { save_event_target_as = spy_mission_target_country }
		this.leader = { save_event_target_as = spy_mission_spy }
		owner = {
			save_event_target_as = spy_owner
			set_country_flag = cp26_romulan_synth_in_progress
		}
		from = { set_planet_flag = cp26_synth_targeted_flag }
	}

	on_cancel = {
		owner = { remove_country_flag = cp26_romulan_synth_in_progress }
		from = { remove_planet_flag = cp26_synth_targeted_flag }
	}

	on_success = {
		owner = {
			set_country_flag = cp26_romulan_synth_cooldown
			remove_country_flag = cp26_romulan_synth_in_progress
			country_event = { id = cp26_synth_rom.20 }
			country_event = { id = cp26_synth_rom.30 days = 720 }
		}
		create_country = {
			name = "NAME_A500_Rebels"
			type = global_event
			auto_delete = yes
			effect = {
				set_faction_hostility = {
					set_hostile = yes
					set_neutral = no
					set_friendly = no
				}
			}
		}
		last_created_country = { save_global_event_target_as = cp26_synth_rebel_country }
		from = {
			save_global_event_target_as = cp26_synth_target_planet
			add_planet_devastation = 15
			add_modifier = { modifier = cp26_synth_unrest days = 720 }
			set_planet_flag = cp26_synth_rebellion_active
			remove_planet_flag = cp26_synth_targeted_flag
			while = {
				count = 3
				if = {
					limit = { exists = event_target:et_a500_species_global }
					create_army = {
						owner = event_target:cp26_synth_rebel_country
						type = rebel_army
						species = event_target:et_a500_species_global
					}
				}
				else = {
					create_army = {
						owner = event_target:cp26_synth_rebel_country
						type = rebel_army
					}
				}
			}
			if = {
				limit = { owner = { is_ufop = yes } }
				owner = {
					if = {
						limit = { NOT = { has_event_chain = cp26_synth_rebellion_chain } }
						begin_event_chain = {
							event_chain = "cp26_synth_rebellion_chain"
							target = THIS
						}
					}
					enable_special_project = {
						name = "CP26_SYNTH_GROUND_ASSAULT"
						location = event_target:cp26_synth_target_planet
						owner = this
					}
					country_event = { id = cp26_synth_rom.25 }
				}
			}
		}
		spy_mission_misc_effects = yes
	}

	on_partial_success = {
		owner = {
			set_country_flag = cp26_romulan_synth_cooldown
			remove_country_flag = cp26_romulan_synth_in_progress
			country_event = { id = cp26_synth_rom.20 }
			country_event = { id = cp26_synth_rom.30 days = 720 }
		}
		create_country = {
			name = "NAME_A500_Rebels"
			type = global_event
			auto_delete = yes
			effect = {
				set_faction_hostility = {
					set_hostile = yes
					set_neutral = no
					set_friendly = no
				}
			}
		}
		last_created_country = { save_global_event_target_as = cp26_synth_rebel_country }
		from = {
			save_global_event_target_as = cp26_synth_target_planet
			add_planet_devastation = 15
			add_modifier = { modifier = cp26_synth_unrest days = 720 }
			set_planet_flag = cp26_synth_rebellion_active
			remove_planet_flag = cp26_synth_targeted_flag
			while = {
				count = 3
				if = {
					limit = { exists = event_target:et_a500_species_global }
					create_army = {
						owner = event_target:cp26_synth_rebel_country
						type = rebel_army
						species = event_target:et_a500_species_global
					}
				}
				else = {
					create_army = {
						owner = event_target:cp26_synth_rebel_country
						type = rebel_army
					}
				}
			}
			if = {
				limit = { owner = { is_ufop = yes } }
				owner = {
					if = {
						limit = { NOT = { has_event_chain = cp26_synth_rebellion_chain } }
						begin_event_chain = {
							event_chain = "cp26_synth_rebellion_chain"
							target = THIS
						}
					}
					enable_special_project = {
						name = "CP26_SYNTH_GROUND_ASSAULT"
						location = event_target:cp26_synth_target_planet
						owner = this
					}
					country_event = { id = cp26_synth_rom.25 }
				}
			}
		}
		event_target:spy_mission_target_country = {
			trigger_silent_alarm = { location = event_target:spy_mission_system }
		}
		spy_mission_misc_effects = yes
	}

	on_fail = {
		owner = {
			remove_country_flag = cp26_romulan_synth_in_progress
			country_event = { id = cp26_synth_rom.21 }
		}

		event_target:spy_mission_target_country = {

# Counter Espionage
# Detects every Spy Ship that is not ours
mission_counter_espionage = {
	mission_type = counter_espionage_mission
	scope = galactic_object
	length = @counter_espionage_mission_lenght

	experience = value:spy_mission_experience|MONTHLY_PROGRESS|@mission_monthly_progress_base|

	# Always success
	success_dc = 2
	partial_success_dc = 1

	tooltip_desc = mission_counter_espionage_effect_desc

	possible = {
		exists = from
		exists = from.owner

		owner = {
			is_same_value = prev.from.owner
		}
	}

	on_start = {
		from = {
			solar_system = {
				every_system_planet = {
					limit = {
						is_colony = yes
					}
					add_modifier = { modifier = spy_added_stability days = 31 }
				}
			}
	    }
	}

	on_cancel = {
	}

	on_success = {
		from = {
			every_fleet_in_system = {
				limit = {
					is_ship_class = shipclass_spy_ship

					owner = {
						NOT = { is_same_value = prevprevprev.owner }
					}

					leader = {
						NOT = { leader_opposed_skill_check = prevprevprev.leader }
					}
				}

				solar_system = {
					every_fleet_in_system = {
						limit = {
							owner = { 
								has_civic = civic_diplomatic_expertise
								is_same_value = prevprevprev.owner
							}
							is_ship_class = shipclass_spy_ship
						}
						add_modifier = { modifier = spy_mission_civic_cunning_subterfuge }
					}
				}

				if = {
					limit = {
						NOT = { was_detected = { detected_by = prevprev.owner } }
					}

					prevprev.owner = {
						change_variable = {
							which = "spies_detected_count"
							value = 1
						}
						
						change_variable = {
							which = "counter_espionage_win_count"
							value = 1
						}
					}
				}

				if = {
					limit = {
						prevprev.owner = {
						 	has_civic = civic_paranoia
						}
					}
					
					prevprev.owner = {
						remove_modifier = paranoia_counter_espionage
						add_modifier = { 
							modifier = paranoia_counter_espionage 
							days = 720
						}
					}
				}

				if = {
					limit = { exists = solar_system.starbase }
					solar_system.starbase.fleet = {
						add_modifier = { modifier = spy_detected_starbase_range_boost days = @spy_detected_starbase_range_boost_days }
					}
				}

				set_detected = { detected_by = prevprev.owner }

				add_global_tension = {
					amount = @spy_mission_fail_global_tension
					type = spy_mission_detected
					country = prevprev.owner
				} 

				prevprev.owner = {
					add_opinion_modifier = {
						who = prev.owner
						modifier = opinion_spy_mission_fail
					}
				}

				prev.from = { save_event_target_as = spy_detected_system }
				root.owner = { save_event_target_as = spy_detected_system_owner }
				owner = {
					country_event = { id = country.27000 }
				}
				
				fail_mission = yes
			}
			
			counter_espionage_finished = yes
		}

		owner = {
			if = {
				limit = {
					check_variable = {
						which = "spies_detected_count"
						value > 0
					}
				}

				set_variable = {
					which = "spies_detected_count"
					value = 0
				}
			}
		}
	}

	on_partial_success = {
	}

	on_fail = {
	}
}

# Counter Propaganda
mission_counter_propaganda = {
	mission_type = espionage_mission
	scope = planet
	length = @mission_lenght

	experience = value:spy_mission_experience|MONTHLY_PROGRESS|@mission_monthly_progress_base|
	
	success_dc = 8
	partial_success_dc = 5

	tooltip_desc = mission_counter_propaganda_effect_desc

	possible = {
		exists = from.owner
		from.owner = { is_same_value = root.owner }
	}

	on_start = {
		this.leader = { save_event_target_as = spy_mission_spy }
	}

	on_cancel = {
	}

	on_success = {
		from = {
			convert_unloyal_pops = { 
				percentage = 1.0
				spy = event_target:spy_mission_spy
			}
		}
	}

	on_partial_success = {
		from = {
			resettle_unloyal_pops = { 
				percentage = 0.5
				spy = event_target:spy_mission_spy
			}
			convert_unloyal_pops = { 
				percentage = 1.0
				spy = event_target:spy_mission_spy
			}
		}
	}

	on_fail = {
		from = {
			resettle_unloyal_pops = { 
				percentage = 1.0
				spy = event_target:spy_mission_spy
			}
		}
		fleet_event = { id = mission.1000 }
	}
}
