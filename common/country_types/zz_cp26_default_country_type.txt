		# Vanilla used hard-coded min/max for constructor/science/colonizer,
		# which left late-game AIs under-staffed once their empire grew.
		#
		# The cp26_empire_size_* country flags are updated yearly by event
		# cp26_ai_pulse.1 (events/zz_cp26_ai_pulse_events.txt) based on
		# count_owned_planet :

		#   medium : 4 - 7
		#   large  : 8 - 14
		#   huge   : 15+

		# Each ship category below uses these flags as fraction modifiers,
		# growing the AI's construction priorities as its empire scales.
