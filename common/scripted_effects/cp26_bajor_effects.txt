cp26_bajor_bootstrap_targets_effect = {
	every_country = {
		limit = { has_country_flag = bajoran }
		save_global_event_target_as = bajoran_country
		capital_scope = {
			save_global_event_target_as = bajoran_planet
			solar_system = { save_global_event_target_as = bajoran_system }
		}
	}
	every_country = {
		limit = {
			OR = {
				has_country_flag = cardassia
				is_cardassian = yes
			}
		}
		save_global_event_target_as = cardassian_country
	}
	every_country = {
		limit = { has_country_flag = ufop }
		save_global_event_target_as = ufop_country
	}
	every_country = {
		limit = { has_country_flag = klingon }
		save_global_event_target_as = klingon_country
	}
	every_country = {
		limit = { has_country_flag = romulan }
		save_global_event_target_as = romulan_country
	}
	every_country = {
		limit = { has_country_flag = ferengi }
		save_global_event_target_as = ferengi_country
	}
}


cp26_spawn_terok_nor_at_bajor = {
	cp26_bajor_bootstrap_targets_effect = yes
	if = {
		limit = {
			exists = event_target:bajoran_system
			exists = event_target:bajoran_planet
			exists = event_target:cardassian_country
			event_target:cardassian_country = {
				NOT = { has_country_flag = cp26_terok_nor_built }
			}
			event_target:bajoran_system = {
				NOT = {
					any_system_megastructure = {
						is_megastructure_type = cp26_terok_nor
					}
				}
			}
		}
		event_target:bajoran_system = {
			spawn_megastructure = {
				type = "cp26_terok_nor"
				planet = event_target:bajoran_planet
				owner = event_target:cardassian_country
				orbit_distance = 8
				orbit_angle = 45
			}
		}
		event_target:cardassian_country = {
			set_country_flag = cp26_terok_nor_built
		}
		event_target:bajoran_system = {
			set_star_flag = cp26_terok_nor_system
		}
	}
}

cp26_bajor_start_rebellion_setup = {
	if = {
		limit = { NOT = { has_global_flag = cp26_bajor_rebellion_active } }
		set_global_flag = cp26_bajor_rebellion_active
		if = {
			limit = { exists = event_target:bajoran_planet }
			event_target:bajoran_planet = {
				set_planet_flag = cp26_bajor_in_rebellion
				add_planet_devastation = 25
			}
		}
	}
}
