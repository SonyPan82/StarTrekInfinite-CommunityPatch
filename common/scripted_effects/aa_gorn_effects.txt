setup_gorn_capital = {
	if = {
		limit = {
			exists = event_target:gorn_country
		}
		# CP26 — Gorn combat doctrine : tough but slow reptilian fleets.
		event_target:gorn_country = {
			if = {
				limit = { NOT = { has_modifier = cp26_minor_gorn_combat_doctrine } }
				add_modifier = { modifier = cp26_minor_gorn_combat_doctrine days = -1 }
			}
		}
		set_surveyed = { surveyed = yes surveyor = event_target:gorn_country }
		if = {
			limit = { exists = event_target:ufop_country }
			set_surveyed = { surveyed = yes surveyor = event_target:ufop_country }
		}
		if = {
			limit = { exists = event_target:klingon_country }
			set_surveyed = { surveyed = yes surveyor = event_target:klingon_country }
		}
		if = {
			limit = { exists = event_target:romulan_country }
			set_surveyed = { surveyed = yes surveyor = event_target:romulan_country }
		}
		if = {
			limit = { exists = event_target:cardassian_country }
			set_surveyed = { surveyed = yes surveyor = event_target:cardassian_country }
		}
		random_system_planet = {
			limit = {
				has_planet_flag = gorn_capital				
			}
				while = {
					count = 21
					create_pop = { species = "Gorn" }
				}
				add_deposit = d_ore_rich_slopes
				add_deposit = d_rich_mountain
				add_deposit = d_hot_springs
				add_deposit = d_geothermal_vent
				clear_blockers = yes
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
			add_district_and_planet_size_if_needed_effect = { district = district_mining }
			add_district_and_planet_size_if_needed_effect = { district = district_mining }
			add_district_and_planet_size_if_needed_effect = { district = district_industrial }
			add_district_and_planet_size_if_needed_effect = { district = district_generator }
			add_district_and_planet_size_if_needed_effect = { district = district_generator }
			add_district_and_planet_size_if_needed_effect = { district = district_gorn_hunting }
			add_district_and_planet_size_if_needed_effect = { district = district_gorn_hunting }
			add_district_and_planet_size_if_needed_effect = { district = district_gorn_hunting }
			add_building = building_capital
		}
			if = {
				limit = {
					NOT = { exists = starbase }
				}
				create_starbase = {
					size = starbase_starport
					owner = event_target:gorn_country
					module = shipyard
					building = crew_quarters
				}
			}
	}
}

setup_gorn_territory = {
	if = {

			set_surveyed = { surveyed = yes surveyor = event_target:ufop_country }
		}
		if = {
			limit = { exists = event_target:klingon_country }
			set_surveyed = { surveyed = yes surveyor = event_target:klingon_country }
		}
		if = {
			limit = { exists = event_target:romulan_country }
			set_surveyed = { surveyed = yes surveyor = event_target:romulan_country }
		}

			set_surveyed = { surveyed = yes surveyor = event_target:cardassian_country }
		}
			if = {

					NOT = { exists = starbase }
				}

				}
			}
	}
