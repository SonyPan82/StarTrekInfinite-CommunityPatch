setup_breen_capital = {
	if = {
		limit = { exists = event_target:breen_country }
		# CP26 - Breen combat doctrine : heavy cryo armor, slow cadence.
		event_target:breen_country = {
			if = {
				limit = { NOT = { has_modifier = cp26_minor_breen_combat_doctrine } }
				add_modifier = { modifier = cp26_minor_breen_combat_doctrine days = -1 }
			}
		}
		random_system_planet = {
			limit = {
				has_planet_flag = breen_capital
			}
			while = {
				count = 23
				create_pop = { species = "Breen" }
			}
			clear_blockers = yes
			add_district_and_planet_size_if_needed_effect = { district = district_city }
			add_district_and_planet_size_if_needed_effect = { district = district_city }
			add_district_and_planet_size_if_needed_effect = { district = district_mining }
			add_district_and_planet_size_if_needed_effect = { district = district_mining }
			add_district_and_planet_size_if_needed_effect = { district = district_industrial }
			add_district_and_planet_size_if_needed_effect = { district = district_generator }
			add_district_and_planet_size_if_needed_effect = { district = district_generator }
			add_district_and_planet_size_if_needed_effect = { district = district_farming }
			add_district_and_planet_size_if_needed_effect = { district = district_farming }
			add_district_and_planet_size_if_needed_effect = { district = district_farming }
			add_district_and_planet_size_if_needed_effect = { district = district_civil_forum_breen }
			add_building = building_capital
		}
		random_system_planet = {
			limit = {
				has_planet_flag = breenus_secundus
			}
			set_owner = event_target:breen_country
			set_colony_type = col_default
			add_deposit = d_underworld_caverns
			add_deposit = d_ore_rich_slopes
			clear_blockers = yes

			add_district_and_planet_size_if_needed_effect = { district = district_city }
			add_district_and_planet_size_if_needed_effect = { district = district_labor_camp_breen }
			add_district_and_planet_size_if_needed_effect = { district = district_labor_camp_breen }

			while = {
				count = 5
				create_pop = { species = "Breen" }
			}

			create_pop = { species = "Bajoran" }
			create_pop = { species = "Romulan" }
			create_pop = { species = "Cardassian" }
			create_pop = { species = "Klingon" }
			add_building = building_capital
		}
			if = {
				limit = {
					NOT = { exists = starbase }
				}
				create_starbase = {
					size = starbase_starport
					owner = event_target:breen_country
					module = shipyard
					building = crew_quarters
				}
			}
	}
}

setup_breen_territory = {
	if = {
		limit = { exists = event_target:breen_country }
		set_surveyed = { surveyed = yes surveyor = event_target:breen_country }
		create_starbase = {
			size = starbase_outpost
			owner = event_target:breen_country
		}
	}
}
