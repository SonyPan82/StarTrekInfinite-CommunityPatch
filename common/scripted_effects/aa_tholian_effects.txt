spawn_tholian_system = {
	spawn_system = {
		min_jumps = 1
		max_jumps = 1
		min_distance = 3
		max_distance = 14
		initializer = tholian_system_initializer
		hyperlane = yes
		is_discovered = yes
	}
}

		min_distance = 3
		max_distance = 14
		initializer = berellium_system_initializer
		hyperlane = yes
		is_discovered = yes
	}
}

setup_tholian_capital = {
	if = {
		limit = {
			exists = event_target:tholian_country
		}
		random_system_planet = {
			limit = {
				has_planet_flag = tholian_capital
			}
			add_deposit = d_ore_rich_slopes
			add_deposit = d_rich_mountain
			add_deposit = d_hot_springs
			add_deposit = d_geothermal_vent
			while = {
				count = 20
				create_pop = { species = "Tholian" }
			}
			clear_blockers = yes
			add_district_and_planet_size_if_needed_effect = { district = district_city_tholian }
			add_district_and_planet_size_if_needed_effect = { district = district_city }
			add_district_and_planet_size_if_needed_effect = { district = district_city }
			add_district_and_planet_size_if_needed_effect = { district = district_generator }
			add_district_and_planet_size_if_needed_effect = { district = district_generator }
			add_district_and_planet_size_if_needed_effect = { district = district_mining }
			add_district_and_planet_size_if_needed_effect = { district = district_mining }
			add_district_and_planet_size_if_needed_effect = { district = district_industrial }
			add_district_and_planet_size_if_needed_effect = { district = district_industrial }
			add_district_and_planet_size_if_needed_effect = { district = district_farming }
			remove_building = building_colony_shelter
			add_building = building_capital
		}
		create_starbase = {
			size = starbase_starport
			owner = event_target:tholian_country
			module = shipyard
			building = crew_quarters
		}
	}
}

setup_tholian_territory = {
	if = {
		limit = {
			exists = event_target:tholian_country
		}
		set_surveyed = { surveyed = yes surveyor = event_target:tholian_country }
		create_starbase = {
			size = starbase_outpost
			owner = event_target:tholian_country
		}
	}
}
