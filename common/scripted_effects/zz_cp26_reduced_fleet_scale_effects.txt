# CP26 reduced-fleet scale overrides.
# These effects replace high-volume vanilla/Infinite scripted spawns with
# compact task forces so late-game object counts stay sane.

create_war_situation_mercenary_fleet_effect = {
	IF = {
		limit = { command_limit <= 20 }
		create_fleet = {
			effect = {
				set_owner = root
				while = {
					count = 2
					create_ship = {
						name = random
						design = "NAME_Small_War_Sit_Mercenary"
						prefix = no
						graphical_culture = nausicaan_01
						initial_officers_perc = 1
					}
				}
				while = {
					count = 2
					create_ship = {
						name = random
						design = "NAME_Medium_War_Sit_Mercenary"
						prefix = no
						graphical_culture = nausicaan_01
						initial_officers_perc = 1
					}
				}
			}
		}
	}
	else_if = {
		limit = { command_limit <= 35 }
		create_fleet = {
			effect = {
				set_owner = root
				while = {
					count = 3
					create_ship = {
						name = random
						design = "NAME_Small_War_Sit_Mercenary"
						prefix = no
						graphical_culture = nausicaan_01
						initial_officers_perc = 1
					}
				}
				while = {
					count = 3
					create_ship = {
						name = random
						design = "NAME_Medium_War_Sit_Mercenary"
						prefix = no
						graphical_culture = nausicaan_01
						initial_officers_perc = 1
					}
				}
			}
		}
	}
	else = {
		create_fleet = {
			effect = {
				set_owner = root
				while = {
					count = 4
					create_ship = {
						name = random
						design = "NAME_Small_War_Sit_Mercenary"
						prefix = no
						graphical_culture = nausicaan_01
						initial_officers_perc = 1
					}
				}
				while = {
					count = 4
					create_ship = {
						name = random
						design = "NAME_Medium_War_Sit_Mercenary"
						prefix = no
						graphical_culture = nausicaan_01
						initial_officers_perc = 1
					}
				}
			}
		}
	}
	last_created_fleet = {
		set_fleet_flag = mercenary_fleet@event_target:current_war_scope
	}
}

create_ufop_starting_military_fleet = {
	create_fleet = {
		name = "NAME_UFOP_First_Fleet"
		settings = {
			spawn_debris = no
		}

		effect = {
			set_fleet_stance = passive
			set_owner = root

			while = {
				count = 2
				create_ship = {
					name = ""
					prefix = no
					random_existing_design = miranda
					initial_officers_perc = 1
					graphical_culture = prev
				}
			}

			set_location = event_target:ufop_military_station
		}
	}

	last_created_fleet = {
		save_global_event_target_as = ufop_main_fleet
	}
}

create_marauder_raiders = {
	if = {
		limit = { years_passed < 40 }
		create_first_storm_wing = yes
		set_fleet_flag = marauder_raider_fleet_1
	}
	if = {
		limit = {
			years_passed > 39
			years_passed < 60
		}
		create_star_rider_flotilla = yes
		set_fleet_flag = marauder_raider_fleet_2
	}
	if = {
		limit = {
			years_passed > 59
			years_passed < 80
		}
		create_armada_of_the_voidborn = yes
		set_fleet_flag = marauder_raider_fleet_3
	}
	if = {
		limit = { years_passed > 80 }
		create_armada_of_the_voidborn = yes
		set_fleet_flag = marauder_raider_fleet_4
	}
}

spawn_borg_cube_mid_wave_at_location = {
	create_fleet = {
		name = "NAME_BORG_Wave_Fleet"
		settings = {
			spawn_debris = no
			can_upgrade = no
			can_disband = no
			can_change_composition = no
			can_change_leader = no
		}

		effect = {
			set_owner = event_target:borg_crisis_country

			create_ship = {
				name = "NAME_FROM_BORG"
				prefix = no
				design = "NAME_BorgCube_mid"
				graphical_culture = event_target:borg_crisis_country
				initial_officers_perc = @starting_officers
			}

			set_location = {
				target = $LOCATION$
				distance = 300
				angle = random
			}

			queue_actions = {
				move_to = $LOCATION$
			}
		}
	}
}
