spawn_ferengi_xuracorp_system = {
	spawn_system = {
		min_jumps = 1
		max_jumps = 1
		min_distance = 3
		max_distance = 21
		initializer = ferengi_xuracorp_init
		hyperlane = yes
		is_discovered = yes
	}
}

setup_ferengi_territory = {
	if = {
		limit = {
			NOT = { exists = event_target:ferengi_country }
		}
		every_country = {
			limit = {
				has_country_flag = ferengi
			}
			save_global_event_target_as = ferengi_country
		}
	}
	if = {
		limit = {
			exists = event_target:ferengi_country
		}
		set_surveyed = { surveyed = yes surveyor = event_target:ferengi_country }
		create_starbase = {
			size = starbase_outpost
			owner = event_target:ferengi_country
		}
	}
}
