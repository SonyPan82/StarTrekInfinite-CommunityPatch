# CP26 — Dominion Jem'Hadar spawns that do not rely on naval_cap (create_fleet_from_naval_cap often yields empty fleets).

# Bajor system — small scout picket (Phase 1 / DTL reactions).
cp26_spawn_jem_scouts_bajor = {
	random_galaxy_planet = {
		limit = {
			has_planet_flag = denorios_belt_target
			solar_system = { has_star_flag = bajoran_system }
		}
		save_event_target_as = cp26_dom_fleet_spawn_system
	}
	random_country = {
		limit = { cp26_is_dominion_empire = yes }
		save_event_target_as = cp26_dom_fleet_owner
	}
	if = {
		limit = {
			exists = event_target:cp26_dom_fleet_owner
			exists = event_target:cp26_dom_fleet_spawn_system
		}
		event_target:cp26_dom_fleet_owner = {
			create_fleet = {
				name = "Jem'Hadar Scout"
				effect = {
					set_owner = event_target:cp26_dom_fleet_owner
					while = {
						count = 3
						create_ship = {
							name = random
							random_existing_design = corvette
							graphical_culture = dominion_01
							prefix = no
						}
					}
					set_location = event_target:cp26_dom_fleet_spawn_system
				}
				settings = {
					can_upgrade = no
					can_disband = yes
					spawn_debris = yes
				}
			}
		}
	}
}
