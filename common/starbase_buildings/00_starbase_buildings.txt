		}

		modifier = {
			factor = 0

					any_system_planet = {
						exists = owner
						owner = { is_country_type = default }
					}
				}
			}
		}
	}

ice_mining_station = {
	icon = "GFX_starbase_ice_mining_station"
	construction_days = 360

	potential = {
		exists = owner
		owner = { has_technology = tech_ice_mining_station }
	}

	possible = {
		custom_tooltip = {
			fail_text = "requires_starport"
			has_starbase_size >= starbase_starport
		}

		custom_tooltip = {
			fail_text = requires_ice_source
			solar_system = {
				OR = {
					has_planet_class = pc_ice_asteroid
					has_planet_class = pc_tundra
					has_planet_class = pc_arctic
					has_planet_class = pc_alpine
					has_planet_class = pc_frozen
					has_planet_class = pc_ice
				}
			}
		}
	}

	resources = {
		category = starbase_buildings
		cost = {
			alloys = 100
		}

		produces = {
			deuterium = 10
		}

		upkeep = {
			energy = 2
		}
	}

	# custom_tooltip = ice_mining_station_tooltip

	# system_modifier = {
	# 	station_gatherers_produces_mult = 0.15
	# }

	show_in_tech = "tech_ice_mining_station"

	ai_build_at_chokepoint = yes
	ai_build_outside_chokepoint = yes
	ai_weight = {
		weight = 100 #use same weight here - ai weighting is done in starbase_types.txt
		modifier = {
			factor = 0
			owner = {
				count_owned_starbase = {
					count >= 2
					limit = {
						has_starbase_building = ice_mining_station
					}
				}
			}
		}
	}
}
