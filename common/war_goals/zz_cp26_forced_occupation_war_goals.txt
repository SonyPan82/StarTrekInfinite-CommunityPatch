wg_cp26_forced_occupation = {
	casus_belli = cb_cp26_forced_occupation
	hide_if_no_cb = no
	total_war = no
	needs_total_occupation = no
	cede_claims = occupied_only
	threat_multiplier = 1.5

	allowed_peace_offers = {
		status_quo
		surrender
		demand_surrender
	}

	surrender_acceptance = -50

	potential = {
		has_total_war_cb = no
		OR = {
			is_country_type = default
			is_country_type = contender
		}
	}

	possible = {
		from = {
			is_subject = no
			OR = {
				is_country_type = default
				is_country_type = contender
			}
		}
	}

	on_wargoal_set = {
		from = {
			every_system_within_border = {
				limit = {
					NOT = {
						root = { has_claim = prev }
					}
				}
				add_claims = { who = root }
			}
		}
	}

	on_accept = {
		from = {
			set_subject_of = {
				who = root
				subject_type = occupied_state
			}
			hidden_effect = {
				every_system_within_border = {
					limit = {
						root = { has_claim = prev }
					}
					remove_claims = { who = root }
				}
			}
		}
	}

	on_status_quo = {
	}

	ai_weight = {
		weight = 1
	}
}
