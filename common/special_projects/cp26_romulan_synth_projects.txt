# Mission d'espionnage sur Earth
# Scope: THIS = Earth planet, OWNER = Romulan country
special_project = {
	key = "CP26_SOL_SPY_MISSION"
	cost = 0
	days_to_research = 30
	timelimit = 215
	tech_department = society_technology
	location = yes

	picture = GFX_evt_spy_mission_romulan
	picture = GFX_evt_orbital_research_barren_world

	event_scope = ship_event
	delete_on_finish = yes

	requirements = {
		shipclass_spy_ship = 1
	}

	on_success = {
		if = {
			limit = { exists = OWNER }
			OWNER = { country_event = { id = cp26_synth_rom.10 } }
		}
	}

	on_fail = {
		if = {
			limit = { exists = OWNER }
			OWNER = { country_event = { id = cp26_synth_rom.2 } }
		}
	}
}

# Projet Synth - rébellion synthétique à distance
# Scope: THIS = target planet, OWNER = Romulan country
special_project = {
	key = "CP26_PROJET_SYNTH"
	cost = 0
	days_to_research = 30
	timelimit = 90
	tech_department = society_technology
	location = yes

	picture = GFX_evt_spy_mission_romulan
	picture = GFX_evt_machine_space_battle

	event_scope = ship_event
	delete_on_finish = yes

	requirements = {
		shipclass_spy_ship = 1
	}

	on_success = {
		if = {
			limit = { exists = OWNER }
			create_country = {
				name = "NAME_A500_Rebels"
				type = global_event
				auto_delete = yes
				effect = {
					set_faction_hostility = {
						set_hostile = yes
						set_neutral = no
						set_friendly = no
					}
				}
			}
			last_created_country = { save_global_event_target_as = cp26_synth_rebel_country }
			THIS = {
				add_planet_devastation = 15
				add_modifier = { modifier = cp26_synth_unrest days = 720 }
				set_planet_flag = cp26_synth_rebellion_active
				while = {
					count = 3
					if = {
						limit = { exists = event_target:et_a500_species_global }
						create_army = {
							owner = event_target:cp26_synth_rebel_country
							type = rebel_army
							species = event_target:et_a500_species_global
						}
					}
					else = {
						create_army = {
							owner = event_target:cp26_synth_rebel_country
							type = rebel_army
						}
					}
				}
				if = {
					limit = { owner = { is_ufop = yes } }
					owner = { country_event = { id = cp26_synth_rom.25 } }
				}
			}
			OWNER = { country_event = { id = cp26_synth_rom.20 } }
		}
	}

	on_fail = {
		if = {
			limit = { exists = OWNER }
			OWNER = { country_event = { id = cp26_synth_rom.21 } }
		}
	}
}
