special_project = {
	key = "CP26_A500_SECURITY_RESPONSE"
	cost = 0
	days_to_research = 120
	timelimit = 365
	tech_department = engineering_technology
	picture = GFX_evt_cp26_a500_rebellion
	event_chain = "cp26_a500_rebellion_chain"
	location = yes
	event_scope = ship_event
	delete_on_finish = yes

	requirements = {
		shipclass_military = 1
	}

	on_success = {
		hidden_effect = {
			OWNER = {
				country_event = { id = cp26_a500.15 days = 1 }
			}
		}
	}

	on_fail = {
		hidden_effect = {
			OWNER = {
				set_country_flag = cp26_a500_no_response
				country_event = { id = cp26_a500.2 days = 1 }
			}
		}
	}
}
