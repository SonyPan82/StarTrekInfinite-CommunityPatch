# CP26 — BAJORAN WORMHOLE DISCOVERY PROJECT
# Enabled when a playable major (contender/default) fleet first enters the
# Bajor system (event .11), or yearly for AI Bajor (.20). Science ship +
# scientist on site; on success → STH_bajoran_story.1010 (CP26 override).

special_project = {
	key = "CP26_WORMHOLE_SURVEY"
	days_to_research = 90
	timelimit = -1
	tech_department = physics_technology
	picture = GFX_evt_first_contact_ufop_bajoran_stage3
	event_scope = ship_event

	requirements = {
		shipclass_science_ship = 1
		leader = scientist
	}

	on_success = {
		owner = { country_event = { id = STH_bajoran_story.1010 } }
	}
	on_fail = { }

	abort_trigger = {
		owner = { has_global_flag = bajoran_wormhole_discovered }
	}
}
