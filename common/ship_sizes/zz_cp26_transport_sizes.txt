# CP26 — Transport ship override
# Adds ignores_warp_range = yes so troop transports can travel beyond the default
# warp range radius, matching science ships and military ships.
# Borders are still respected — transports cannot cross closed frontiers.

transport = {
	max_speed = @speed_very_fast
	acceleration = 0.6
	rotation_speed = 0.2
	max_hitpoints = @transport_hp
	modifier = {
		ship_evasion_add = @transport_evasion
	}
	size_multiplier = 1
	fleet_slot_size = 1
	section_slots = { "mid" = { locator = "part1" } }
	num_target_locators = 0
	is_space_station = no
	icon_frame = 11
	text_icon = military_size_1
	fleet_view_background = "transport"

	combat_disengage_chance = 1.5

	enable_default_design = yes
	is_designable = no
	components_add_to_cost = no
	class = shipclass_transport
	required_component_set = "power_core"
	required_component_set = "ftl_components"
	required_component_set = "thruster_components"
	required_component_set = "sensor_components"
	auto_upgrade = yes

	ignores_warp_range = yes

	ship_modifier = {
		ship_hyperlane_range_add = 1
	}

	resources = {
		category = ships

		cost = {
			trigger = {
				from = {
					country_uses_artificial_singularities = yes
				}
			}
			artificial_singularities = 1
		}

		upkeep = {
			trigger = {
				NOT = {

					}
				}
			}
			deuterium = 1
		}

		upkeep = {
			trigger = {
				ROOT = {
					exists = FROM

				}
			}
			artificial_singularities = 1
		}

		upkeep = {
			alloys = 1
		}
	}
