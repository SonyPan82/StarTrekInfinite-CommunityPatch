# CP26 - Expansion Pressure: AI Influence budget correction
#
# Overrides only `influence_expenditure_stations` from vanilla's
# common/ai_budget/00_influence_budget.txt (key-based override - all other
# entries in that file are untouched and still apply from vanilla).
#
# Vanilla hardcodes static_min/static_max at 75/150, matching
# NEconomy.EXPANSION_COST_BASE (70) and EXPANSION_COST_MAX (150) - i.e. the
# AI's assumption of "how much Influence should I reserve before trying to
# expand" is baked to the un-modified vanilla cost range.
#
# With Expansion Pressure active, a country's real outpost cost can reach
# CP26's 200 Influence ceiling at Tier 4. Left uncorrected, the AI would
# under-reserve Influence for its expansion plan and stall or behave
# erratically once it reaches the higher tiers - this file exists
# specifically to prevent that, independent of difficulty.
#
# static_max is sized for the 200 Influence cap with modest headroom;
# static_min is nudged up so the AI starts saving a little earlier once
# it is past Tier 0, without hoarding Influence during the early game.
#
# See project wiki, page "Expansion Pressure", for the full cost table.

influence_expenditure_stations = {
	resource = influence
	type = expenditure
	category = stations

	potential = {
		has_ai_expansion_plan = yes
	}

	fraction = {
		weight = 0.3
	}

	static_min = {
		base = 90
	}

	static_max = {
		base = 250
	}
}
