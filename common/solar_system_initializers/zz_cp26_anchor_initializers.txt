cp26_xindus_anchor_initializer = {
	class = "rl_all_stars"
	flags = { outpost ufop_system ufop cp26_xindus_anchor }
	usage = static_system_init

	planet = {
		class = star
		orbit_distance = 0
		orbit_angle = 0
		size = { min = 20 max = 30 }
		has_ring = no
	}

	change_orbit = 50

	planet = {
		count = { min = 3 max = 8 }
		class = rl_unhabitable_planets
		orbit_distance = { min = 20 max = 30 }
		home_planet = no
	}

	init_effect = {
		every_neighbor_system = { set_star_flag = empire_cluster }
	}

	late_init_effect = {
		create_starbase = {
			size = starbase_outpost
			owner = event_target:ufop_country
		}
	}
}
