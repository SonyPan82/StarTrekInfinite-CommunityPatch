ferengi_player_initializer = {
	name = "NAME_Ferenginar"
	class = "rl_all_stars"
	usage = custom_empire
	flags = { empire_home_system ferengi ferengi_system ferengi_home_system capital }

	planet = {
		count = 1
		class = star
		orbit_distance = 0
		orbit_angle = 1
		size = { min = 20 max = 30 }
		has_ring = no

		init_effect = {
			change_pc = pc_k_star
			save_global_event_target_as = ferengi_primary_star
		}
	}

	change_orbit = 30

	planet = {
		count = 1
		orbit_distance = 12
		size = 4
		orbit_angle = { min = 0 max = 300 }
		class = pc_molten
		home_planet = no
	}

	planet = {
		count = 1
		orbit_distance = 10
		size = 5
		orbit_angle = { min = 0 max = 300 }
		class = pc_barren
		home_planet = no
	}

	planet = {
		name = "NAME_Ferenginar"
		count = 1
		class = pc_continental
		subclass = psc_class_m
		size = 16
		orbit_distance = 22
		starting_planet = yes
		home_planet = yes
		deposit_blockers = none
		modifiers = none
		flags = { mapo_colony ferengi_capital }

		init_effect = {
			prevent_anomaly = yes
			save_global_event_target_as = ferengi_homeworld
			every_country = {
				if = {
					limit = {
						OR = {
							has_country_flag = ferengi
							has_country_id = "ferengi"
						}
					}
					save_global_event_target_as = ferengi_country
				}
			}
			add_modifier = {
				modifier = home_planet
				days = -1
			}
		}

			late_init_effect = {
				if = {
					limit = { is_colony = no }
					set_colony_type = col_default
					clear_blockers = yes
					add_building = building_capital

					while = {
						count = 2
						add_district_and_planet_size_if_needed_effect = {
							district = district_city
						}
					}
					while = {
						count = 1
						add_district_and_planet_size_if_needed_effect = {
							district = district_generator
						}
					}
					while = {
						count = 1
						add_district_and_planet_size_if_needed_effect = {
							district = district_industrial
						}
					}
					while = {
						count = 1
						add_district_and_planet_size_if_needed_effect = {
							district = district_farming
						}
					}
					while = {
						count = 16
						create_pop = {
							species = owner_main_species
						}
					}
					if = {
						limit = { NOT = { has_building = building_energy_grid } }
						add_building = building_energy_grid
					}
					if = {
						limit = { NOT = { has_building = cp26_ferengi_energy_consortium } }
						add_building = cp26_ferengi_energy_consortium
					}
					if = {
						limit = { NOT = { has_building = cp26_ferengi_food_consortium } }
						add_building = cp26_ferengi_food_consortium
					}
				}
				if = {
					limit = { NOT = { has_building = building_capital } }
					add_building = building_capital
				}
				if = {
					limit = { NOT = { has_building = cp26_ferengi_energy_consortium } }
					add_building = cp26_ferengi_energy_consortium
				}
				if = {
					limit = { NOT = { has_building = cp26_ferengi_food_consortium } }
					add_building = cp26_ferengi_food_consortium
				}
				if = {
					limit = { has_building = cp26_ferengi_mining_consortium }
					remove_building = cp26_ferengi_mining_consortium
				}
			}

	}

	planet = {
		count = 1
		orbit_distance = 30
		size = 8
		orbit_angle = { min = 0 max = 300 }
		class = pc_arid
		subclass = psc_class_l
		home_planet = no
	}

	init_effect = {
		generate_home_system_resources = yes
		set_star_flag = ferengi_home_system
		set_star_flag = ferengi_system
		set_star_flag = ferengi
		every_neighbor_system = { set_star_flag = empire_cluster }
	}

	}

						}
					}
				}

			if = {
				limit = {
					NOT = { exists = event_target:ferengi_country }
				}
				create_country = {
					species = "Ferengi"
					country_id = "ferengi"
					type = ferengi_alliance

					name_list = "FERE"
					authority = auth_corporate

					flag = {

							category = "emblem"
							file = "flag_Ferengi_35.dds"
						}
						background = {
							category = "backgrounds"
							file = "00_solid.dds"
						}
						colors = {
							"Cardassian"
							"black"
							"null"
							"null"
						}
					}
					ethos = {
						ethic = ethic_fanatic_materialist
						ethic = ethic_xenophile
					}
					ignore_initial_colony_error = yes
				}
				last_created_country = {
					set_name = "Alliance Ferengi"
					set_graphical_culture = ferengi
					save_global_event_target_as = ferengi_country
					set_country_flag = ferengi
					set_country_flag = ferengi_alliance
					set_custom_capital_location = prev

					create_leader = {
						class = ruler
						species = this
						name = random
						skill = 3
					}
					set_leader = last_created_leader
				}
			}
		}

			late_init_effect = {
				if = {
					limit = {
						is_colony = no
						exists = event_target:ferengi_country
				}
				set_owner = event_target:ferengi_country
				set_colony_type = col_default
			}
		}
	}

	planet = {

			if = {
				limit = {
					has_country_flag = ferengi
				}
				save_global_event_target_as = ferengi_country
			}
		}
	}

	late_init_effect = {
		if = {
			limit = {
				NOT = { exists = starbase }
				exists = event_target:ferengi_country
			}
			create_starbase = {
				size = starbase_starport
				owner = event_target:ferengi_country
			}
		}
	}
