bajoran_bajor_initializer = {
	name = "NAME_Bajoran_System"
	class = sc_g
	usage = custom_empire
	flags = { bajoran bajoran_system bajoran_home_system capital }

	planet = {
		name = "NAME_Bajor_Star"
		count = 1
		class = star
		orbit_distance = 0
		orbit_angle = 1
		size = { min = 20 max = 30 }
		has_ring = no

		init_effect = {
			change_pc = pc_g_star
		}
	}

	change_orbit = 20

	planet = {
		count = 10
		orbit_distance = { min = 20 max = 25 }
		class = "rl_unhabitable_planets"
	}

	planet = {
		name = "NAME_Bajor_XI"
		orbit_distance = 35
		class = pc_continental
		subclass = psc_class_m
		size = 15
		deposit_blockers = none
		modifiers = none
		has_ring = no
		home_planet = yes
		flags = { bajor mipo_colony }

		init_effect = {
			prevent_anomaly = yes
			add_modifier = {
				modifier = home_planet
				days = -1
			}
			save_global_event_target_as = bajoran_planet
		}

		moon = {
			count = 5
			orbit_distance = 6
			class = rl_unhabitable_planets
			has_ring = no
			size = 5
		}
	}

	init_effect = {
		generate_home_system_resources = yes
		every_neighbor_system = { set_star_flag = empire_cluster }

		every_country = {
			if = {
				limit = {
					has_country_flag = bajoran
				}
				save_global_event_target_as = bajoran_country
			}
		}
		save_global_event_target_as = bajoran_system
	}

	late_init_effect = {
		create_starbase = {
			size = starbase_starport
			owner = event_target:bajoran_country
		}
		if = {
			limit = { has_star_flag = bajoran_system }
			every_country = {
				if = {
					limit = {
						OR = {
							has_country_flag = cardassia
							is_cardassian = yes
						}
						NOT = { has_country_flag = cp26_terok_nor_built }
					}
					save_event_target_as = cp26_cardassian_country_for_terok_nor
				}
			}
			every_system_planet = {
				limit = { has_planet_flag = bajor }
				save_event_target_as = cp26_bajor_planet_for_terok_nor
			}
			if = {
				limit = {
					exists = event_target:cp26_cardassian_country_for_terok_nor
					exists = event_target:cp26_bajor_planet_for_terok_nor
					NOT = { has_star_flag = cp26_terok_nor_system }
				}
				spawn_megastructure = {
					type = "cp26_terok_nor"
					planet = event_target:cp26_bajor_planet_for_terok_nor
					owner = event_target:cp26_cardassian_country_for_terok_nor
					orbit_distance = 8
					orbit_angle = 45
				}
				set_star_flag = cp26_terok_nor_system
				event_target:cp26_cardassian_country_for_terok_nor = {
					set_country_flag = cp26_terok_nor_built
				}
			}
		}
	}
}
