		potential = { exists = owner }

		potential = {
			exists = owner

			exists = owner
            owner = { has_civic = civic_systematized_structure }
		}

			planet_amenities_add = 2
		}
	}

		category = planet_buildings

			energy = @b1_upkeep
		}
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_tradition = tr_ufop_finish

			}
			officers = 2
		}
	}

		building_capital
	}

	}
