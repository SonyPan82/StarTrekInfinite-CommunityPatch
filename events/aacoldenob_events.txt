namespace = aa_coldenob

event = {
	id = aa_coldenob.1000
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		create_species = {
			class = "AADENOB"
			portrait = "denobulan"
			name = "Denobulan"
			plural = "Denobulans"
			adjective = "Denobula"
			name_list = "BETA"
			traits = {
				trait = "trait_denob_communal"
				trait = "trait_denob_polyamorous"
				trait = "trait_denob_family"
			}
			effect = {
				save_global_event_target_as = denobulan_species
			}
		}
		
		random_galaxy_planet = {
			limit = {
				has_planet_flag = planet_denobula
			}
			set_owner = event_target:ufop_country
			set_colony_type = col_default
			add_building = building_capital
		}

		event_target:ufop_country = {
			random_owned_planet = {
				limit = {
					has_planet_flag = planet_denobula
				}
				while = {
					count = 13
					create_pop = { species = event_target:denobulan_species }
				}
				add_deposit = d_bountiful_plains
				add_deposit = d_hot_springs
				add_deposit = d_geothermal_vent
				add_district_and_planet_size_if_needed_effect = {
					district = district_farming
				}
				add_district_and_planet_size_if_needed_effect = {
					district = district_generator
				}
				add_building = building_clinic
			}
		}
	}
}
