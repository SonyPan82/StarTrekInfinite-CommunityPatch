### COLONY P1E1: METAMORPHOSIS - TRIGGER
planet_event = {
	id = hrb_colony.10
	is_triggered_only = yes
	hide_window = yes

	pre_triggers = {
		has_owner = yes
		is_homeworld = no
		original_owner = yes
		is_ai = no
		has_ground_combat = no
		is_capital = no
		is_occupied_flag = no
	}

	trigger = {
		num_pops > 1
		NOT = {
			is_planet_class = pc_habitat
		}
		NOR = {
			has_planet_flag = colony_event
			has_planet_flag = bespoke_colony
			owner = { 
				has_country_flag = metamorphosis_event 
			}
		}
	}

	immediate = {
		set_variable = {
			which = killed_pops
			value = 0
		}

		set_planet_flag = colony_event
		save_event_target_as = colony_metamorphosis
		owner = { 
			set_country_flag = metamorphosis_event 
		}
		planet_event = { id = hrb_colony.11 }
	}
}

### COLONY P1E1: METAMORPHOSIS - KILL ALL POPS
planet_event = {
	id = hrb_colony.11
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		if = {
			limit = {
				num_pops > 1
			}
			random_owned_pop = {
				kill_pop = yes
			}
			owner = {
				change_variable = {
					which = "killed_pops"
					value = 1
				}
			}
			planet_event = { id = hrb_colony.11 }
		}
		else = {
			planet_event = { id = hrb_colony.12 }
		}
	}
}

	} 
	picture = {

		picture = GFX_evt_hologram_automaton_rocky_planet_klingon
	}
	picture = {
		trigger = { is_romulan = yes } 
		picture = GFX_evt_hologram_automaton_rocky_planet_romulan
	} 
	picture = {
		trigger = { is_cardassian = yes }
		picture = GFX_evt_hologram_automaton_rocky_planet_cardassian
	}

	# Default image on fail
	picture = GFX_evt_archaeology_excavation

	### All: Study the automaton's programs.
	option = {

		name = hrb_colony.83.a
		custom_tooltip = hrb_colony.83.a.tt
		
		# CP26 fix : the upstream special_project AQUEDUCT_LIFEFORM only saves
		# event_target:aqueduct_scientist when the science ship has an assigned
		# leader. If that capture failed, accessing the target raised an
		# Invalid context switch warning. Gate the trait reward on existence.
		if = {
			limit = { exists = event_target:aqueduct_scientist }
			event_target:aqueduct_scientist = {
				add_trait = "leader_trait_pathologist"
				add_experience = 200
			}
		}
	}

	### All: Crush the beads, and leave.
	option = {

		name = hrb_colony.83.b
		custom_tooltip = hrb_colony.83.b.tt
		
		add_monthly_resource_mult = { 
			resource = unity 
			value = @tier2unityreward 
			min = @tier2unitymin 
			max = @tier2unitymax 
		}
		
		if = {
			limit = { 
				has_technology = tech_frontier_hospital 
			}

			add_monthly_resource_mult = { 
				resource = research 
				value = @tier2researchreward 
				min = @tier2researchmin 
				max = @tier2researchmax 
			}
		}
		else_if = {
			limit = { 
				has_technology = tech_frontier_health 
			}

			add_tech_progress = {
				tech = tech_frontier_hospital
				progress = 0.25
			}
		}
		else = {
			add_tech_progress = {
				tech = tech_frontier_health
				progress = 0.25
			}
		}
	}

	### All: Replenish the aqueduct.

		name = hrb_colony.83.c
		custom_tooltip = hrb_colony.83.c.tt
		
		allow = {
			resource_stockpile_compare = {
				resource = energy

		event_target:colony_stella_maris = {
			add_modifier = { 
				modifier = stella_maris_revitalized_ecosystem 
			}
		}
	}
