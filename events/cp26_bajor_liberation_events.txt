namespace = cp26_bajor

##############################################################
# Liberation de Bajor - 2369
#
# Mecanique :
#   - 1 seul popup par empire avec les vrais choix
#   - Immediate : ouvre frontieres Bajor + Cardassia (forced=180)
#     pour permettre aux flottes de traverser l'espace cardassien
#   - Options militaires : set cp26_bajor_fleet_needed
#     → Journal de bord affiche l'objectif "envoyer une flotte"
#     → special_project CP26_BAJOR_LIBERATION_* surveille l'orbite de Bajor
#     → Quand flotte detectee → cp26_bajor.10 (liberation)
#   - Options non-militaires : timer immediat → cp26_bajor.20
#   - cp26_bajor.20 referme les frontieres forcees
##############################################################

# ============================================================
# FEDERATION
# ============================================================
country_event = {
	id = cp26_bajor.1
	title = "cp26_bajor.title"
	desc = "cp26_bajor.1.desc"
	picture = GFX_evt_empire_fleet_ufop
	show_sound = event_empire_fleet

	mean_time_to_happen = { days = 1 }

	trigger = {
		years_passed >= 23
		NOT = { has_global_flag = cp26_block_bajor_arc }
		NOT = { has_country_flag = cp26_bajor_event_fired }
		is_ufop = yes
	}

	immediate = {
		cp26_bajor_bootstrap_targets_effect = yes
		set_country_flag = cp26_bajor_event_fired
		set_global_flag = cp26_bajor_2369_active
		# Ouvrir frontieres Bajor - forced=180 jours pour que ca tienne
		if = {
			limit = { exists = event_target:bajoran_country }
			establish_communications_no_message = event_target:bajoran_country
			set_closed_borders = { who = event_target:bajoran_country status = no forced = 180 }
			event_target:bajoran_country = {
				establish_communications_no_message = root
				set_closed_borders = { who = root status = no forced = 180 }
			}
		}
		# Ouvrir frontieres Cardassia pour traverser leur espace jusqu'a Bajor
		if = {
			limit = { exists = event_target:cardassian_country }
			establish_communications_no_message = event_target:cardassian_country
			set_closed_borders = { who = event_target:cardassian_country status = no forced = 180 }
			event_target:cardassian_country = {
				establish_communications_no_message = root
				set_closed_borders = { who = root status = no forced = 180 }
			}
		}
		cp26_bajor_start_rebellion_setup = yes
	}

	# Non-intervention
	option = {
		name = "cp26_bajor.1.a"
		add_resource = { influence = 100 }
		if = {
			limit = { exists = event_target:cardassian_country }
			event_target:cardassian_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_humanitarian_aid }
			}
		}
		hidden_effect = {
			country_event = { id = cp26_bajor.20 days = 1 }
		}
	}

	# Intervention militaire
	option = {
		name = "cp26_bajor.1.b"
		add_resource = { influence = 200 alloys = -500 }
		add_resource = { unity = 500 }
		if = {
			limit = { exists = event_target:cardassian_country }
			event_target:cardassian_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_opposed_liberation }
			}
		}
		hidden_effect = {
			set_country_flag = cp26_bajor_fleet_military
			set_country_flag = cp26_bajor_opt_fed_mil
			enable_special_project = {
				name = "CP26_BAJOR_LIBERATION_MIL"
				location = event_target:bajoran_system
				owner = ROOT
			}
		}
		custom_tooltip = cp26_bajor.1.b.tt
	}

	# Aide humanitaire
	option = {
		name = "cp26_bajor.1.c"
		add_resource = { influence = 50 food = -1000 }
		add_resource = { unity = 150 }
		if = {
			limit = { exists = event_target:bajoran_country }
			event_target:bajoran_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_humanitarian_aid }
			}
		}
		if = {
			limit = { exists = event_target:cardassian_country }
			event_target:cardassian_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_opposed_liberation }
			}
		}
		hidden_effect = {
			set_country_flag = cp26_bajor_fleet_needed
			set_country_flag = cp26_bajor_opt_fed_hum
			enable_special_project = {
				name = "CP26_BAJOR_LIBERATION_ANY"
				location = event_target:bajoran_system
				owner = ROOT
			}
		}
		custom_tooltip = cp26_bajor.1.c.tt
	}
}

# REVOLTE BAJORANNE - si Cardassia n'envoie pas de troupes a temps
# Fires 30 jours apres la decision 2.a/2.b si aucun transport
# n'est arrive avec des armees embarquees
# ============================================================
country_event = {
	id = cp26_bajor.52
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_bajor_troops_suppressed }
		NOT = { has_global_flag = cp26_bajor_liberated }
		exists = event_target:bajoran_planet
	}

	immediate = {
		cp26_bajor_bootstrap_targets_effect = yes

		set_global_flag = cp26_bajor_liberated

		# Tuer tous les pops cardassiens sur Bajor
		every_owned_pop = {
			limit = { planet = { is_same_value = event_target:bajoran_planet } }
			kill_pop = yes
		}

		event_target:bajoran_planet = {
			add_planet_devastation = 100
			if = {
				limit = { exists = event_target:bajoran_country }
				set_owner = event_target:bajoran_country
				event_target:bajoran_country = {
					add_opinion_modifier = { who = root modifier = cp26_bajor_opposed_liberation }
				}
			}
		}
	}
}

country_event = {
	id = cp26_bajor.10
	title = "cp26_bajor.10.name"
	desc = "cp26_bajor.10.desc"
	picture = GFX_evt_civil_unrest_cardassian
	show_sound = event_empire_fleet
	is_triggered_only = yes

	immediate = {
		cp26_bajor_bootstrap_targets_effect = yes

		if = {
			limit = {
				NOT = { has_global_flag = cp26_bajor_liberated }
				exists = event_target:bajoran_country
			}
			set_global_flag = cp26_bajor_liberated
			remove_global_flag = cp26_bajor_rebellion_active
			if = {
				limit = { exists = event_target:bajoran_planet }
				event_target:bajoran_planet = { remove_planet_flag = cp26_bajor_in_rebellion }
			}

			# Liberer Bajor de l'occupation cardassienne
			event_target:bajoran_country = {
				set_subject_of = { who = none }
			}

			# Frontieres permanentes avec l'empire liberateur
			if = {
				limit = { NOT = { is_cardassian = yes } }
				set_closed_borders = { who = event_target:bajoran_country status = no forced = 5 }
				event_target:bajoran_country = {
					set_closed_borders = { who = root status = no forced = 5 }
					add_opinion_modifier = { who = root modifier = cp26_bajor_supported_liberation }
				}
			}

			# ---- Bonuses par choix ----

			# Federation militaire
			if = {
				limit = {
					has_country_flag = cp26_bajor_opt_fed_mil
					exists = event_target:bajoran_planet
				}
				event_target:bajoran_planet = {
					add_modifier = { modifier = "cp26_bajor_fed_liberation" days = 1800 }
				}
			}

			# Federation humanitaire
			if = {
				limit = {
					has_country_flag = cp26_bajor_opt_fed_hum
					exists = event_target:bajoran_planet
				}
				event_target:bajoran_planet = {
					add_modifier = { modifier = "cp26_bajor_humanitarian_relief" days = 1800 }
				}
			}

			# Klingon
			if = {
				limit = { has_country_flag = cp26_bajor_opt_kling }
				if = {
					limit = { exists = event_target:bajoran_planet }
					event_target:bajoran_planet = {
						add_modifier = { modifier = "cp26_bajor_klingon_honor" days = 720 }
					}
				}
				event_target:bajoran_country = {
					add_opinion_modifier = { who = root modifier = cp26_bajor_supported_liberation }
				}
			}

			# Romulien
			if = {
				limit = {
					has_country_flag = cp26_bajor_opt_rom
					exists = event_target:bajoran_planet
				}
				event_target:bajoran_planet = {
					add_modifier = { modifier = "cp26_bajor_romulan_influence" days = 1800 }
				}
			}

			# Ferengi (vente aux Bajorans)
			if = {
				limit = {
					has_country_flag = cp26_bajor_opt_feren
					exists = event_target:bajoran_planet
				}
				event_target:bajoran_planet = {
					add_modifier = { modifier = "cp26_bajor_ferengi_trade" days = 1095 }
				}
			}

			# Ferengi (vente aux deux camps) - devastation partielle, profit quand meme
			if = {
				limit = {
					has_country_flag = cp26_bajor_opt_feren_both
					exists = event_target:bajoran_planet
				}
				event_target:bajoran_planet = {
					add_planet_devastation = 30
					add_modifier = { modifier = "cp26_bajor_ferengi_trade" days = 1095 }
				}
			}

# Résolution aléatoire si aucune intervention (MTTH 500j)
planet_event = {
	id = cp26_bajor.32
	hide_window = yes

	trigger = {
		has_planet_flag = cp26_bajor_in_rebellion
		has_global_flag = cp26_bajor_rebellion_active
		NOT = { has_global_flag = cp26_bajor_liberated }
		NOT = { any_country = { has_country_flag = cp26_bajor_fleet_military } }
		NOT = { any_country = { has_country_flag = cp26_bajor_fleet_needed } }
		NOT = { any_country = { has_country_flag = cp26_bajor_troops_needed } }
		NOT = { any_country = { has_country_flag = cp26_bajor_cardassia_delivery } }
		NOT = { any_country = { has_country_flag = cp26_bajor_feren_both } }
	}

	mean_time_to_happen = { days = 500 }

	immediate = {
		random_list = {
			# 60% — la résistance remporte la victoire
			60 = {
				remove_planet_flag = cp26_bajor_in_rebellion
				remove_global_flag = cp26_bajor_rebellion_active
				set_global_flag = cp26_bajor_liberated
				every_owned_pop = {
					limit = { NOT = { is_same_species = owner } }
					kill_pop = yes
				}
				owner = { set_subject_of = { who = none } }
				every_country = {
					limit = {
						has_communications = owner
						NOT = { is_same_value = owner }
					}
					country_event = { id = cp26_bajor.33 }
				}
			}
			# 40% — Cardassia écrase la rébellion
			40 = {
				remove_planet_flag = cp26_bajor_in_rebellion
				remove_global_flag = cp26_bajor_rebellion_active
				add_modifier = { modifier = "cp26_bajor_suppressed_garrison" days = 1800 }
				every_country = {
					limit = {
						has_communications = owner
						NOT = { is_same_value = owner }
					}
					country_event = { id = cp26_bajor.34 }
				}
			}
		}
	}
}
