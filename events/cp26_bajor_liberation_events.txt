# Liberation de Bajor - 2369
#
# Mecanique :
#   - 1 seul popup par empire avec les vrais choix
#   - Immediate : ouvre frontieres Bajor + Cardassia (forced=180)
#     pour permettre aux flottes de traverser l'espace cardassien
#   - Options militaires : set cp26_bajor_fleet_needed
#     → Journal de bord affiche l'objectif "envoyer une flotte"
#     → cp26_bajor.50 surveille l'orbite de Bajor en continu
#     → Quand flotte detectee → cp26_bajor.10 (liberation)

# ============================================================
# FEDERATION
# ============================================================
country_event = {
	id = cp26_bajor.1
	title = "cp26_bajor.title"
	desc = "cp26_bajor.1.desc"
	picture = GFX_evt_war_ufop
	show_sound = event_war_news

	mean_time_to_happen = { days = 1 }

	trigger = {
		NOT = { has_global_flag = cp26_block_bajor_arc }
		NOT = { has_country_flag = cp26_bajor_event_fired }
		is_ufop = yes
		date > 2369.1.1
		NOT = { date > 2370.1.1 }
	}

	immediate = {
		set_country_flag = cp26_bajor_event_fired
		set_global_flag = cp26_bajor_2369_active
		# Ouvrir frontieres Bajor - forced=180 jours pour que ca tienne
		if = {
			limit = { exists = event_target:bajoran_country }
			establish_communications_no_message = event_target:bajoran_country
			set_closed_borders = { who = event_target:bajoran_country status = no forced = 180 }
			event_target:bajoran_country = {
				establish_communications_no_message = root
				set_closed_borders = { who = root status = no forced = 180 }
			}
		}
		# Ouvrir frontieres Cardassia pour traverser leur espace jusqu'a Bajor
		if = {
			limit = { exists = event_target:cardassian_country }
			establish_communications_no_message = event_target:cardassian_country
			set_closed_borders = { who = event_target:cardassian_country status = no forced = 180 }
			event_target:cardassian_country = {
				establish_communications_no_message = root
				set_closed_borders = { who = root status = no forced = 180 }
			}
		}
		begin_event_chain = {
			event_chain = "cp26_bajor_arc_chain"
			target = THIS
		}
	}

	# Non-intervention
	option = {
		name = "cp26_bajor.1.a"
		add_resource = { influence = 100 }
		if = {
			limit = { exists = event_target:cardassian_country }
			event_target:cardassian_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_humanitarian_aid }
			}
		}
		hidden_effect = {
			country_event = { id = cp26_bajor.20 days = 1 }
		}
	}

	# Intervention militaire
	option = {
		name = "cp26_bajor.1.b"
		add_resource = { influence = 200 alloys = -1500 }
		add_resource = { unity = 500 }
		if = {
			limit = { exists = event_target:cardassian_country }
			event_target:cardassian_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_opposed_liberation }
			}
		}
		hidden_effect = {
			set_country_flag = cp26_bajor_fleet_military
			set_country_flag = cp26_bajor_opt_fed_mil
		}
		custom_tooltip = cp26_bajor.1.b.tt
	}

	# Aide humanitaire
	option = {
		name = "cp26_bajor.1.c"
		add_resource = { influence = 50 food = -1000 }
		add_resource = { unity = 150 }
		if = {
			limit = { exists = event_target:bajoran_country }
			event_target:bajoran_country = {
				add_opinion_modifier = { who = root modifier = cp26_bajor_humanitarian_aid }
			}
		}
		hidden_effect = {
			set_country_flag = cp26_bajor_fleet_needed
			set_country_flag = cp26_bajor_opt_fed_hum
		}
		custom_tooltip = cp26_bajor.1.c.tt
	}
}

# ============================================================
# DETECTION VAISSEAU DANS LE SYSTEME DE BAJOR
# cp26_bajor_fleet_needed  → n'importe quel vaisseau (convoi, trade)
# cp26_bajor_fleet_military → idem detection mais texte journal different
# ============================================================
country_event = {
	id = cp26_bajor.50
	hide_window = yes

	trigger = {
		OR = {
			has_country_flag = cp26_bajor_fleet_needed
			has_country_flag = cp26_bajor_fleet_military
		}
		NOT = { has_country_flag = cp26_bajor_fleet_arrived }
		exists = event_target:bajoran_system
		event_target:bajoran_system = {
			any_fleet_in_system = {
				owner = { is_same_value = root }
			}
		}
	}

	mean_time_to_happen = { days = 30 }

	immediate = {
		set_country_flag = cp26_bajor_fleet_arrived
		country_event = { id = cp26_bajor.10 days = 1 }
	}
}

# ============================================================
# DETECTION TRANSPORT DE TROUPES A BAJOR (Cardassiens)
# cp26_bajor_troops_needed → fleet avec armees embarquees
# Resultat : fermeture arc + modificateur garrison
# ============================================================
country_event = {
	id = cp26_bajor.51
	hide_window = yes

	trigger = {
		has_country_flag = cp26_bajor_troops_needed
		NOT = { has_country_flag = cp26_bajor_fleet_arrived }
		exists = event_target:bajoran_system
		event_target:bajoran_system = {
			any_fleet_in_system = {
				owner = { is_same_value = root }
				num_armies >= 3
			}
		}
	}

	mean_time_to_happen = { days = 30 }

	immediate = {
		set_country_flag = cp26_bajor_fleet_arrived
		set_country_flag = cp26_bajor_troops_suppressed
		# Appliquer le modificateur selon le choix fait
		if = {
			limit = {
				has_country_flag = cp26_bajor_opt_crush
				exists = event_target:bajoran_planet
			}
			event_target:bajoran_planet = {
				add_modifier = { modifier = "cp26_bajor_suppressed_garrison" days = 1800 }
			}
		}
		country_event = { id = cp26_bajor.20 days = 1 }
	}
}

# ============================================================
# REVOLTE BAJORANNE - si Cardassia n'envoie pas de troupes a temps
# Fires 30 jours apres la decision 2.a/2.b si aucun transport
# n'est arrive avec des armees embarquees
# ============================================================
country_event = {
	id = cp26_bajor.52
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_bajor_troops_suppressed }
		NOT = { has_global_flag = cp26_bajor_liberated }
		exists = event_target:bajoran_planet
	}

	immediate = {
		end_event_chain = cp26_bajor_arc_chain
		set_global_flag = cp26_bajor_liberated

		# Tuer tous les pops cardassiens sur Bajor
		every_owned_pop = {
			limit = { planet = { is_same_value = event_target:bajoran_planet } }
			kill_pop = yes
		}

		event_target:bajoran_planet = {
			set_planet_devastation = 100
			if = {
				limit = { exists = event_target:bajoran_country }
				set_owner = event_target:bajoran_country
				event_target:bajoran_country = {
					add_opinion_modifier = { who = root modifier = cp26_bajor_opposed_liberation }
				}
			}
		}
	}
}

# BAJOR LIBERE - resolution propre (autres empires)
# ============================================================
country_event = {
	id = cp26_bajor.10
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		end_event_chain = cp26_bajor_arc_chain
		if = {
			limit = {
				NOT = { has_global_flag = cp26_bajor_liberated }
				exists = event_target:bajoran_planet
				exists = event_target:bajoran_country
			}
			set_global_flag = cp26_bajor_liberated

			event_target:bajoran_planet = {
				if = {
					limit = {
						exists = owner
						owner = { is_cardassian = yes }
					}
					set_owner = event_target:bajoran_country
				}
			}

			# Frontieres permanentes avec l'empire liberateur
			if = {
				limit = { NOT = { is_cardassian = yes } }
				set_closed_borders = { who = event_target:bajoran_country status = no forced = 5 }
				event_target:bajoran_country = {
					set_closed_borders = { who = root status = no forced = 5 }
					add_opinion_modifier = { who = root modifier = cp26_bajor_supported_liberation }
				}
			}

			# ---- Bonuses par choix ----

			# Federation militaire
			if = {
				limit = { has_country_flag = cp26_bajor_opt_fed_mil }
				event_target:bajoran_planet = {
					add_modifier = { modifier = "cp26_bajor_fed_liberation" days = 1800 }
				}
			}

					add_modifier = { modifier = "cp26_bajor_ferengi_trade" days = 1095 }

			}
