namespace = cp26_minor_mission

country_event = {
	id = cp26_minor_mission.43
	title = "cp26_minor_mission.43.name"
	desc = "cp26_minor_mission.43.desc"
	show_sound = event_generic_success
	location = FROM

	is_triggered_only = yes

	picture = GFX_evt_lab_research_ufop

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					exists = event_target:minor_state_event_country
				}
				finished_major_minor_mission = {
					minor_country = event_target:minor_state_event_country
					finished_successfully = yes
				}
				if = {
					limit = {
						exists = event_target:minor_state_event_planet
					}
					event_target:minor_state_event_planet = {
						add_modifier = {
							modifier = cp26_minor_outbreak_contained
							days = 720
						}
					}
				}
			}
		}
	}

	option = {
		name = cp26_minor_mission.43.a
		custom_tooltip = cp26_minor_mission.43.a.tt
	}
}
