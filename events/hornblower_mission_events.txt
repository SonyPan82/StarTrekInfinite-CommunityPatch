# Check Fleet Spawn Conditions
# If yes = trigger mission.13
# If no = loop itself (mission.12)
country_event = {
	id = mission.12
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		pick_random_system_for_fleet_rescue_mission_effect = {
			DISTANCE = 50
		}

		if = {
			limit = { NOT = { exists = event_target:minor_state_event_system } }

			pick_random_system_for_fleet_rescue_mission_effect = {
				DISTANCE = 100
			}
		}

		if = {
			limit = { NOT = { exists = event_target:minor_state_event_system } }

			pick_random_system_for_fleet_rescue_mission_effect = {
				DISTANCE = 150
			}
		}

		if = {
			limit = { NOT = { exists = event_target:minor_state_event_system } }

			pick_random_system_for_fleet_rescue_mission_effect = {
				DISTANCE = 200
			}
		}

		# Fallback
		if = {
			limit = { NOT = { exists = event_target:minor_state_event_system } }

			pick_random_system_for_fleet_rescue_mission_effect = {
				DISTANCE = 500
			}
		}

		# Mission not possible right now, pick another one
		if = {
			limit = { NOT = { exists = event_target:minor_state_event_system } }
			country_event = { id = mission.02 }
			set_country_flag = vessel_adrift_not_triggered
		}
		# All good
		else = {
			country_event = { id = mission.13 } #Triggers Spawn Disabled Minor State Fleet
			remove_all_mipo_mission_flags = yes
		}
	}	
}

# Spawn Disabled Minor State Fleet
country_event = {
	id = mission.13
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		event_target:minor_state_event_system = {
			# Can only spawn a fleet within a planet scope
			random_system_planet = { 
				save_event_target_as = minor_state_event_fleet_location
			}
		}

		create_fleet = {		
			effect = {
				set_owner = event_target:minor_state_event_country
				
				create_ship = {
					design = "NAME_C2"
					effect = {
						set_disabled = yes
					}
				}
				
				set_location = {
					target = event_target:minor_state_event_fleet_location
					distance = 140
					angle = random
				}
				
				set_fleet_stance = passive
			}

			settings = {
				can_upgrade = no
				can_disband = no
				can_change_composition = no
				can_change_leader = no
				spawn_debris = yes
			}
		}

		last_created_fleet = {
			save_event_target_as = minor_state_event_fleet

			create_ambient_object = {
				type = "ship_spawn_object"
				location = THIS
				use_3d_location = yes
				duration = 1
			}
		}

		hidden_effect = {
			country_event = { id = mission.14 } #Triggers Mission Event Chain & Special Project
		}
	}
}

# Pick Starbase for the Event
# If yes = trigger mission.23
# If no = loop itself (mission.22)
country_event = {
	id = mission.22
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		random_system_within_warp_area = {
			limit = {
				exists = owner
				owner = {
					is_same_value = event_target:minor_state_event_country
				}
				has_access_fleet = this
				has_access_fleet = event_target:minor_state_event_country
				NOT = {
					any_fleet_in_system = {
						owner = { 
							is_hostile = this
							is_hostile = event_target:minor_state_event_country
						}
					}
				}
			}	
			set_star_flag = starbase_to_repair_system
			save_event_target_as = minor_state_event_system
		}

		if = {
			limit = { exists = event_target:minor_state_event_system }
			country_event = { id = mission.23 } #Triggers Starbase Disableing
			event_target:minor_state_event_country = {
				set_closed_borders = {
					who = event_target:major_state_country
					status = no
					forced = 1
				}
			}
			remove_all_mipo_mission_flags = yes
		}
		else = {
			country_event = { id = mission.02 days = 1 } #Loops Check Starbase Conditions
			set_country_flag = disaster_relief_not_triggered
		}
	}	
}
