# CP26 — Fix vanilla colony events (2 scope errors)
# ============================================================
#
# Two vanilla events in `events/hornblower_colony_events.txt` produce
# script errors in the live error.log of every long playthrough :
#
# 1. hrb_colony.75  (line 2978 vanilla, option a)
#      Effect Type: country_event
#      Scope: planet
#      → "attempted to execute an effect on an unsupported scope"
#    The option enters scope `planet` via `event_target:colony_ruins_uncovered`,
#    then tries to fire a `country_event = { ... }` from within. `country_event`
#    expects a `country` scope, so the engine logs the unsupported-scope error
#    and the follow-up event (`hrb_colony.75001`, the 10-year delayed return
#    visit) is silently dropped.
#
# 2. hrb_colony.83  (line 3484 vanilla, option a)
#      Undefined event target: aqueduct_scientist
#      → "Invalid context switch [event_target:aqueduct_scientist] from <country>"
#    The option does `event_target:aqueduct_scientist = { add_trait ... }`, but
#    the target is only saved by the upstream special_project AQUEDUCT_LIFEFORM
#    when the science ship has an assigned leader. If the leader is missing
#    (USS Olympia case in error.log), the target is never saved and this option
#    fails with an undefined-target error. The pathologist bonus is silently
#    dropped.
#
# Both are vanilla bugs. We redefine the two events here so the engine's
# singleobject database keeps OUR fixed versions (zz_* loads after vanilla).

# --- Event 1 : hrb_colony.75 — Planet of Ruins / Bomb Fields ---------------
country_event = {
	id = hrb_colony.75
	title = "hrb_colony.75.name"
	desc = "hrb_colony.75.desc"
	picture = GFX_evt_facility_capsules_androids_ai_alert
	show_sound = event_facility_capsules_androids_ai_alert
	location = ROOT
	trackable = yes
	is_triggered_only = yes

	immediate = {
		set_country_flag = terminal_terminal_flag
	}

	### All: Uh-oh.
	option = {

		name = hrb_colony.75.a
		custom_tooltip = hrb_colony.75.a.tt

		event_target:colony_ruins_uncovered = {
			remove_deposit = f_planet_of_ruins
			add_deposit = f_bomb_fields
		}

		# CP26 fix : country_event needs a country scope; the original
		# fired it from inside the planet block which produced an
		# unsupported-scope warning and the delayed return event was lost.
		# We now fire it from the option's country scope (root) explicitly.
		hidden_effect = {
			country_event = { id = hrb_colony.75001 days = 3600 }
		}
	}
}

# --- Event 2 : hrb_colony.83 — Stella Maris Automaton ---------------------
country_event = {
	id = hrb_colony.83
	title = "hrb_colony.83.name"
	desc = "hrb_colony.83.desc"
	show_sound = event_hologram_automaton_rocky_planet
	location = ROOT
	trackable = yes
	is_triggered_only = yes

	picture = {
		trigger = { is_ufop = yes }
		picture = GFX_evt_hologram_automaton_rocky_planet_ufop
	}
	picture = {
		trigger = { is_klingon = yes }
		picture = GFX_evt_hologram_automaton_rocky_planet_klingon
	}
	picture = {
		trigger = { is_romulan = yes }
		picture = GFX_evt_hologram_automaton_rocky_planet_romulan
	}
	picture = {
		trigger = { is_cardassian = yes }
		picture = GFX_evt_hologram_automaton_rocky_planet_cardassian
	}

	picture = GFX_evt_archaeology_excavation

	### All: Study the automaton's programs.
	option = {

		name = hrb_colony.83.a
		custom_tooltip = hrb_colony.83.a.tt

		# CP26 fix : the upstream special_project AQUEDUCT_LIFEFORM only
		# saves event_target:aqueduct_scientist when the science ship has
		# an assigned leader. If that step failed (no leader, reassigned
		# scientist, etc.) the target is undefined and trying to access
		# it logged an Invalid context switch. We now gate the trait
		# reward on the target's existence so the option still works
		# even when the engine couldn't capture the scientist earlier.
		if = {
			limit = { exists = event_target:aqueduct_scientist }
			event_target:aqueduct_scientist = {
				add_trait = "leader_trait_pathologist"
				add_experience = 200
			}
		}
	}

	### All: Crush the beads, and leave.
	option = {

		name = hrb_colony.83.b
		custom_tooltip = hrb_colony.83.b.tt

		add_monthly_resource_mult = {
			resource = unity
			value = @tier2unityreward
			min = @tier2unitymin
			max = @tier2unitymax
		}

		if = {
			limit = {
				has_technology = tech_frontier_hospital
			}

			add_monthly_resource_mult = {
				resource = research
				value = @tier2researchreward
				min = @tier2researchmin
				max = @tier2researchmax
			}
		}
		else_if = {
			limit = {
				has_technology = tech_frontier_health
			}

			add_tech_progress = {
				tech = tech_frontier_hospital
				progress = 0.25
			}
		}
		else = {
			add_tech_progress = {
				tech = tech_frontier_health
				progress = 0.25
			}
		}
	}

	### All: Replenish the aqueduct.
	option = {

		name = hrb_colony.83.c
		custom_tooltip = hrb_colony.83.c.tt

		allow = {
			resource_stockpile_compare = {
				resource = energy
				value >= 1000
			}
		}

		add_resource = { energy = -1000 }

		event_target:colony_stella_maris = {
			add_modifier = {
				modifier = stella_maris_revitalized_ecosystem
			}
		}
	}
}
