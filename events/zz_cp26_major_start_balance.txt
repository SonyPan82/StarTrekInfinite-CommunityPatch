namespace = cp26_start_balance

# CP26 - équilibrage départ (2346.01.01, même tick que game_start).
# Tampons stockpile + petits ajustements ciblés ; les Romuliens restent la
# référence « haute » (pas de buff). Féd : districts générateurs renforcés sur
# les mondes fondateurs (Terre:6, Vulcan:4, Andoria:4, Tellar:5) + stations minières.
# Admin UFP : charte sectorielle inchangée.

country_event = {
	id = cp26_start_balance.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# United Federation of Planets - exploite les dépôts existants + réseau énergie.
		if = {
			limit = { has_country_id = ufop }
				add_resource = {
					energy = 5000
					minerals = 900
					food = 400
				}
				add_modifier = { modifier = cp26_ufop_sector_governance_charter }
				add_modifier = {
					modifier = cp26_ufop_federal_energy_grid
					days = 5475
				}
				add_modifier = {
					modifier = cp26_ufop_federal_power_bootstrap
					days = 2190
				}
			add_modifier = {
				modifier = cp26_ufop_early_unity_balance
			}
			country_event = { id = cp26_start_balance.2 }
			country_event = { id = cp26_start_balance.3 }
		}

		# Klingon Empire - cushion IA sobre pour relancer l'expansion sans gonfler le joueur.
		if = {
			limit = { has_country_id = klingon }
			add_resource = {
				energy = 500
				minerals = 800
				food = 200
			}
			if = {
				limit = { is_ai = yes }
				add_resource = {
					alloys = 150
					influence = 50
				}
			}
		}

		# Romulan Star Empire - capture : déjà le plus confortable (900 énergie +9, etc.)
		# Aucun add_resource volontaire.

		# Cardassian Union - vrai sous-seed d'expansion, surtout pour l'IA.
		if = {
			limit = { has_country_id = cardassian }
			add_resource = {
				energy = 500
				minerals = 200
				food = 450
			}
			if = {
				limit = { is_ai = yes }
				add_resource = {
					alloys = 200
					influence = 75
				}
			}
		}

		# Ferengi Alliance - commerce OK, mais pas assez d'alloys pour réellement pousser l'expansion.
		if = {
			limit = { has_country_id = ferengi }
			add_resource = {
				energy = 400
				minerals = 900
				influence = 75
			}
			if = {
				limit = { is_ai = yes }
				add_resource = {
					alloys = 200
				}
			}
		}
	}
}

country_event = {
	id = cp26_start_balance.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ufop = yes
		NOT = { has_country_flag = cp26_ufop_starting_generator_grid_rebalanced }
	}

	immediate = {
		set_country_flag = cp26_ufop_starting_generator_grid_rebalanced

		random_owned_planet = {
			limit = { has_planet_flag = planet_earth }
			if = {
				limit = {
					NOR = {
						has_building = building_energy_grid
						has_building = building_energy_nexus
					}
				}
				add_building = building_energy_grid
			}
			while = {
				count = 6
				add_district_and_planet_size_if_needed_effect = { district = district_generator }
			}
		}

		random_owned_planet = {
			limit = { has_planet_flag = planet_vulcan }
			if = {
				limit = {
					NOR = {
						has_building = building_energy_grid
						has_building = building_energy_nexus
					}
				}
				add_building = building_energy_grid
			}
			while = {
				count = 4
				add_district_and_planet_size_if_needed_effect = { district = district_generator }
			}
		}

		random_owned_planet = {
			limit = { has_planet_flag = planet_andoria }
			if = {
				limit = {
					NOR = {
						has_building = building_energy_grid
						has_building = building_energy_nexus
					}
				}
				add_building = building_energy_grid
			}
			while = {
				count = 4
				add_district_and_planet_size_if_needed_effect = { district = district_generator }
			}
		}

		random_owned_planet = {

			if = {

					NOR = {
						has_building = building_energy_grid
						has_building = building_energy_nexus

			}
			while = {
				count = 5
				add_district_and_planet_size_if_needed_effect = { district = district_generator }
			}
		}

	}
}

country_event = {
	id = cp26_start_balance.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ufop = yes
		NOT = { has_country_flag = cp26_ufop_starting_mining_grid_repaired }
	}

	immediate = {
		set_country_flag = cp26_ufop_starting_mining_grid_repaired

		every_planet_within_border = {
			limit = {
				has_deposit_for = shipclass_mining_station
				has_mining_station = no
			}
			create_mining_station = {
				owner = root
			}
		}

		event_target:ufop_homeworld.solar_system = {
			random_system_planet = {
				limit = {
					has_mining_station = yes
					OR = {
						has_deposit = d_minerals_5
						has_deposit = d_minerals_3
						has_deposit = d_minerals_2
					}
				}
				random_mining_station = { dismantle = yes }
				remove_planet_flag = starting_mining_target
				remove_planet_flag = starting_deposit
			}
		}
	}
}
