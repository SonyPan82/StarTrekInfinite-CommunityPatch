namespace = cp26_start_balance

# CP26 - équilibrage départ (2346.01.01, même tick que game_start).
# Tampons stockpile + petits ajustements ciblés ; les Romuliens restent la
# référence « haute » (pas de buff). Féd : déficit énergie mensuel traité par
# des stations minières sur les dépôts déjà présents + un petit filet temporaire.
# Admin UFP : charte sectorielle inchangée.

country_event = {
	id = cp26_start_balance.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# United Federation of Planets - exploite les dépôts existants + léger filet énergie.
		if = {
			limit = { has_country_id = ufop }
			add_resource = {
				energy = 5000
				minerals = 900
				food = 400
			}
			add_modifier = { modifier = cp26_ufop_sector_governance_charter }
			add_modifier = {
				modifier = cp26_ufop_federal_power_bootstrap
				days = 2190
			}
			add_modifier = {
				modifier = cp26_ufop_early_unity_balance
			}
			country_event = { id = cp26_start_balance.2 }
		}

		# Klingon Empire - capture : 300 énergie +11, 100 minéraux ~0, 200 nourriture
		if = {
			limit = { has_country_id = klingon }
			add_resource = {
				energy = 500
				minerals = 800
				food = 200
			}
		}

		# Romulan Star Empire - capture : déjà le plus confortable (900 énergie +9, etc.)
		# Aucun add_resource volontaire.

		# Cardassian Union - capture : 700 énergie -1/mois, 100 nourriture, minéraux OK
		if = {
			limit = { has_country_id = cardassian }
			add_resource = {
				energy = 500
				minerals = 200
				food = 450
			}
		}

		# Ferengi Alliance - seed capital pour premières succursales.
		# Minéraux : coût d'ouverture d'une succursale (400-500) × 2 premiers deals.
		# Énergie : trésorerie d'exploitation (revenus faibles en early game).
		# Influence : diplomatie commerciale, premier pacte commercial dès l'an 1.
		if = {
			limit = { has_country_id = ferengi }
			add_resource = {
				energy = 400
				minerals = 900
				influence = 75
			}
		}
	}
}

country_event = {
	id = cp26_start_balance.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ufop = yes
		NOT = { has_country_flag = cp26_ufop_starting_mining_grid_repaired }
	}

	immediate = {
		set_country_flag = cp26_ufop_starting_mining_grid_repaired

		every_planet_within_border = {
			limit = {
				has_deposit_for = shipclass_mining_station
				has_mining_station = no
			}
			create_mining_station = {
				owner = root
			}
		}
	}
}
