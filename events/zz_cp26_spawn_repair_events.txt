namespace = cp26_spawn_repair

country_event = {
	id = cp26_spawn_repair.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		OR = {
			is_ufop = yes
			is_klingon = yes
			is_romulan = yes
			is_cardassian = yes
			is_ferengi_empire = yes
		}
	}

	immediate = {
		if = {
			limit = { is_ufop = yes }
			every_system = {
				limit = {
					has_star_flag = ufop_system
					has_star_flag = outpost
				}
				if = {
					limit = { exists = starbase }
					starbase = { set_owner = root }
				}
				else = {
					create_starbase = {
						size = starbase_outpost
						owner = root
					}
				}
			}
		}

		if = {
			limit = { is_klingon = yes }
			every_system = {
				limit = {
					has_star_flag = klingon_system
					has_star_flag = outpost
				}
				if = {
					limit = { exists = starbase }
					starbase = { set_owner = root }
				}
				else = {
					create_starbase = {
						size = starbase_outpost
						owner = root
					}
				}
			}
			random_galaxy_planet = {
				limit = { has_planet_flag = planet_rura_penthe }
				if = {
					limit = { is_colony = no }
					create_rura_penthe_colony = yes
				}
				solar_system = {
					if = {
						limit = { exists = starbase }
						starbase = { set_owner = root }
					}
					else = {
						create_starbase = {
							size = starbase_outpost
							owner = root
						}
					}
					set_star_flag = outpost
					set_star_flag = klingon_system
					set_star_flag = klingon
					set_star_flag = rura_penthe_system
				}
			}
		}

		if = {
			limit = { is_romulan = yes }
			every_system = {
				limit = {
					has_star_flag = romulan_system
					has_star_flag = outpost
				}
				if = {
					limit = { exists = starbase }
					starbase = { set_owner = root }
				}
				else = {
					create_starbase = {
						size = starbase_outpost
						owner = root
					}
				}
			}
		}

		if = {
			limit = { is_cardassian = yes }
			every_system = {
				limit = {
					has_star_flag = cardassian_system
					has_star_flag = outpost
				}
				if = {
					limit = { exists = starbase }
					starbase = { set_owner = root }
				}
				else = {
					create_starbase = {
						size = starbase_outpost
						owner = root
					}
				}
			}
		}

		if = {
			limit = { is_ferengi_empire = yes }
			random_galaxy_planet = {
				limit = {
					has_planet_flag = ferengi_capital
					exists = owner
					owner = { is_same_value = root }
				}
				if = {
					limit = { is_colony = no }
					create_ferenginar_colony = yes
				}
				solar_system = {
					if = {
						limit = { exists = starbase }
						starbase = { set_owner = root }
					}
					else = {
						create_starbase = {
							size = starbase_starport
							owner = root
						}
					}
					set_star_flag = outpost
					set_star_flag = ferengi_home_system
					set_star_flag = ferengi_system
					set_star_flag = ferengi
				}
			}
		}
	}
}

country_event = {
	id = cp26_spawn_repair.2
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		OR = {
			is_ufop = yes
			is_klingon = yes
			is_romulan = yes
			is_cardassian = yes
			is_ferengi_empire = yes
		}
	}

	immediate = {
		country_event = { id = cp26_spawn_repair.3 days = 5 }
	}
}

country_event = {
	id = cp26_spawn_repair.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = cp26_rura_penthe_global_repaired }
	}

	immediate = {
		every_country = {
			if = {
				limit = { has_country_flag = klingon }
				save_global_event_target_as = klingon_country
			}
		}

		if = {
			limit = { exists = event_target:klingon_country }
			random_galaxy_planet = {
				limit = { has_planet_flag = planet_rura_penthe }

				if = {
					limit = { is_colony = no }
					create_rura_penthe_colony = yes
				}
				else = {
					set_owner = event_target:klingon_country
					set_colony_type = col_default
				}

				solar_system = {
					if = {
						limit = { exists = starbase }
						starbase = { set_owner = event_target:klingon_country }
					}
					else = {
						create_starbase = {
							size = starbase_outpost
							owner = event_target:klingon_country
						}
					}
					set_star_flag = outpost
					set_star_flag = klingon_system
					set_star_flag = klingon
					set_star_flag = rura_penthe_system
				}
			}
			set_global_flag = cp26_rura_penthe_global_repaired
		}
	}
}
