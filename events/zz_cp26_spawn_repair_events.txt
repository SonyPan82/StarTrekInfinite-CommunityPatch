# Establish reciprocal communications for the CP26 major powers introduced by
# custom country types. Their fleets can otherwise be encountered while the
# diplomacy/contact UI has no valid counterpart to display.
country_event = {
	id = cp26_spawn_repair.5
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { has_global_flag = cp26_cp26_contacts_repaired }
	}

	immediate = {
		every_country = {
			limit = {
				OR = {
					is_cp26_major_mission_country = yes
					is_ferengi_empire = yes
					is_gorn_hegemony = yes
					is_orion_syndicate = yes
					has_country_flag = tholian
				}
			}
			save_event_target_as = cp26_contact_source

			every_country = {
				limit = {
					OR = {
						is_gorn_hegemony = yes
						is_orion_syndicate = yes
						has_country_flag = tholian
					}
					NOT = { is_same_value = event_target:cp26_contact_source }
				}
				save_event_target_as = cp26_contact_target

				event_target:cp26_contact_source = {
					if = {
						limit = { NOT = { has_communications = event_target:cp26_contact_target } }
						establish_contact = { who = event_target:cp26_contact_target hidden = yes }
						establish_communications_no_message = event_target:cp26_contact_target
					}
				}
				event_target:cp26_contact_target = {
					if = {
						limit = { NOT = { has_communications = event_target:cp26_contact_source } }
						establish_contact = { who = event_target:cp26_contact_source hidden = yes }
						establish_communications_no_message = event_target:cp26_contact_source
					}
				}
			}
		}

		every_country = {
			if = {
				limit = { is_ufop = yes }
				save_global_event_target_as = ufop_country
			}
			if = {
				limit = { is_klingon = yes }
				save_global_event_target_as = klingon_country
			}
		}

		# game_start.103 and game_start.260 define this relationship but are not
		# called by Infinite's start on_actions. Restore their intended effect once.
		if = {
			limit = {
				exists = event_target:ufop_country
				exists = event_target:klingon_country
			}
			event_target:ufop_country = {
				if = {
					limit = { NOT = { has_communications = event_target:klingon_country } }
					establish_contact = { who = event_target:klingon_country hidden = yes }
					establish_communications_no_message = event_target:klingon_country
				}
				set_closed_borders = { who = event_target:klingon_country status = no }
				if = {
					limit = { NOT = { has_defensive_pact = event_target:klingon_country } }
					set_defensive_pact = { target = event_target:klingon_country }
				}
			}
			event_target:klingon_country = {
				if = {
					limit = { NOT = { has_communications = event_target:ufop_country } }
					establish_contact = { who = event_target:ufop_country hidden = yes }
					establish_communications_no_message = event_target:ufop_country
				}
				set_closed_borders = { who = event_target:ufop_country status = no }
			}
		}

		set_global_flag = cp26_cp26_contacts_repaired
	}
}

country_event = {
	id = cp26_spawn_repair.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = cp26_rura_penthe_global_repaired }
	}

	immediate = {
		every_country = {
			if = {
				limit = { has_country_flag = klingon }
				save_global_event_target_as = klingon_country
			}
		}

		if = {
			limit = { exists = event_target:klingon_country }
			random_galaxy_planet = {
				limit = { has_planet_flag = planet_rura_penthe }

				if = {
					limit = { is_colony = no }
					create_rura_penthe_colony = yes
				}
				else = {
					set_owner = event_target:klingon_country
					set_colony_type = col_default
				}

				solar_system = {
					if = {
						limit = { exists = starbase }
						starbase = { set_owner = event_target:klingon_country }
					}
					else = {
						create_starbase = {
							size = starbase_outpost
							owner = event_target:klingon_country
						}
					}
					set_star_flag = outpost
					set_star_flag = klingon_system
					set_star_flag = klingon
					set_star_flag = rura_penthe_system
				}
			}
			set_global_flag = cp26_rura_penthe_global_repaired
		}
	}
}
