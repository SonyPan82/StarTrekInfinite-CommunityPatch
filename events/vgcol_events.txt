namespace = vg_col

event = {
	id = vg_col.1000
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		event_target:ufop_country = {	
			random_galaxy_planet = {
				limit = {
					has_planet_flag = planet_earth
				}
				spawn_system = {
					min_jumps = 1
					max_jumps = 3
					min_distance = 15
					max_distance = 38
					initializer = planet_vega_initializer
					hyperlane = yes
					is_discovered = yes
					}	
			}
			last_created_system = {
				set_surveyed = { surveyed = yes surveyor = event_target:ufop_country }
				create_starbase = {
					size = starbase_outpost # We can use any Starbase size. The sizes are located on "game\common\starbase_types"
					owner = event_target:ufop_country # Using the event_target saved on ufop_system_initializer
				 }
				random_system_planet = {
					limit = {
						has_planet_flag = vega_ix_planet
					}
					
					create_colony = {
						owner = event_target:ufop_country
					}
					set_planet_flag = cp26_needs_capital_upgrade
					set_planet_flag = cp26_vega_needs_foundry

					while = {
						count = 9
						create_pop = { species = "Human" }
					}
					add_district_and_planet_size_if_needed_effect = {
						district = district_city
					}
					add_district_and_planet_size_if_needed_effect = {
						district = district_generator
					}
					add_district_and_planet_size_if_needed_effect = {
						district = district_generator
					}
					add_district_and_planet_size_if_needed_effect = {
						district = district_farming
					}
					add_district_and_planet_size_if_needed_effect = {
						district = district_culture
					}
				}
			}
		}
	}
}
