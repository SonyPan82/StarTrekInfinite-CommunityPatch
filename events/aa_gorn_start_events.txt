namespace = aa_gorn_game_start

event = {
	id = aa_gorn_game_start.120
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		every_country = {
			limit = {
				has_country_flag = gorn
			}
			save_global_event_target_as = gorn_country
		}

		if = {
			limit = {
				NOT = { exists = event_target:gorn_country }
			}
			every_system = {
				limit = {
					OR = {
					has_star_flag = gorn_anchor
					has_star_flag = sf_gorncandidate1
					has_star_flag = sf_gorncandidate2
					has_star_flag = sf_gorncandidate3
					has_star_flag = sf_gorncandidate4
					has_star_flag = sf_gorncandidate5
					has_star_flag = sf_gorncandidate6
					has_star_flag = sf_gorncandidatex
				}
			}
			remove_star_flag = gorn_anchor
			remove_star_flag = sf_gorncandidate1
			remove_star_flag = sf_gorncandidate2
			remove_star_flag = sf_gorncandidate3
			remove_star_flag = sf_gorncandidate4
			remove_star_flag = sf_gorncandidate5
			remove_star_flag = sf_gorncandidate6
			remove_star_flag = sf_gorncandidatex
		}

		# Build a broad Gorn pocket in the lower central Federation/Klingon frontier.
		# The goal is reliability first: any valid spawn in this zone is acceptable.
		every_system = {
			limit = {
				is_not_used_space = yes
				any_neighbor_system = {
					has_star_flag = ufop_system
				}
			}
			set_star_flag = sf_gorncandidate1
			set_star_flag = sf_gorncandidatex
		}

		every_system = {
			limit = {
				is_not_used_space = yes
				any_neighbor_system = {
					has_star_flag = klingon_system
				}
			}
			set_star_flag = sf_gorncandidate2
			set_star_flag = sf_gorncandidatex
		}

		every_system = {
			limit = {
				has_star_flag = sf_gorncandidate1
			}
			every_neighbor_system = {
				limit = {
					is_not_used_space = yes
					NOT = { has_star_flag = sf_gorncandidate1 }
				}
				set_star_flag = sf_gorncandidate3
				set_star_flag = sf_gorncandidatex
			}
		}

		every_system = {
			limit = {
				has_star_flag = sf_gorncandidate2
			}
			every_neighbor_system = {
				limit = {
					is_not_used_space = yes
					NOT = { has_star_flag = sf_gorncandidate2 }
				}
				set_star_flag = sf_gorncandidate4
				set_star_flag = sf_gorncandidatex
			}
		}

		if = {
			limit = {
				NOT = { any_system = { has_star_flag = gorn_anchor } }
			}
			random_system = {
				limit = {
					has_star_flag = sf_gorncandidate3
					has_star_flag = sf_gorncandidate4
					NOT = { has_star_flag = sf_gorncandidate1 }
					NOT = { has_star_flag = sf_gorncandidate2 }
				}
				set_star_flag = gorn_anchor
			}
		}

		if = {
			limit = {
				NOT = { any_system = { has_star_flag = gorn_anchor } }
			}
			random_system = {
				limit = {
					has_star_flag = sf_gorncandidate1
					has_star_flag = sf_gorncandidate4
				}
				set_star_flag = gorn_anchor
			}
		}

		if = {
			limit = {
				NOT = { any_system = { has_star_flag = gorn_anchor } }
			}
			random_system = {
				limit = {
					has_star_flag = sf_gorncandidate2
					has_star_flag = sf_gorncandidate3
				}
				set_star_flag = gorn_anchor
			}
		}

		if = {
			limit = {
				NOT = { any_system = { has_star_flag = gorn_anchor } }
			}
			random_system = {
				limit = {
					has_star_flag = sf_gorncandidate1
					has_star_flag = sf_gorncandidate2
				}
				set_star_flag = gorn_anchor
			}
		}

		if = {
			limit = {
				NOT = { any_system = { has_star_flag = gorn_anchor } }
			}
			random_system = {
				limit = {
					OR = {
						has_star_flag = sf_gorncandidate3
						has_star_flag = sf_gorncandidate4
					}
				}
				set_star_flag = gorn_anchor
			}
		}

		if = {
			limit = {
				NOT = { any_system = { has_star_flag = gorn_anchor } }
			}
			random_system = {
				limit = {
					has_star_flag = sf_gorncandidatex
				}
				set_star_flag = gorn_anchor
			}
		}

		if = {
			limit = {
				any_system = { has_star_flag = gorn_anchor }
			}
			random_system = {
				limit = {
					has_star_flag = gorn_anchor
				}
				spawn_gorn_system = yes
				last_created_system = {
					setup_gorn_capital = yes

					spawn_finibus_system = yes
					last_created_system = {
						set_name = "Finibus"
						setup_gorn_territory = yes
					}

					spawn_galdonterre_system = yes
					last_created_system = {
						setup_gorn_territory = yes
					}

					spawn_cestus_system = yes
					last_created_system = {
						setup_gorn_territory = yes
					}
				}
			}
		}

		if = {
			limit = {
				NOT = { exists = event_target:gorn_country }
				any_system = { has_star_flag = gorn_anchor }
			}
			random_system = {
				limit = {
					has_star_flag = gorn_anchor
				}
				spawn_gorn_system_fallback = yes
				last_created_system = {
					setup_gorn_capital = yes

					spawn_finibus_system = yes
					last_created_system = {
						set_name = "Finibus"
						setup_gorn_territory = yes
					}

					spawn_galdonterre_system = yes
					last_created_system = {
						setup_gorn_territory = yes
					}

					spawn_cestus_system = yes
					last_created_system = {
						setup_gorn_territory = yes
					}
					}
				}
			}
		}

		if = {
			limit = {
				exists = event_target:gorn_country
				NOT = { any_system = { has_star_flag = gorn_anchor } }
				any_system = {
					has_star_flag = gorn_home_system
				}
			}
			random_system = {
				limit = {
					has_star_flag = gorn_home_system
				}
				set_star_flag = gorn_anchor
				setup_gorn_capital = yes

				spawn_finibus_system = yes
				last_created_system = {
					set_name = "Finibus"
					setup_gorn_territory = yes
				}

				spawn_galdonterre_system = yes
				last_created_system = {
					setup_gorn_territory = yes
				}

				spawn_cestus_system = yes
				last_created_system = {
					setup_gorn_territory = yes
				}
			}
		}

		if = {
			limit = {
				exists = event_target:gorn_country
			}
			random_system = {
				limit = {
					has_star_flag = gorn_anchor
				}
				setup_gorn_territory = yes
				while = {
					count = 2
					random_neighbor_system = {
						limit = {
							is_not_used_space = yes
						}
						setup_gorn_territory = yes
					}
				}
			}

			every_country = {
				limit = {
					is_major = yes
				}
				establish_contact = { who = event_target:gorn_country hidden = yes }
				establish_communications_no_message = event_target:gorn_country
			}

			if = {
				limit = {
					exists = event_target:ufop_country
				}
				event_target:ufop_country = {
					add_intel = { amount = 35 who = event_target:gorn_country }
				}
			}

			event_target:gorn_country = {
				set_adjective = "Gorn"
				give_technology = { tech = tech_destroyer_unlock message = no }
				give_technology = { tech = tech_cruiser_unlock message = no }

				if = {
					limit = { exists = event_target:ufop_country }
					establish_contact = { who = event_target:ufop_country hidden = yes }
					establish_communications_no_message = event_target:ufop_country
					set_truce = {
						target = event_target:ufop_country
						type = none
						start_date = 2340.06.03
					}
					add_opinion_modifier = {
						who = event_target:ufop_country
						modifier = opinion_insult
					}
					add_intel = { amount = 35 who = event_target:ufop_country }
				}
				if = {
					limit = { exists = event_target:cardassian_country }
					establish_contact = { who = event_target:cardassian_country hidden = yes }
					establish_communications_no_message = event_target:cardassian_country
				}
				if = {
					limit = { exists = event_target:romulan_country }
					establish_contact = { who = event_target:romulan_country hidden = yes }
					establish_communications_no_message = event_target:romulan_country
				}
				if = {
					limit = { exists = event_target:klingon_country }
					establish_contact = { who = event_target:klingon_country hidden = yes }
					establish_communications_no_message = event_target:klingon_country
				}

				create_civilian_fleet = yes

				create_fleet = {
					set_take_point = yes
					effect = {
						set_owner = event_target:gorn_country

						while = {
							count = 4
							create_ship = {
								name = random
								random_existing_design = gorncorvette
								initial_officers_perc = 0.6
							}
						}
						while = {
							count = 3
							create_ship = {
								name = random
								random_existing_design = gorndestroyer
								initial_officers_perc = 0.15
							}
						}
						create_ship = {
							name = random
							random_existing_design = gornbattleship
							initial_officers_perc = 0.1
						}

						set_location = prev.capital_star
					}
				}
			}
		}
	}
}
