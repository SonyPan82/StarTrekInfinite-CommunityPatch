# CP26 Fix — country.27500 (UFOP First Contact Bonus)
# 
# BUG vanilla : in the original event (hornblower_country_events.txt:6136),
# the second add_opinion_modifier block uses 'who = root.from' which can
# resolve to the same country as the scope executing the effect, causing
# the engine to log "Trying to add_opinion_modifier on oneself" and skip
# the opinion bonus entirely.
#
# FIX history :
# v1 (CP26 audit) : Replaced 'who = root.from' with 'who = from' and added
#                   is_same_value guards at the outer limit.
# v2 (2026-05-12) : Moved the is_same_value check INSIDE the target scope.
#                   The previous outer-limit check was compared against the
#                   root scope of the event (e.g. the player country), not
#                   against fromfrom/from at the inner scope where the effect
#                   is actually applied. When the chained first_contact
#                   targets resolved to the same country (rare but happens
#                   on self-triggered chains), the engine still logged the
#                   "add_opinion_modifier on oneself" warning. Moving the
#                   check inside guarantees the comparison uses the right
#                   contextual scope.

first_contact_event = {
	id = country.27500
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			FROM = { has_civic = civic_diplomatic_expertise }
			FROMFROM = { has_civic = civic_diplomatic_expertise }
		}
	}

	immediate = {
		if = {
			limit = {
				exists = from
				exists = fromfrom
			}
			from = {
				if = {
					limit = { NOT = { is_same_value = fromfrom } }
					add_opinion_modifier = {
						who = fromfrom
						modifier = opinion_civic_diplomatic_expertise
					}
				}
			}
		}

		if = {
			limit = {
				exists = fromfrom
				exists = from
			}
			fromfrom = {
				if = {
					limit = { NOT = { is_same_value = from } }
					add_opinion_modifier = {
						who = from
						modifier = opinion_civic_diplomatic_expertise
					}
				}
			}
		}
	}
}
