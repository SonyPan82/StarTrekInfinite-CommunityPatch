namespace = cp26_capital_upgrade

country_event = {
	id = cp26_capital_upgrade.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		country_event = {
			id = cp26_capital_upgrade.2
			days = 3
		}
	}
}

country_event = {
	id = cp26_capital_upgrade.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_owned_planet = {
			limit = {
				has_planet_flag = cp26_needs_capital_upgrade
			}
			remove_building = building_colony_shelter
			add_building = building_capital
			remove_planet_flag = cp26_needs_capital_upgrade
			if = {
				limit = { has_planet_flag = cp26_vega_needs_foundry }
				add_building = building_foundry_1
				remove_planet_flag = cp26_vega_needs_foundry
			}
		}
	}
}
