## war_situation starts
country_event = {
	id = global_situation.17
	title = "global_situation.17.name"
	show_sound = event_generic_warfare
	is_triggered_only = yes

	trigger = {
		always = yes
	}

	# Dynamic Images (Major Power variants)
	picture = {
		trigger = {
		owner = { is_ufop = yes } 
		}
		picture = GFX_evt_empire_fleet_ufop
	} 
	picture = {
		trigger = {
		owner = { is_klingon = yes }
		}
		picture = GFX_evt_empire_fleet_klingon
	}
	picture = {
		trigger = {
		owner = { is_romulan = yes } 
		}
		picture = GFX_evt_empire_fleet_romulan
	} 
	picture = {
		trigger = {
		owner = { is_cardassian = yes }
		}
		picture = GFX_evt_empire_fleet_cardassian
	}
	
	# Default Image on Fail 
	picture = GFX_evt_warscore_background

	desc = {
		trigger = {
			is_ufop = yes
		}
		text = "global_situation.17.ufop.desc"
	} 

	desc = {
		trigger = {
			is_klingon = yes
		}
		text = "global_situation.17.klingon.desc"
	}

	desc = {
		trigger = {
			is_romulan = yes
		}
		text = "global_situation.17.romulan.desc"
	}

	desc = {
		trigger = {
			is_cardassian = yes
		}
		text = "global_situation.17.cardassian.desc"
	}

	desc = {
		trigger = {
			is_major = no
		}
		text = "global_situation.17.hidden.desc"
	} 

	after = {

		# save_event_target_as = war_participant@situation@country

		start_global_situation = {
			type = war_situation

			# No target, see divination_locus or divination_visitors for example
			# target = root

			# No extra effects for now
			# effect = {
			# }
		}

		# CP26 FIX: save war_situation_scope at the after-block level so it is
		# accessible in the every_country block below (outside hidden_effect).
		# In Infinite, event targets saved inside hidden_effect do not persist
		# into sibling blocks — saving it here ensures line 175 can use it.
		last_created_situation = {
			save_event_target_as = war_situation_scope
		}

		hidden_effect = {
			last_created_situation = {
				save_event_target_as = war_situation_scope
			}
	
			# Adding SITUATION PARTICIPANTS
			fromfrom = { 
				set_situation = event_target:war_situation_scope
	
				every_war_participant = {
					event_target:war_situation_scope = {
						add_participants_to_global_situation = {
							who = prev
						}
					}
					set_country_flag = participant_in_situation@event_target:war_situation_scope
					# CP26 FIX: simple flag to avoid situation-scope switch in every_country trigger limit below
					set_country_flag = cp26_war_situation_participant
				}
				every_war_participant = {
					limit = {
						is_war_leader = yes
						is_war_participant = {
							war = fromfrom
							side = attackers
						}
					}
					save_event_target_as = war_situation_attackers_leader
					log = "war_situation_attackers_leader set"
					set_country_flag = attacker_war_leader_flag_@event_target:war_situation_scope
				}
				every_war_participant = {
					limit = {
						is_war_leader = yes
						is_war_participant = {
							war = fromfrom
							side = defenders
						}
					}
					save_event_target_as = war_situation_defenders_leader
					log = "war_situation_defenders_leader set"
					set_country_flag = defender_war_leader_flag_@event_target:war_situation_scope
				}
			}
	
			every_country = {
				limit = {
					not = {
						OR = {
							owner = {
								is_same_value = event_target:war_situation_attackers_leader
							}
							owner = {
								is_same_value = event_target:war_situation_defenders_leader
							}
						}
					}
	
					# HRB: Commenting opinion requirement for now
					# OR = {
					# 	NOT = {
					# 		opinion_level = {
					# 			who = event_target:war_situation_attackers_leader
					# 			level = neutral
					# 		}
					# 	}
					# 	NOT = {
					# 		opinion_level = {
					# 			who = event_target:war_situation_defenders_leader
					# 			level = neutral
					# 		}
					# 	}
					# }
					
					has_communications = event_target:war_situation_attackers_leader
					has_communications = event_target:war_situation_defenders_leader
					is_country_type = default
					NOT = {
						is_war_participant = fromfrom
					}
					# CP26 FIX: replaced "event_target:war_situation_scope = { NOT = { any_situation_participant_country ... } }"
					# which caused "Invalid context switch from [country]" errors at line 175 (vanilla).
					# Infinite does not support situation-scope context switches in trigger limit blocks.
					# Using the simpler cp26_war_situation_participant flag set on all war participants above.
					NOT = { has_country_flag = cp26_war_situation_participant }
				}
				event_target:war_situation_scope = {
					add_participants_to_global_situation = {
						who = prev
					}
				}
				set_country_flag = participant_in_situation@event_target:war_situation_scope
				set_country_flag = cp26_war_situation_participant
			}
			# CP26 FIX: clean up the temporary flag after every_country is done
			every_country = {
				limit = { has_country_flag = cp26_war_situation_participant }
				remove_country_flag = cp26_war_situation_participant
			}
		}
	}

	### UFOP: We will fight.
	option = { 

		trigger = {
			is_ufop = yes
		}

		name = global_situation.17.a
		custom_tooltip = global_situation.17.a.tt
	}

	### Klingon: wo'batlhvad!
	option = { 

		trigger = {
			is_klingon = yes
		}

		name = global_situation.17.b
		custom_tooltip = global_situation.17.b.tt
	}

	### Romulan: Let them come.
	option = { 

		trigger = {
			is_romulan = yes
		}

		name = global_situation.17.c
		custom_tooltip = global_situation.17.c.tt
	}

	### Cardassian: Mobilize forces.
	option = { 

		trigger = {
			is_cardassian = yes
		}

		name = global_situation.17.d
		custom_tooltip = global_situation.17.d.tt
	}

	### Hidden option for MiPos
	option = { 

		trigger = {
			is_major = no
		}

		name = global_situation.17.d
		custom_tooltip = global_situation.17.d.tt
	}
}

	id = global_situation.200	
	title = global_situation.200.name	
	desc = global_situation.200.desc	
	
	# Dynamic Images (Major Power variants)
	picture = {
		trigger = {
		owner = { is_ufop = yes } 
		}
		picture = GFX_evt_red_alert_ufop
	} 
	picture = {
		trigger = {
		owner = { is_klingon = yes }
		}
		picture = GFX_evt_red_alert_klingon
	}
	picture = {
		trigger = {
		owner = { is_romulan = yes } 
		}
		picture = GFX_evt_red_alert_romulan
	} 
	picture = {
		trigger = {
		owner = { is_cardassian = yes }
		}
		picture = GFX_evt_red_alert_cardassian
	}
		
	# Default Image on Fail 
	picture = GFX_evt_warscore_background
	
	show_sound = event_ship_bridge	
	is_triggered_only = yes	
		
	option = {	

		#Gain opinion with everyone else
		hidden_effect = {
			every_country = {
				limit = {
					NOR = {
						is_same_value = root
						is_same_value = event_target:main_attacker_scope
					}
				}
	
				add_opinion_modifier = {
					who = root
					modifier = opinion_defender_stage
				}
			}
		}

		name = global_situation.200.a
		custom_tooltip = global_situation.200.a.tt

		ai_chance = {
			factor = 1
		}
	}	
		
	option = {	

		hidden_effect = {
			#Create Elite armies
			if = {
				limit = {
					exists = this
					is_ufop = yes
				}
				random_owned_planet = {
					limit = {
						is_capital = yes
					}
					while = {
						count = 4
						create_army = {
							owner = THIS
							type = ufop_elite_assault_army
							species = owner_main_species
						}
					}
				}
			}
			if = {
				limit = {
					exists = this
