namespace = cp26_bajoran_wormhole

# ============================================================================
# DISCOVERY CHAIN — DS9-era lore-friendly window (year 2369+)
# ============================================================================

# Yearly check: if Denorios was surveyed before 2369, the project appears once
# the date matches the DS9-era discovery window.
country_event = {
	id = cp26_bajoran_wormhole.110
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		years_passed > 22
		OR = {
			has_country_flag = cp26_denorios_surveyed
			any_system = {
				has_star_flag = denorios_system
				is_surveyed = {
					who = ROOT
					status = yes
				}
			}
		}
		NOT = { has_country_flag = cp26_bajoran_wormhole_project_enabled }
		NOT = { has_global_flag = cp26_bajoran_wormhole_bypass_created }
	}

	immediate = {
		set_country_flag = cp26_denorios_surveyed
		country_event = { id = cp26_bajoran_wormhole.120 }
	}
}

# Denorios survey hook. Before 2369 it only records the discovery; from 2369
# onward it opens the investigation in the situation log.
country_event = {
	id = cp26_bajoran_wormhole.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = from
		from = { has_star_flag = denorios_system }
		NOT = { has_country_flag = cp26_bajoran_wormhole_project_enabled }
		NOT = { has_global_flag = cp26_bajoran_wormhole_bypass_created }
	}

	immediate = {
		set_country_flag = cp26_denorios_surveyed
		if = {
			limit = { years_passed > 22 }
			country_event = { id = cp26_bajoran_wormhole.120 }
		}
	}
}

# Investigation popup, asks the player to start the special project.
country_event = {
	id = cp26_bajoran_wormhole.120
	title = cp26_bajoran_wormhole.120.name
	desc = cp26_bajoran_wormhole.120.desc
	picture = GFX_evt_space_travel_vortex_enterprise_ufop
	is_triggered_only = yes

	immediate = {
		random_system = {
			limit = { has_star_flag = denorios_system }
			save_event_target_as = cp26_denorios_system
		}
	}

	option = {
		name = cp26_bajoran_wormhole.120.ok
		set_country_flag = cp26_bajoran_wormhole_project_enabled
		enable_special_project = {
			name = "CP26_BAJORAN_WORMHOLE_INVESTIGATION"
			location = event_target:cp26_denorios_system
			owner = ROOT
		}
	}

	option = {
		name = cp26_bajoran_wormhole.120.no
	}
}

# ============================================================================
# WORMHOLE ACTIVATION — fired once when special project completes (.901 → .200)
# ============================================================================

# Spawns a linked CP26 bypass between Denorios and Idran, unseals the Gamma
# pocket so internal navigation works, but keeps Idran sealed so the ONLY way
# in/out of Idran from the Alpha side is the bypass itself (no jump/warp).
country_event = {
	id = cp26_bajoran_wormhole.200
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = cp26_bajoran_wormhole_bypass_created }
				NOT = { has_global_flag = bajoran_wormhole_discovered }
			}
			set_global_flag = cp26_bajoran_wormhole_bypass_created
			set_global_flag = bajoran_wormhole_discovered

			random_system = {
				limit = { has_star_flag = denorios_system }
				if = {
					limit = { has_natural_wormhole = no }
					spawn_natural_wormhole = {
						bypass_type = cp26_bajoran_bypass
						random_pos = no
						orbit_angle = 100
						orbit_distance = 120
					}
				}
				save_event_target_as = cp26_denorios_system
			}

			random_system = {
				limit = { has_star_flag = idran_system }
				set_name = "NAME_Idran"
				if = {
					limit = { has_natural_wormhole = no }
					spawn_natural_wormhole = {
						bypass_type = cp26_bajoran_bypass
						random_pos = no
						orbit_angle = 360
						orbit_distance = 20
					}
					link_wormholes = event_target:cp26_denorios_system
				}
			}

			# Unseal the Gamma pocket (except Idran itself) so fleets can
			# navigate it via the internal hyperlanes once they cross.
			# Idran stays sealed — only accessible via the bypass.
			every_system = {
				limit = {
					has_star_flag = bajoran_pocket_system
					NOT = { has_star_flag = idran_system }
				}
				remove_star_flag = sealed_system
			}
		}

		# Make sure the discoverer can see the bypass on the galaxy map.
		add_seen_bypass_type = cp26_bajoran_bypass
		random_system = {
			limit = { has_star_flag = idran_system }
			save_event_target_as = cp26_idran_reveal_system
		}
		set_visited = event_target:cp26_idran_reveal_system

		country_event = { id = cp26_bajoran_wormhole.2 }
	}
}

# Notification: wormhole is now open.
country_event = {
	id = cp26_bajoran_wormhole.2
	title = cp26_bajoran_wormhole.2.name
	desc = cp26_bajoran_wormhole.2.desc
	picture = GFX_evt_space_travel_vortex_enterprise_ufop
	is_triggered_only = yes

	option = {
		name = cp26_bajoran_wormhole.2.ok
	}
}

# Fleet hook: when a fleet enters Denorios or Idran, give its owner the
# bypass type so they can pathfind through it. THIS = fleet, FROM = system.
fleet_event = {
	id = cp26_bajoran_wormhole.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = cp26_bajoran_wormhole_bypass_created
		from = {
			OR = {
				has_star_flag = denorios_system
				has_star_flag = idran_system
			}
		}
		exists = owner
		owner = {
			is_country_type = default
			NOT = { has_country_flag = cp26_seen_bajoran_bypass }
		}
	}

	immediate = {
		owner = {
			set_country_flag = cp26_seen_bajoran_bypass
			add_seen_bypass_type = cp26_bajoran_bypass
		}
	}
}
