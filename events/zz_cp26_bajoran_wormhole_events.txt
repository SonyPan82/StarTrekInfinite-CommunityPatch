namespace = cp26_bajoran_wormhole

# ============================================================================
# WORMHOLE ACTIVATION — called by STH_bajoran_story.1010 CP26 override
# ============================================================================

# Sets the global "wormhole active" flag and shows the discovery notification.
# No bypass or Gamma pocket is spawned — the wormhole exists only cosmetically
# and as a gameplay flag that gates the Gamma expedition pop-up chain.
country_event = {
	id = cp26_bajoran_wormhole.200
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_global_flag = cp26_bajoran_wormhole_bypass_created } }
			set_global_flag = cp26_bajoran_wormhole_bypass_created
			set_global_flag = bajoran_wormhole_discovered
			# Ambient object approach removed — sth_denorios_belt_wormhole not available in Infinite.
			# The star planet in bajoran_bajor_initializer carries set_planet_flag = denorios_belt_target.
			# Initialize DTL on the Dominion country so the yearly tick is ready.
			random_country = {
				limit = {
					has_country_flag = the_dominion
					NOT = { is_variable_set = cp26_dtl }
				}
				set_variable = { which = cp26_dtl value = 5 }
			}
		}
		country_event = { id = cp26_bajoran_wormhole.2 }
	}
}

# Notification: wormhole confirmed active.
country_event = {
	id = cp26_bajoran_wormhole.2
	title = cp26_bajoran_wormhole.2.name
	desc = cp26_bajoran_wormhole.2.desc
	picture = GFX_evt_space_travel_vortex_enterprise_ufop
	is_triggered_only = yes

	option = {
		name = cp26_bajoran_wormhole.2.ok
	}
}

# ============================================================================
# WORMHOLE DISCOVERY — first fleet enters Denorios before discovery
# ============================================================================

# Fires when any fleet enters Denorios while the wormhole is still undiscovered.
# Enables the CP26_WORMHOLE_SURVEY special project at the wormhole ambient object,
# then notifies the owner that a scientific investigation is warranted.
fleet_event = {
	id = cp26_bajoran_wormhole.11
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		from = { has_star_flag = bajoran_system }
		NOT = { has_global_flag = bajoran_wormhole_discovered }

		exists = owner
		owner = {
			is_country_type = default
			NOT = { has_country_flag = bajoran_republic }
			NOT = { has_special_project = CP26_WORMHOLE_SURVEY }
		}
		from = { any_system_planet = { has_planet_flag = denorios_belt_target } }
	}

	immediate = {
		from = {
			save_event_target_as = cp26_wormhole_survey_location
		}
		owner = {
			enable_special_project = {
				name = CP26_WORMHOLE_SURVEY
				location = event_target:cp26_wormhole_survey_location
				owner = this
			}
			country_event = { id = cp26_bajoran_wormhole.12 }
		}
	}
}

# ============================================================================
# GAMMA EXPEDITION — fleet enters Denorios after wormhole is active
# ============================================================================

# Fired by on_entering_system_fleet when a fleet enters Denorios.
# Offers the owning empire a chance to dispatch an expedition through the wormhole.
# Cooldown of 10 years prevents it firing every visit.
fleet_event = {
	id = cp26_bajoran_wormhole.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = cp26_bajoran_wormhole_bypass_created
		has_global_flag = cp26_phase1_resolved
		from = { has_star_flag = bajoran_system }
		exists = owner
		owner = {
			is_country_type = default
			NOT = { has_country_flag = cp26_gamma_expedition_cooldown }
		}
	}

	immediate = {
		owner = {
			set_timed_country_flag = { flag = cp26_gamma_expedition_cooldown days = 3650 }
			country_event = { id = cp26_bajoran_wormhole.300 }
		}
	}
}

# Hostile outcome: Dominion retaliates — Jem'Hadar patrol emerges near Denorios.
country_event = {
	id = cp26_bajoran_wormhole.321
	title = cp26_bajoran_wormhole.321.name
	desc = cp26_bajoran_wormhole.321.desc
	picture = GFX_evt_space_travel_vortex_enterprise_ufop
	is_triggered_only = yes

	immediate = {
		# Spawn a Jem'Hadar patrol from the Dominion empire at Bajor (Denorios Belt).
		random_system = {
			limit = { has_star_flag = bajoran_system }
			save_event_target_as = cp26_denorios_system
		}
		random_country = {
			limit = { has_country_flag = the_dominion }
			create_fleet_from_naval_cap = 0.03
			last_created_fleet = {
				set_name = "Jem'Hadar Strike Force"
				set_location = event_target:cp26_denorios_system
				set_fleet_stance = aggressive
			}
		}
		# Shorten cooldown so the Dominion keeps pressing
		remove_country_flag = cp26_gamma_expedition_cooldown
		set_timed_country_flag = { flag = cp26_gamma_expedition_cooldown days = 1825 }
	}

	option = {
		name = cp26_bajoran_wormhole.321.ok
	}
}

# ============================================================================
# CANONICAL 2369 TRIGGER — Bajor gets the survey project automatically
# ============================================================================

# Fires on_yearly_pulse_country in 2369 for the Bajoran empire.
# Enables CP26_WORMHOLE_SURVEY at the wormhole location — canonical discovery.
country_event = {
	id = cp26_bajoran_wormhole.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = bajoran_republic


		NOT = { has_global_flag = bajoran_wormhole_discovered }
		NOT = { has_special_project = CP26_WORMHOLE_SURVEY }
	}

	immediate = {
		random_system = {
			limit = { has_star_flag = bajoran_system }
			save_event_target_as = cp26_wormhole_2369_location
		}
		if = {
			limit = { exists = event_target:cp26_wormhole_2369_location }
			enable_special_project = {
				name = CP26_WORMHOLE_SURVEY
				location = event_target:cp26_wormhole_2369_location
				owner = this
			}
			country_event = { id = cp26_bajoran_wormhole.21 }
		}
	}
}
