# Romulan starting relationship with the UFOP + treaties
event = {
	id = game_start.240
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		exists = event_target:romulan_country
		any_country = { has_country_flag = ufop }
	}

	immediate = {
		every_country = {
			limit = { has_country_flag = ufop }
			save_global_event_target_as = ufop_country
		}
		event_target:romulan_country = {
			
			add_opinion_modifier = {
				who = event_target:ufop_country
				modifier = opinion_romulan_generation
			}
		}
	}
}
