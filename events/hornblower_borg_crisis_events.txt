### Aftermath Initializer (Called when Ship Enters a System)
# Called from on_entering_system_fleet
# Scope = Fleet
# From = System
fleet_event = {
	id = borg_crisis.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		owner = { has_country_flag = ship_can_go_MIA }
	}
	immediate = {
		if = {
			# The ship must be a Science ship
			limit = {
				is_ship_class = shipclass_science_ship

				# Needs to be outside the Country's borders and also be a Highway
				from = {
					NOT = { is_inside_border = prev.owner }
				    is_star_class = sc_highway_node 
				}
			}

			owner = { remove_country_flag = ship_can_go_MIA }

			save_global_event_target_as = borg_crisis_science_fleet
			from = { save_global_event_target_as = borg_crisis_initial_system }
			
			# Borg Debris System
			event_target:borg_crisis_initial_system = {
				random_system_planet = {
					save_global_event_target_as = borg_crisis_debris_spawn_point
				}
			}

			event_target:borg_crisis_science_fleet = {
				set_fleet_flag = borg_crisis_science_fleet
				set_mia = yes

				fleet_event = { id = borg_crisis.35 days = 1 }
			}
			
			# Also adds the borg_crisis_ambient_debris event target
			spawn_borg_debris = yes
		}
	}
}

	is_triggered_only = yes

		if = {
			limit = {
				NOT = { has_global_flag = borg_crisis_won }
			}

			while = {
				count = @brog_waves_per_tick # Currently 5
				
				# HRB_TODO: create a scripted effect
				# Spawn a Borg Cube Wave
				event_target:borg_crisis_country = {
					random_owned_fleet = {
						limit = {
							has_fleet_flag = imperatoris_borg_cube
							NOT = { has_fleet_flag = wave_spawned_here }
							# is_in_combat = no # HRB: could be useful
						}
						save_global_event_target_as = imperatoris_fleet_spawner
					}

					# There's no spawner left
					if = {
						limit = {
							NOT = { exists = event_target:imperatoris_fleet_spawner }
						}

						# Clean flags
						every_owned_fleet = {
							limit = {
								has_fleet_flag = imperatoris_borg_cube
							}
							remove_fleet_flag = wave_spawned_here
						}

						# Try to select a new Spawner
						random_owned_fleet = {
							limit = {
								has_fleet_flag = imperatoris_borg_cube
								# is_in_combat = no
							}
							save_global_event_target_as = imperatoris_fleet_spawner
						}
					}

					# Spawn Borg Cube Wave
					if = {
						limit = {
							exists = event_target:imperatoris_fleet_spawner
						}

						event_target:imperatoris_fleet_spawner = {
							set_fleet_flag = wave_spawned_here
						
							#HRB_TODO: Every time the event repeats it chooses the same outcome
							random_list = {
								30 = { 
									spawn_borg_cube_small_wave_at_location = {
										LOCATION = event_target:imperatoris_fleet_spawner
									}
								}
								70 = {
									spawn_borg_cube_mid_wave_at_location = {
										LOCATION = event_target:imperatoris_fleet_spawner
									}
								}
							}

							last_created_fleet = {
								set_fleet_flag = wave_invasion_fleet
								fleet_event = { id = borg_crisis.560 }
							}
						}

						clear_global_event_target = imperatoris_fleet_spawner
					}
				}
			}


			# Loop itself
			country_event = { 
				id = borg_crisis.550 
				days = @days_to_borg_cube_reinforcements
			}
		}
	}
}

# Borg Cube Orbits Colony Check
fleet_event = {
	id = borg_crisis.580
	hide_window = yes
	is_triggered_only = yes 
	
	immediate = {
		if = {
			limit = {
				exists = event_target:current_colony_destination
				
				event_target:current_colony_destination = {
					has_owner = yes
					
					NOT = { 
						owner = {
							is_same_value = event_target:borg_crisis_country
						}
					}
				}
			}

			# Colony is valid
			# Check if orbiting the planet
			if = {
				limit = {
					exists = orbit
					orbit = { is_same_value = event_target:current_colony_destination }
				}

				# Initialize days_orbiting_colony Variable
				if = {
					limit = {
						NOT = { is_variable_set = days_orbiting_colony } 
					}
	
					set_variable = {
						which = days_orbiting_colony
						value = 0
					}
				}
	
				# +1 days orbiting
				change_variable = {
					which = days_orbiting_colony
					value = 1
				}

				if = {
					limit = {
						check_variable = {
							which = days_orbiting_colony
							value > @days_to_assimilation
						}
					}
	
					clear_variable = days_orbiting_colony
	
					# Launch End-Game pop-up
					event_target:current_colony_destination = {
						if = {
							limit = {
								is_homeworld = yes
							}

							# HRB_TODO: flag Crisis as finished for that country?

							owner = { country_event = { id = borg_crisis.460 } }
						}
					}

					# Abduct Pops
					event_target:current_colony_destination = {
						set_owner = event_target:borg_crisis_country
						set_controller = event_target:borg_crisis_country
					}
				}
			}

			# Loop itself
			fleet_event = { id = borg_crisis.580 days = 1 }
		}
		else = {
			# Check if there's another Colony in the System
			fleet_event = { id = borg_crisis.576 }
		}
	}
}
