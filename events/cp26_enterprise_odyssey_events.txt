# Enterprise Odyssey Refit  -  Complete
country_event = {
	id = cp26_enterprise_odyssey.2
	title = "cp26_enterprise_odyssey.2.name"
	desc = "cp26_enterprise_odyssey.2.desc"
	picture = GFX_evt_highway_ftl_travel_ufop
	show_sound = event_generic_success

	is_triggered_only = yes

	option = {
		name = "cp26_enterprise_odyssey.2.a"
		custom_tooltip = "cp26_enterprise_odyssey.2.a.tt"

		if = {
			limit = { has_country_flag = cp26_fleet_museum_spawned }
			country_event = { id = cp26_museum.3 days = 30 }
		}

		create_fleet = {
			name = "NAME_Enterprise_Fleet"
			effect = {
				set_owner = ROOT

				create_ship = {
					name = "NAME_Enterprise_Odyssey"
					design = NAME_Odyssey_Class
				}

				set_location = event_target:ufop_sol_starbase_ship

				assign_leader = event_target:enterprise_leader
				set_fleet_flag = enterprise
			}
			settings = {
				can_upgrade = yes
				can_change_leader = no
				can_change_composition = no
				can_disband = no
			}
		}

		event_target:enterprise_fleet = {
			delete_fleet = {
				target = this
				destroy_template = yes
				ignore_is_destroyable = yes
			}
		}

		last_created_ship = {
			save_global_event_target_as = enterprise_ship
		}

		last_created_fleet = {
			save_global_event_target_as = enterprise_fleet
		}
	}
}
