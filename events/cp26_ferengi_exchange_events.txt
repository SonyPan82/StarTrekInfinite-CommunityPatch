namespace = cp26_ferengi_exchange

country_event = {
	id = cp26_ferengi_exchange.90
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_country = {
			limit = {
				OR = {
					is_ferengi_empire = yes
					any_owned_planet = { has_planet_flag = ferengi_capital }
				}
				NOT = { is_country_type = enclave }
				NOT = { has_country_flag = ferengi_xuracorp }
			}
			save_global_event_target_as = ferengi_country
		}

		every_country = {
			limit = { has_country_flag = ferengi_xuracorp }
			save_global_event_target_as = ferengi_xuracorp_country
		}

		if = {
			limit = {
				exists = event_target:ferengi_country
				NOT = { exists = event_target:ferengi_xuracorp_country }
			}

			event_target:ferengi_country = {
				random_owned_planet = {
					limit = { is_capital = yes }
					save_global_event_target_as = cp26_ferengi_exchange_world
				}
			}

			if = {
				limit = { exists = event_target:cp26_ferengi_exchange_world }

				event_target:cp26_ferengi_exchange_world = {
					solar_system = {
						save_global_event_target_as = cp26_ferengi_exchange_system
						set_star_flag = guardians_traders_system
						set_star_flag = enclave
					}
					set_planet_flag = guardians_traders_planet

					create_country = {
						name = "NAME_XuraCorp"
						species = "Ferengi"
						country_id = "ferengi_xuracorp"
						type = enclave
						authority = "auth_oligarchic_ferengi_1"
						civics = {
							civic = civic_wealth_trust
							civic = civic_cumulative_knowledge
						}
						origin = "origin_default"
						flag = {
							icon = {
								category = "enclaves"
								file = "enclaves_flag_trader.dds"
							}
							background = {
								category = "backgrounds"
								file = "00_solid.dds"
							}
							colors = {
								"green"
								"dark_green"
								"null"
								"null"
							}
						}
						ethos = {
							ethic = ethic_xenophile
							ethic = ethic_fanatic_materialist
						}
						ignore_initial_colony_error = yes
					}

					last_created_country = {
						set_country_flag = trader_enclave_country
						set_country_flag = trader_enclave_country_3
						set_country_flag = ferengi
						set_country_flag = ferengi_xuracorp
						save_global_event_target_as = ferengi_xuracorp_country
						set_graphical_culture = mammalian_01

						create_fleet = {
							name = "NAME_XuraCorp_HQ"
							settings = { spawn_debris = no }
							effect = {
								set_owner = prev
								create_ship = {
									name = random
									design = "NAME_Trader_Enclave_Station"
									initial_officers_perc = 1
									graphical_culture = mammalian_01
								}
								set_location = {
									target = prevprev
									distance = 90
								}
								save_event_target_as = CustomCapital
								prev = {
									set_custom_capital_location = event_target:CustomCapital
								}
								set_fleet_flag = trader_enclave_station
							}
						}

						create_leader = {
							class = ruler
							species = this
							name = "PRESCRIPTED_ruler_name_ferengi_2"
							skill = 5
						}
						set_leader = last_created_leader

						establish_communications_no_message = event_target:ferengi_country
						event_target:ferengi_country = {
							establish_communications_no_message = prev
						}
					}
				}
			}
		}
	}
}

country_event = {
	id = cp26_ferengi_exchange.92
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		OR = {
			is_ferengi_empire = yes
			any_owned_planet = { has_planet_flag = ferengi_capital }
		}
		NOT = { is_country_type = enclave }
		NOT = { has_country_flag = ferengi_xuracorp }
	}

	immediate = {
		set_country_flag = ferengi
		every_country = {
			limit = { has_country_flag = ferengi_xuracorp }
			save_global_event_target_as = ferengi_xuracorp_country
		}

		if = {
			limit = { NOT = { exists = event_target:ferengi_xuracorp_country } }
			random_owned_planet = {
				limit = { is_capital = yes }
				solar_system = {
					spawn_system = {
						min_jumps = 1
						max_jumps = 2
						min_distance = 1
						max_distance = 24
						initializer = ferengi_xuracorp_init
						hyperlane = yes
						is_discovered = yes
					}
				}
			}

			last_created_system = {
				set_star_flag = ferengi_system
				set_star_flag = outpost
				set_surveyed = { surveyed = yes surveyor = root }
				create_starbase = {
					size = starbase_outpost
					owner = root
				}
			}

			every_country = {
				limit = { has_country_flag = ferengi_xuracorp }
				save_global_event_target_as = ferengi_xuracorp_country
			}
		}

		if = {
			limit = { exists = event_target:ferengi_xuracorp_country }
			establish_communications_no_message = event_target:ferengi_xuracorp_country
			event_target:ferengi_xuracorp_country = {
				establish_communications_no_message = root
			}
		}
	}
}

country_event = {
	id = cp26_ferengi_exchange.91
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		country_event = { id = cp26_ferengi_exchange.90 }

		if = {
			limit = { exists = event_target:ferengi_xuracorp_country }
			establish_communications_no_message = event_target:ferengi_xuracorp_country
			event_target:ferengi_xuracorp_country = {
				establish_communications_no_message = root
				save_global_event_target_as = cp26_ferengi_exchange_host
				save_event_target_as = traders
				save_global_event_target_as = global_support_country
			}
			save_global_event_target_as = cp26_ferengi_exchange_customer
			save_event_target_as = player_country
			country_event = { id = cp26_ferengi_exchange.1 days = 1 }
		}
	}
}
