namespace = cp26_war_access

# Root = country
# From = war
country_event = {
	id = cp26_war_access.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		save_event_target_as = cp26_war_access_country
		from = {
			every_war_participant = {
				limit = {
					event_target:cp26_war_access_country = {
						is_at_war_with = prev
					}
				}
				event_target:cp26_war_access_country = {
					set_closed_borders = {
						who = prev
						status = no
						forced = 3600
					}
				}
				set_closed_borders = {
					who = event_target:cp26_war_access_country
					status = no
					forced = 3600
				}
			}
		}
		every_country = {
			limit = {
				is_country_type = neutral_zone
			}
			set_closed_borders = {
				who = event_target:cp26_war_access_country
				status = no
				forced = 3600
			}
		}
	}
}
