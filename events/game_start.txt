country_event = {
	id = game_start.9
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# Science
		create_fleet = {
			effect = {
				set_owner = prev

				create_ship = {
					name = random
					random_existing_design = science
				}

				set_fleet_stance = evasive
				#set_location = prev.capital_star
				set_location = prev.capital_scope.star

				owner = {
					create_leader = {
						class = scientist
						sub_type = survey
						name = random
						species = owner_main_species
					}
				}
				set_leader = last_created_leader
			}
		}

		# Constructor
		create_fleet = {
			effect = {
				set_owner = prev

				create_ship = {
					name = random
					random_existing_design = constructor
				}

				set_fleet_stance = evasive
				set_location = prev.capital_star
			}
		}

		# Military
		create_fleet = {
			set_take_point = yes
			effect = {
				set_owner = prev

				while = {
					count = 2
					create_ship = {
						name = random
						random_existing_design = corvette
					}
				}

				set_location = prev.capital_star
			}
		}

		# Spy ships
		if = {
			limit = {
				OR = {
					is_major = yes
					has_country_flag = starting_spy
				}
			}

			create_spy_and_ship = yes
		}
	}
}

	id = game_start.12
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_planet_flag = special_start_planet } }
			if = {
				limit = {
					OR = {
						NOT = { has_planet_flag = planet_earth }
						owner = { has_origin = origin_remnants } #no special earth for you, sorry
					}
				}
				generate_start_deposits_and_blockers = yes
			}
			else_if = {
				limit = { 
					has_planet_flag = planet_earth
					NOT = { 
						owner = { has_origin = origin_remnants } 		 #custom blockers
					}
				}
				generate_earth_deposits_and_blockers = yes
			}
			
			owner = {
				switch = {
					trigger = has_country_id
					ufop = {
						log = "Generating Buildings and Districts for the UFOP"
						PREV = { generate_start_buildings_and_districts_ufop = yes } 
					}
					klingon = {
						log = "Generating Buildings and Districts for the Klingons"
						PREV = { generate_start_buildings_and_districts_klingon = yes } 
					}
					romulan = {
						log = "Generating Buildings and Districts for the Romulans"
						PREV = { generate_start_buildings_and_districts_romulan = yes } 
					}
					cardassian = {
						log = "Generating Buildings and Districts for the Cardassians"
						PREV = { generate_start_buildings_and_districts_cardassian = yes } 
					}
					default = {
						log = "Generating Buildings and Districts for the [this.GetName]s"
						PREV = { generate_start_buildings_and_districts = yes }
					}
				}
			}

			generate_start_pops = yes
		}
	}
}
