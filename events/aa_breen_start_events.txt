namespace = aa_breen_game_start

event = {
	id = aa_breen_game_start.120
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		every_country = {
			limit = {
				has_country_flag = breen
			}
			save_global_event_target_as = breen_country
		}

		if = {
			limit = {
				exists = event_target:breen_country
				any_system = {
					has_star_flag = breen_home_system
				}
			}
			random_system = {
				limit = {
					has_star_flag = breen_home_system
				}
				set_surveyed = { surveyed = yes surveyor = event_target:breen_country }
				if = {
					limit = { exists = event_target:ufop_country }
					set_surveyed = { surveyed = yes surveyor = event_target:ufop_country }
				}
				if = {
					limit = { exists = event_target:cardassian_country }
					set_surveyed = { surveyed = yes surveyor = event_target:cardassian_country }
				}
				if = {
					limit = { exists = event_target:romulan_country }
					set_surveyed = { surveyed = yes surveyor = event_target:romulan_country }
				}
				if = {
					limit = { exists = event_target:klingon_country }
					set_surveyed = { surveyed = yes surveyor = event_target:klingon_country }
				}
				setup_breen_capital = yes
				every_neighbor_system = {
					limit = { is_not_used_space = yes }
					setup_breen_territory = yes
				}
			}
		}

		if = {
			limit = { not = { exists = event_target:breen_country }}

			every_system = {
				limit = {
					is_inside_border = event_target:romulan_country
				}
				every_neighbor_system = {
					limit = { is_not_used_space = yes }
					set_star_flag = sf_breencandidate1
				}
			}
			every_system = {
				limit = {
					is_inside_border = event_target:cardassian_country
				}
				every_neighbor_system = {
					limit = { is_not_used_space = yes }
					set_star_flag = sf_breencandidatex
				}
			}
			every_system = {
				limit = {
					is_inside_border = event_target:cardassian_country
					has_star_flag = colony
					any_system_planet = { has_planet_flag = planet_septimus }
				}
				every_neighbor_system = {
					limit = { is_not_used_space = yes }
					set_star_flag = sf_breencandidate2
				}
				while = {
					count = 2
					random_neighbor_system = {
						limit = {
							is_not_used_space = yes
							not = { has_star_flag = sf_breencandidatex }
						}
						spawn_close_system = yes
						last_created_system = {
							set_star_flag = sf_breencandidate2
						}
					}
				}
			}

			random_system = {
				limit = {
					has_star_flag = sf_breencandidate2
					has_star_flag = sf_breencandidate1
				}
				spawn_close_system = yes
				last_created_system = {
					set_star_flag = sf_breencandidate2
				}
			}

			random_system = {
				limit = {
					has_star_flag = sf_breencandidate2
					has_star_flag = sf_breencandidate1
				}
				spawn_close_system = yes
				last_created_system = {
					set_star_flag = sf_breencandidate1
				}
			}

			every_system = {
				limit = { has_star_flag = sf_breencandidate1 }
				every_neighbor_system = {
					limit = {
						is_not_used_space = yes
						not = { has_star_flag = sf_breencandidate1 }
					}
					set_star_flag = sf_breencandidate3
				}
			}
			every_system = {
				limit = { has_star_flag = sf_breencandidate2 }
				every_neighbor_system = {
					limit = {
						is_not_used_space = yes
						not = { has_star_flag = sf_breencandidate2 }
					}
					set_star_flag = sf_breencandidate4
				}
			}

			random_system = {
				limit = {
					or = {
						and = {
							has_star_flag = sf_breencandidate3
							has_star_flag = sf_breencandidate4
						}
						and = {
							has_star_flag = sf_breencandidate1
							has_star_flag = sf_breencandidate3
						}
						and = {
							has_star_flag = sf_breencandidate2
							has_star_flag = sf_breencandidate4
						}
					}
				}
				spawn_breen_system = yes
				last_created_system = {
					setup_breen_capital = yes
					while = {
						count = 2
						spawn_mid_system = yes
						last_created_system = {
							setup_breen_territory = yes
						}
					}
					every_neighbor_system = {
						limit = { is_not_used_space = yes }
						setup_breen_territory = yes
					}
				}
			}
		}

		### No Breen? Try again with wider search ###
		if = {
			limit = { not = { exists = event_target:breen_country }}
			random_system = {
				limit = {
					or = {
						has_star_flag = sf_breencandidate3
						has_star_flag = sf_breencandidate4
					}
				}
				spawn_breen_system_fallback = yes
				last_created_system = {
					setup_breen_capital = yes
					while = {
						count = 2
						spawn_mid_system = yes
						last_created_system = {
							setup_breen_territory = yes
						}
					}
					every_neighbor_system = {
						limit = { is_not_used_space = yes }
						setup_breen_territory = yes
					}
				}
			}
		}

		if = {
			limit = { exists = event_target:breen_country }

			every_country = {
				limit = { is_major = yes }
				establish_contact = { who = event_target:breen_country hidden = yes }
				establish_communications_no_message = event_target:breen_country
			}

			event_target:breen_country = {
				set_adjective = "Breen"
				give_technology = { tech = tech_destroyer_unlock message = no }
				give_technology = { tech = tech_cruiser_unlock message = no }
				establish_contact = { who = event_target:ufop_country hidden = yes }
				establish_communications_no_message = event_target:ufop_country
				establish_contact = { who = event_target:cardassian_country hidden = yes }
				establish_communications_no_message = event_target:cardassian_country
				establish_contact = { who = event_target:romulan_country hidden = yes }
				establish_communications_no_message = event_target:romulan_country
				establish_contact = { who = event_target:klingon_country hidden = yes }
				establish_communications_no_message = event_target:klingon_country
				#every_first_contact = {
				#	finish_first_contact = yes
				#}
				set_truce = {
					target = event_target:cardassian_country
					type = none
					start_date = 2342.06.03
				}
				set_truce = {
					target = event_target:romulan_country
					type = none
					start_date = 2343.06.03
				}
				add_opinion_modifier = {
					who = event_target:romulan_country
					modifier = opinion_insult
				}
				add_opinion_modifier = {
					who = event_target:cardassian_country
					modifier = opinion_insult
				}

				every_planet_within_border = {
					limit = {
						has_deposit_for = shipclass_mining_station
						has_mining_station = no
					}
					create_mining_station = {
						owner = event_target:breen_country
					}
				}

				create_civilian_fleet = yes

				# Military
				create_fleet = {
					set_take_point = yes
					effect = {
						set_owner = prev

						while = {
							count = 4
							create_ship = {
								name = random
								random_existing_design = breencorvette
								initial_officers_perc = 0.6
							}
						}
						while = {
							count = 3
							create_ship = {
								name = random
								random_existing_design = breendestroyer
								initial_officers_perc = 0.15
							}
						}
						create_ship = {
							name = random
							random_existing_design = breenbattleship
							initial_officers_perc = 0.1
						}

						set_location = prev.capital_star
					}
				}
			}

			event_target:ufop_country = {
				add_intel = { amount = 15 who = event_target:breen_country }
			}
			event_target:cardassian_country = {
				add_opinion_modifier = {
					who = event_target:breen_country
					modifier = opinion_insult
				}
				add_intel = { amount = 50 who = event_target:breen_country }
			}
			event_target:romulan_country = {
				add_opinion_modifier = {
					who = event_target:breen_country
					modifier = opinion_insult
				}
				add_intel = { amount = 40 who = event_target:breen_country }
			}
		}
	}
}
