namespace = cp26_dominion_threat

# ============================================================================
# DOMINION THREAT LEVEL SYSTEM — Phase 1: Discovery & First Contact
#
# DTL is a variable stored on the Dominion country (0-100).
#   0-20  : Curiosity / Scouts
#   20-40 : Infiltration / Incidents      (Phase 2 — future)
#   40-60 : Military Tensions             (Phase 3 — future)
#   60-80 : Cold War / Dominion War prep  (Phase 4 — future)
#   80-100: Total Invasion                (Phase 5 — future)
#
# Phase 1 flow:
#   .1 (yearly) → fires .100 once for wormhole discoverer
#   .100 → player chooses approach:
#     A (Science)  → enable special project CP26_GAMMA_EXPEDITION_SCIENCE
#                    science ship at Denorios → project completes → .200
#     B (Military) → enable special project CP26_GAMMA_EXPEDITION_MILITARY
#                    military ship at Denorios → project completes → .210
#     C (Isolate)  → immediate effects, .102 Bajor tension after delay
#   Phase resolves → cp26_phase1_resolved set → ongoing expeditions unlock
# ============================================================================

# ============================================================================
# .1 — YEARLY TICK (Dominion country scope via on_yearly_pulse_country)
# ============================================================================

country_event = {
	id = cp26_dominion_threat.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = cp26_dominion_arc_blocked }
		cp26_is_dominion_empire = yes
		has_global_flag = bajoran_wormhole_discovered
	}

	immediate = {
		if = {
			limit = { NOT = { is_variable_set = cp26_dtl } }
			set_variable = { which = cp26_dtl value = 5 }
		}

		# Passive escalation: +1 DTL per year
		change_variable = { which = cp26_dtl value = 1 }
		if = {
			limit = { check_variable = { which = cp26_dtl value > 100 } }
			set_variable = { which = cp26_dtl value = 100 }
		}

		# Threshold flags
		if = {
			limit = {
				check_variable = { which = cp26_dtl value >= 20 }
				NOT = { has_global_flag = cp26_dtl_20_reached }
			}
			set_global_flag = cp26_dtl_20_reached
		}
		if = {
			limit = {
				check_variable = { which = cp26_dtl value >= 40 }
				NOT = { has_global_flag = cp26_dtl_40_reached }
			}
			set_global_flag = cp26_dtl_40_reached
		}
		if = {
			limit = {
				check_variable = { which = cp26_dtl value >= 60 }
				NOT = { has_global_flag = cp26_dtl_60_reached }
			}
			set_global_flag = cp26_dtl_60_reached
		}
		if = {
			limit = {
				check_variable = { which = cp26_dtl value >= 80 }
				NOT = { has_global_flag = cp26_dtl_80_reached }
			}
			set_global_flag = cp26_dtl_80_reached
		}
		if = {
			limit = {
				check_variable = { which = cp26_dtl value >= 100 }
				NOT = { has_global_flag = cp26_dtl_100_reached }
			}
			set_global_flag = cp26_dtl_100_reached
		}

		# Phase 1 — fire approach choice once for the wormhole discoverer
		if = {
			limit = { NOT = { has_global_flag = cp26_phase1_first_contact_fired } }
			set_global_flag = cp26_phase1_first_contact_fired
			random_country = {
				limit = {
					has_country_flag = wormhole_discoverer
					cp26_is_major_player_empire = yes
				}
				country_event = { id = cp26_dominion_threat.100 days = 15 random = 10 }
			}
		}

		# Phase 2 — Jem'Hadar incidents (DTL 20-40)
		if = {
			limit = {
				has_global_flag = cp26_dtl_20_reached
				has_global_flag = cp26_phase1_resolved
				NOT = { has_global_flag = cp26_phase2_intro_fired }
			}
			set_global_flag = cp26_phase2_intro_fired
			random_country = {
				limit = {
					has_country_flag = wormhole_discoverer
					cp26_is_major_player_empire = yes
				}
				country_event = { id = cp26_dominion_threat.400 days = 30 random = 20 }
			}
		}

		# Phase 3 — Changeling infiltration intro (DTL 40+, Phase 2 resolved)
		if = {
			limit = {
				has_global_flag = cp26_dtl_40_reached
				has_global_flag = cp26_phase2_resolved
				NOT = { has_global_flag = cp26_phase3_intro_fired }
			}
			set_global_flag = cp26_phase3_intro_fired
			set_variable = { which = cp26_changeling_infiltration value = 5 }
			random_country = {
				limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
				country_event = { id = cp26_dominion_threat.500 days = 30 random = 20 }
			}
		}

		# Phase 3 — Changeling incident ticker (20% per year while infiltration > 3)
		if = {
			limit = {
				has_global_flag = cp26_phase3_intro_fired
				NOT = { has_global_flag = cp26_phase3_resolved }
				is_variable_set = cp26_changeling_infiltration
				check_variable = { which = cp26_changeling_infiltration value > 3 }
			}
			random = {
				chance = 20
				random_country = {
					limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
					country_event = { id = cp26_dominion_threat.520 days = 5 random = 30 }
				}
			}
		}

		# Phase 4 — Cardassia/Dominion alliance (DTL 60+, Phase 3 resolved)
		if = {
			limit = {
				has_global_flag = cp26_dtl_60_reached
				has_global_flag = cp26_phase3_resolved
				NOT = { has_global_flag = cp26_phase4_intro_fired }
			}
			set_global_flag = cp26_phase4_intro_fired
			random_country = {
				limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
				country_event = { id = cp26_dominion_threat.600 days = 30 random = 20 }
			}
		}

		# Phase 5 — War intro (DTL 80+, Phase 4 resolved)
		if = {
			limit = {
				has_global_flag = cp26_dtl_80_reached
				has_global_flag = cp26_phase4_resolved
				NOT = { has_global_flag = cp26_phase5_intro_fired }
			}
			set_global_flag = cp26_phase5_intro_fired
			random_country = {
				limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
				country_event = { id = cp26_dominion_threat.700 days = 15 random = 10 }
			}
		}

		# Phase 5 — Frontier skirmishes (DTL 80-89, 40% chance/year)
		if = {
			limit = {
				has_global_flag = cp26_phase5_intro_fired
				NOT = { has_global_flag = cp26_phase5_full_assault }
				check_variable = { which = cp26_dtl value >= 80 }
				check_variable = { which = cp26_dtl value < 90 }
			}
			random = {
				chance = 40
				random_country = {
					limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
					country_event = { id = cp26_dominion_threat.701 days = 5 random = 20 }
				}
			}
		}

		# Phase 5 — System strikes (DTL 90-99, 60% chance/year)
		if = {
			limit = {
				has_global_flag = cp26_phase5_intro_fired
				NOT = { has_global_flag = cp26_ds9_attack_fired }
				check_variable = { which = cp26_dtl value >= 90 }
				check_variable = { which = cp26_dtl value < 100 }
			}
			set_global_flag = cp26_phase5_full_assault
			random = {
				chance = 60
				random_country = {
					limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
					country_event = { id = cp26_dominion_threat.710 days = 5 random = 20 }
				}
			}
		}

		# Phase 5 — DS9 assault (DTL 100, once)
		if = {
			limit = {
				has_global_flag = cp26_dtl_100_reached
				has_global_flag = cp26_phase5_intro_fired
				NOT = { has_global_flag = cp26_ds9_attack_fired }
			}
			set_global_flag = cp26_ds9_attack_fired
			random_country = {
				limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
				country_event = { id = cp26_dominion_threat.720 days = 5 }
			}
		}

		# Phase 5 — Prophets intervention (20%/year if conditions met)
		if = {
			limit = {
				has_global_flag = cp26_ds9_attack_fired
				NOT = { has_global_flag = cp26_ds9_captured }
				NOT = { has_global_flag = cp26_wormhole_mined }
				NOT = { has_global_flag = cp26_prophets_fired }
				has_global_flag = cp26_phase6_prophets_possible
			}
			random = {
				chance = 20
				set_global_flag = cp26_prophets_fired
				random_country = {
					limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
					country_event = { id = cp26_dominion_threat.730 days = 5 }
				}
			}
		}

		# Victory check — DTL < 20 after Phase 5 started
		if = {
			limit = {
				has_global_flag = cp26_phase5_intro_fired
				NOT = { has_global_flag = cp26_dominion_war_victory }
				NOT = { has_global_flag = cp26_dominion_war_defeat }
				check_variable = { which = cp26_dtl value < 20 }
			}
			set_global_flag = cp26_dominion_war_victory
			random_country = {
				limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
				country_event = { id = cp26_dominion_threat.800 days = 5 }
			}
		}

		# Defeat check — DS9 captured + DTL 100 for 10+ years
		if = {
			limit = {
				has_global_flag = cp26_ds9_captured
				has_global_flag = cp26_dtl_100_reached
				NOT = { has_global_flag = cp26_dominion_war_victory }
				NOT = { has_global_flag = cp26_dominion_war_defeat }
			}
			change_variable = { which = cp26_occupation_years value = 1 }
			if = {
				limit = { check_variable = { which = cp26_occupation_years value >= 10 } }
				set_global_flag = cp26_dominion_war_defeat
				random_country = {
					limit = { has_country_flag = wormhole_discoverer cp26_is_major_player_empire = yes }
					country_event = { id = cp26_dominion_threat.810 days = 5 }
				}
			}
		}

		# Canon ending — Cardassia revolt (conditions met)
		if = {
			limit = {
				has_global_flag = cp26_cardassia_dominion_ally
				has_global_flag = cp26_phase5_intro_fired
				NOT = { has_global_flag = cp26_dominion_war_victory }
				NOT = { has_global_flag = cp26_dominion_war_defeat }
				NOT = { has_global_flag = cp26_canon_ending_fired }
				check_variable = { which = cp26_dtl value >= 60 }
				check_variable = { which = cp26_dtl value <= 90 }
			}
			random_country = {
				limit = {
					has_country_flag = wormhole_discoverer
					cp26_is_major_player_empire = yes
					any_country = {
						OR = {
							has_country_flag = bajoran_republic
							has_country_flag = bajoran
						}
						opinion = { who = prev value >= 30 }
					}
				}
				set_global_flag = cp26_canon_ending_fired
				country_event = { id = cp26_dominion_threat.820 days = 30 random = 20 }
			}
		}
	}
}

# ============================================================================
# .100 — APPROACH CHOICE (fires for wormhole discoverer)
# Player picks their strategy. The actual outcome depends on the ship they send.
# ============================================================================

country_event = {
	id = cp26_dominion_threat.100
	title = cp26_dominion_threat.100.name
	desc = cp26_dominion_threat.100.desc
	picture = GFX_evt_first_contact_ufop_bajoran_stage3
	is_triggered_only = yes

	# A — SCIENCE EXPEDITION
	option = {
		name = cp26_dominion_threat.100.a
		set_country_flag = cp26_gamma_approach_science
		cp26_dtl_increase = { AMOUNT = 2 }
		# Survey site = Bajor Prime (denorios_belt_target), not the star.
		cp26_save_bajor_denorios_project_location = yes
		enable_special_project = {
			name = "CP26_GAMMA_EXPEDITION_SCIENCE"
			location = event_target:cp26_bajoran_project_system
			owner = ROOT
		}
		hidden_effect = {
			country_event = { id = cp26_dominion_threat.109 days = 3 }
		}
	}

	# B — MILITARY PATROL
	option = {
		name = cp26_dominion_threat.100.b
		set_country_flag = cp26_gamma_approach_military
		cp26_dtl_increase = { AMOUNT = 6 }
		cp26_save_bajor_denorios_project_location = yes
		enable_special_project = {
			name = "CP26_GAMMA_EXPEDITION_MILITARY"
			location = event_target:cp26_bajoran_project_system
			owner = ROOT
		}
		hidden_effect = {
			country_event = { id = cp26_dominion_threat.119 days = 3 }
		}
	}

	# C — ISOLATIONISM
	option = {
		name = cp26_dominion_threat.100.c
		set_country_flag = cp26_gamma_approach_isolation
		add_resource = { energy = -200 }
		cp26_dtl_decrease = { AMOUNT = 5 }
		set_global_flag = cp26_phase1_resolved
		hidden_effect = {
			random_country = {
				limit = {
					OR = {
						has_country_flag = bajoran_republic
						has_country_flag = bajoran
					}
				}
				add_opinion_modifier = {
					who = root
					modifier = opinion_insult
				}
				set_country_flag = cp26_bajor_wormhole_closure_tension
			}
			country_event = { id = cp26_dominion_threat.102 days = 30 }
		}
	}
}

				owner = ROOT
			}
			remove_global_flag = cp26_phase1_resolved
			country_event = { id = cp26_dominion_threat.109 days = 2 }
		}
	}
	option = {
		name = cp26_dominion_threat.102.b
		add_monthly_resource_mult = {
			resource = unity
			value = -2

			max = -20

# .410 — COLONY GONE SILENT (main Phase 2 event)
# Choice drives what ship type the player must deploy.
country_event = {
	id = cp26_dominion_threat.410
	title = cp26_dominion_threat.410.name
	desc = cp26_dominion_threat.410.desc
	picture = GFX_evt_first_contact
	is_triggered_only = yes

	# A — Send a military fleet to investigate
	option = {
		name = cp26_dominion_threat.410.a
		set_country_flag = cp26_phase2_investigating
		hidden_effect = {
			# Pick a frontier colony as the investigation site; fallback to Bajor
			if = {
				limit = {
					any_owned_planet = {
						is_colony = yes
						NOT = { is_capital = yes }
					}
				}
				random_owned_planet = {
					limit = {
						is_colony = yes
						NOT = { is_capital = yes }
					}
					solar_system = {
						save_event_target_as = cp26_colony_invest_system
						save_global_event_target_as = cp26_colony_invest_system
					}
				}
			}
			else = {
				random_system = {
					limit = { has_star_flag = bajoran_system }
					save_event_target_as = cp26_colony_invest_system
					save_global_event_target_as = cp26_colony_invest_system
				}
			}
			enable_special_project = {
				name = "CP26_COLONY_INVESTIGATION"
				location = event_target:cp26_colony_invest_system
				owner = ROOT
			}
			country_event = { id = cp26_dominion_threat.412 days = 3 }
		}
	}

	# B — Ignore the report
	option = {
