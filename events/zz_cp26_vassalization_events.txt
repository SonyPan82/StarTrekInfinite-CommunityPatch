namespace = cp26_vassalization

country_event = {
	id = cp26_vassalization.1
	title = cp26_vassalization.1.name
	desc = cp26_vassalization.1.desc
	picture = GFX_evt_empire_fleet_ufop
	show_sound = event_default
	is_triggered_only = yes

	immediate = {
		save_event_target_as = cp26_vassalization_recipient
	}

	option = {
		name = cp26_vassalization.1.accept
		ai_chance = {
			factor = 10
			modifier = {
				# Superior power + excellent opinion: likely to submit
				factor = 2
				AND = {
					relative_power = { who = event_target:cp26_vassalization_actor value >= superior }
					opinion_level = { who = event_target:cp26_vassalization_actor level >= excellent }
				}
			}
			modifier = {
				# Superior power + any positive signal: moderately likely
				factor = 2
				AND = {
					relative_power = { who = event_target:cp26_vassalization_actor value >= superior }
					OR = {
						opinion_level = { who = event_target:cp26_vassalization_actor level >= good }
						has_embassy = event_target:cp26_vassalization_actor
						trust = {
							who = event_target:cp26_vassalization_actor
							value >= 50
						}
					}
				}
			}
			modifier = {
				factor = 0
				opinion_level = { who = event_target:cp26_vassalization_actor level < neutral }
			}
			modifier = {
				factor = 0
				is_at_war_with = event_target:cp26_vassalization_actor
			}
			modifier = {
				factor = 0
				is_in_federation_with = event_target:cp26_vassalization_actor
			}
			modifier = {
				# Ferengi Alliance is a sovereign power  -  never submits voluntarily
				factor = 0
				is_country_type = ferengi_alliance
			}
		}

		set_subject_of = {
			who = event_target:cp26_vassalization_actor
			subject_type = occupied_state
		}
		remove_country_flag = cp26_vassalization_pending_target
		event_target:cp26_vassalization_actor = {
			remove_country_flag = cp26_vassalization_pending_actor
			if = {
				limit = { is_ai = no }
				country_event = { id = cp26_vassalization.2 }
			}
		}
	}

	option = {
		name = cp26_vassalization.1.decline
		ai_chance = {
			factor = 15
			modifier = {
				# Favourable conditions: much less likely to refuse
				factor = 0.3
				AND = {
					relative_power = { who = event_target:cp26_vassalization_actor value >= superior }
					opinion_level = { who = event_target:cp26_vassalization_actor level >= good }
				}
			}
			modifier = {
				factor = 0.5
				AND = {
					relative_power = { who = event_target:cp26_vassalization_actor value >= superior }
					OR = {
						opinion_level = { who = event_target:cp26_vassalization_actor level >= good }
						has_embassy = event_target:cp26_vassalization_actor
						trust = {
							who = event_target:cp26_vassalization_actor
							value >= 50
						}
					}
				}
			}
			modifier = {
				# Hostile opinion: certain refusal
				factor = 3
				opinion_level = { who = event_target:cp26_vassalization_actor level < neutral }
			}
		}

		event_target:cp26_vassalization_actor = {
			add_opinion_modifier = {
				who = root
				modifier = opinion_demanded_subjectization
			}
			set_timed_country_flag = {
				flag = cp26_subjugation_cb_vs_@root
				days = 360
			}
			add_casus_belli = {
				type = cb_subjugation
				who = root
				days = 360
			}
			remove_country_flag = cp26_vassalization_pending_actor
			if = {
				limit = { is_ai = no }
				country_event = { id = cp26_vassalization.3 }
			}
		}
		remove_country_flag = cp26_vassalization_pending_target
	}
}

country_event = {
	id = cp26_vassalization.4
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		save_event_target_as = cp26_vassalization_subject
		FROM = {

# CP26: keep Liberation War CBs in sync for regular empires that become subjects.
country_event = {
	id = cp26_vassalization.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		this = {
			is_regular_empire = yes
			is_subject = yes
		}
		from = {
			is_regular_empire = yes
		}
	}

	immediate = {
		every_country = {
			limit = {
				is_regular_empire = yes
				NOT = { is_same_value = from }
				NOT = { has_country_flag = liberation_war_casus_belli_on_@root }
				has_communications = from
			}

			set_country_flag = liberation_war_casus_belli_on_@root
			add_casus_belli = {
				type = cb_liberation_war
				who = from
				days = -1
			}
		}
	}
}

country_event = {
	id = cp26_vassalization.21
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		this = {
			is_regular_empire = yes
		}
	}

	immediate = {
		every_country = {
			limit = {
				is_regular_empire = yes
				has_country_flag = liberation_war_casus_belli_on_@root
			}

			remove_country_flag = liberation_war_casus_belli_on_@root
			if = {
				limit = {
					exists = from
				}
				remove_casus_belli = {
					type = cb_liberation_war
					who = from
				}
			}
		}

		if = {
			limit = {
				this = { is_subject = yes }
				exists = from
				from = { is_regular_empire = yes }
			}

			every_country = {
				limit = {
					is_regular_empire = yes
					NOT = { is_same_value = from }
					has_communications = from
				}

				set_country_flag = liberation_war_casus_belli_on_@root
				add_casus_belli = {
					type = cb_liberation_war
					who = from
					days = -1
				}
			}
		}
	}
}
