namespace = aa_colrisa

event = {
	id = aa_colrisa.1000
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_country = {
			limit = { has_country_flag = ufop }
			save_global_event_target_as = ufop_country
		}
		create_species = {
			class = "HRB"
			portrait = "betazoid"
			name = "Risan"
			plural = "Risans"
			adjective = "Risan"
			name_list = "BETA"
			traits = {
				trait = "trait_risan_altruistic"
				trait = "trait_risan_nurturing"
				trait = "trait_risan_sedentary"
			}
			effect = {
				save_global_event_target_as = risan_species
			}
		}
		
		random_galaxy_planet = {
			limit = {
				has_planet_flag = planet_risa
			}
			set_owner = event_target:ufop_country
			set_colony_type = col_pleasure
			add_building = building_capital
		}

		event_target:ufop_country = {
			random_owned_planet = {
				limit = {
					has_planet_flag = planet_risa
				}
				while = {
					count = 6
					create_pop = { species = event_target:risan_species }
				}
				add_district_and_planet_size_if_needed_effect = {
					district = district_city
				}
				add_district_and_planet_size_if_needed_effect = {
					district = district_farming
				}
			}
		}
	}
}
