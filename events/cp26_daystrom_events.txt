country_event = {

	title = cp26_daystrom.1.title
	desc = cp26_daystrom.1.desc
	picture = GFX_evt_lab_research_ufop
	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = cp26_daystrom_station_built
		NOT = { has_country_flag = cp26_daystrom_arc_ended }
		NOT = { has_country_flag = cp26_daystrom_spawned_rift_empire }
	}

	option = {
		name = cp26_daystrom.1.a
		if = {
			limit = { NOT = { has_event_chain = cp26_daystrom_chain } }
			begin_event_chain = {
				event_chain = "cp26_daystrom_chain"
			}
		}
		set_country_flag = cp26_daystrom_project_authorized
		country_event = { id = cp26_daystrom.2 days = 180 }
	}

	option = {
		name = cp26_daystrom.1.b
		if = {
			limit = { NOT = { has_event_chain = cp26_daystrom_chain } }
			begin_event_chain = {
				event_chain = "cp26_daystrom_chain"
			}
		}
		set_country_flag = cp26_daystrom_project_limited
		country_event = { id = cp26_daystrom.2 days = 240 }
	}

	option = {
		name = cp26_daystrom.1.c
		set_country_flag = cp26_daystrom_arc_ended
		if = {
			limit = { has_event_chain = cp26_daystrom_chain }
			end_event_chain = cp26_daystrom_chain
		}
	}
}
