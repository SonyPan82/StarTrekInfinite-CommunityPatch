	fire_only_once = yes

	trigger = {
		has_country_flag = cp26_daystrom_station_built
		NOT = { has_country_flag = cp26_daystrom_arc_ended }
		NOT = { has_country_flag = cp26_daystrom_spawned_rift_empire }

country_event = {
	id = cp26_daystrom.4
	title = cp26_daystrom.4.title
	desc = cp26_daystrom.4.desc
	show_sound = event_explosion_minor_ship
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = cp26_daystrom_final_test
		NOT = { has_country_flag = cp26_daystrom_spawned_rift_empire }
	}

	option = {
		name = OK

		hidden_effect = {
			set_country_flag = cp26_daystrom_spawned_rift_empire
			set_country_flag = cp26_daystrom_arc_ended

			random_system = {
				limit = { has_star_flag = cp26_daystrom_station_system }
				save_global_event_target_as = cp26_daystrom_system
			}

			if = {
				limit = { NOT = { has_built_species = "Subspatial Entity" } }
				create_species = {
					class = "ART"
					portrait = "art1"
					name = "Subspatial Entity"
					plural = "Subspatial Entities"
					adjective = "Subspatial"
					name_list = "ART1"
					traits = {
						trait = trait_strong
						trait = trait_nonadaptive
						trait = trait_enduring
					}
				}
			}

				create_country = {
					name = "NAME_CP26_DAYSTROM_RIFT_COUNTRY"
					species = "Subspatial Entity"
					country_id = "cp26_daystrom_rift"
					type = default
					ship_prefix = "SSI"
					name_list = "ART1"
					authority = auth_imperial
					civics = { civic = civic_military_discipline }
					civics = { civic = civic_secret_police }
				origin = "origin_default"
				flag = {
					icon = {
						category = "special"
						file = "flag_borgcube.dds"
					}
					background = {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors = {
						"orange"
						"black"
						"null"
						"null"
					}
				}
				ethos = {
					ethic = ethic_fanatic_militarist
					ethic = ethic_xenophobe
				}
				ignore_initial_colony_error = yes
				effect = {
					set_faction_hostility = {
						set_hostile = yes
						set_neutral = no
						set_friendly = no
					}
				}
			}

				last_created_country = {
					save_global_event_target_as = cp26_daystrom_rift_country
					set_graphical_culture = arthropoid_01
					set_country_flag = cp26_daystrom_rift_country
					give_technology = { tech = tech_destroyer_unlock message = no }
					give_technology = { tech = tech_cruiser_unlock message = no }
				}

				if = {
					limit = {
						exists = event_target:cp26_daystrom_target_world
						exists = event_target:cp26_daystrom_rift_country
					}
					if = {
						limit = { exists = event_target:cp26_daystrom_station_scope }
						event_target:cp26_daystrom_station_scope = {
							remove_megastructure = this
						}
					}
					event_target:cp26_daystrom_target_world = {
						solar_system = {
							spawn_megastructure = {
								type = "cp26_daystrom_rift_habitat"
								planet = event_target:cp26_daystrom_target_world
								owner = event_target:cp26_daystrom_rift_country
								orbit_distance = 9
								orbit_angle = 45
							}
						}
					}
				}

				if = {
					limit = {
						exists = event_target:cp26_daystrom_system
						exists = event_target:cp26_daystrom_rift_country
					}
					event_target:cp26_daystrom_system = {
						every_system_planet = {
							limit = {
								NOT = { is_star = yes }
								exists = owner
							}
							set_owner = event_target:cp26_daystrom_rift_country
							set_controller = event_target:cp26_daystrom_rift_country
						}
						if = {
							limit = { exists = starbase }
							starbase = {
								set_owner = event_target:cp26_daystrom_rift_country
							}
						}
						else = {
							create_starbase = {
								size = starbase_starport
								owner = event_target:cp26_daystrom_rift_country
							}
						}
					}
				}

			if = {
				limit = {
					exists = event_target:cp26_daystrom_rift_world
					exists = event_target:cp26_daystrom_rift_country
				}
				event_target:cp26_daystrom_rift_world = {
					set_name = "NAME_CP26_DAYSTROM_RIFT_WORLD"
					set_planet_flag = cp26_daystrom_rift_world
					create_colony = {
						owner = event_target:cp26_daystrom_rift_country
					}
					set_controller = event_target:cp26_daystrom_rift_country
					set_colony_type = col_default
					set_capital = yes
					clear_blockers = yes
					while = {
						count = 12
						create_pop = { species = "Subspatial Entity" }
					}
					add_district_and_planet_size_if_needed_effect = { district = district_city }
					add_district_and_planet_size_if_needed_effect = { district = district_city }
					add_district_and_planet_size_if_needed_effect = { district = district_generator }
					add_district_and_planet_size_if_needed_effect = { district = district_generator }
					add_district_and_planet_size_if_needed_effect = { district = district_industrial }
					add_building = building_capital
				}
			}

			if = {
				limit = {
					exists = event_target:cp26_daystrom_rift_world
					exists = event_target:cp26_daystrom_rift_country
				}
				create_fleet = {
					name = "NAME_CP26_DAYSTROM_RIFT_FLEET"
					effect = {
						set_owner = event_target:cp26_daystrom_rift_country
						# Vanilla-style scaling: increase the incursion fleet in power bands
						# based on the player's strongest military fleet.
						if = {
							limit = {
								root = {
									any_owned_fleet = {
										is_ship_class = shipclass_military
										fleet_power >= 6500
									}
								}
							}
							while = {
								count = 16
								create_ship = {
									name = random
									design = "NAME_Small_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 8
								create_ship = {
									name = random
									design = "NAME_Medium_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 4
								create_ship = {
									name = random
									design = "NAME_Large_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
						}
						else_if = {
							limit = {
								root = {
									any_owned_fleet = {
										is_ship_class = shipclass_military
										fleet_power >= 3500
									}
								}
							}
							while = {
								count = 12
								create_ship = {
									name = random
									design = "NAME_Small_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 6
								create_ship = {
									name = random
									design = "NAME_Medium_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 3
								create_ship = {
									name = random
									design = "NAME_Large_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
						}
						else_if = {
							limit = {
								root = {
									any_owned_fleet = {
										is_ship_class = shipclass_military
										fleet_power >= 2500
									}
								}
							}
							while = {
								count = 8
								create_ship = {
									name = random
									design = "NAME_Small_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 4
								create_ship = {
									name = random
									design = "NAME_Medium_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 2
								create_ship = {
									name = random
									design = "NAME_Large_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
						}
						else_if = {
							limit = {
								root = {
									any_owned_fleet = {
										is_ship_class = shipclass_military
										fleet_power >= 1500
									}
								}
							}
							while = {
								count = 6
								create_ship = {
									name = random
									design = "NAME_Small_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 3
								create_ship = {
									name = random
									design = "NAME_Medium_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							create_ship = {
								name = random
								design = "NAME_Large_Minor"
								prefix = no
								graphical_culture = arthropoid_01
								initial_officers_perc = 1
							}
						}
						else = {
							while = {
								count = 3
								create_ship = {
									name = random
									design = "NAME_Small_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							while = {
								count = 2
								create_ship = {
									name = random
									design = "NAME_Medium_Minor"
									prefix = no
									graphical_culture = arthropoid_01
									initial_officers_perc = 1
								}
							}
							create_ship = {
								name = random
								design = "NAME_Large_Minor"
								prefix = no
								graphical_culture = arthropoid_01
								initial_officers_perc = 1
							}
						}
						set_location = event_target:cp26_daystrom_rift_world
					}
					settings = {
						can_upgrade = no
						can_change_composition = no
					}
				}
				last_created_fleet = {
					save_global_event_target_as = cp26_daystrom_rift_fleet
				}
			}

				event_target:cp26_daystrom_rift_country = {
					if = {
						limit = { exists = event_target:cp26_daystrom_rift_world }
						set_custom_capital_location = event_target:cp26_daystrom_rift_world
					}

					if = {
						limit = { exists = root }
						establish_communications_no_message = root
						set_faction_hostility = {
							target = root
							set_hostile = yes
							set_neutral = no
							set_friendly = no
						}
						declare_war = {
							target = root
							attacker_war_goal = wg_humiliation
						}
						root = {
							country_event = { id = cp26_daystrom.5 }
						}
					}
				}

				if = {
					limit = { exists = event_target:cp26_daystrom_rift_country }
					set_faction_hostility = {
						target = event_target:cp26_daystrom_rift_country
						set_hostile = yes
						set_neutral = no
						set_friendly = no
					}
				}

				play_country_event_for_all_major_powers = { ID = cp26_daystrom.100 }
			}
		}
}
