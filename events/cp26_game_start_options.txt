namespace = cp26_options

country_event = {
	id = cp26_options.1
	title = cp26_options.1.name
	desc = cp26_options.1.desc
	picture = GFX_evt_lab_research_ufop
	show_sound = event_space_travel
	location = capital_scope
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOT = { has_country_flag = cp26_game_start_options_confirmed }
	}

	immediate = {
		# The Mars / Synth narrative arc is now always enabled for every campaign.
		remove_global_flag = cp26_block_synth_arc
		if = {
			limit = { NOT = { has_global_flag = cp26_bajor_arc_initialized } }
			set_global_flag = cp26_bajor_arc_initialized
		}
		# Default: diplomatic occupied states enabled.
		if = {
			limit = { NOT = { has_global_flag = cp26_diplomacy_options_set } }
			set_global_flag = cp26_diplo_vassalization_enabled
		}
		# Default: Nausicaan marauders disabled
		if = {
			limit = { NOT = { has_global_flag = cp26_nausicaan_options_set } }
			set_global_flag = cp26_block_nausicaan_marauders
		}
	}

	option = {
		name = cp26_options.1.pirates_on
		trigger = { NOT = { has_global_flag = cp26_no_pirates } }
		custom_tooltip = cp26_options.1.pirates_on.tt
		set_global_flag = cp26_no_pirates
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.pirates_off
		trigger = { has_global_flag = cp26_no_pirates }
		custom_tooltip = cp26_options.1.pirates_off.tt
		remove_global_flag = cp26_no_pirates
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.vassalization_on
		trigger = { NOT = { has_global_flag = cp26_diplo_vassalization_disabled } }
		custom_tooltip = cp26_options.1.vassalization_on.tt
		remove_global_flag = cp26_diplo_vassalization_enabled
		set_global_flag = cp26_diplo_vassalization_disabled
		set_global_flag = cp26_diplomacy_options_set
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.vassalization_off
		trigger = { has_global_flag = cp26_diplo_vassalization_disabled }
		custom_tooltip = cp26_options.1.vassalization_off.tt
		remove_global_flag = cp26_diplo_vassalization_disabled
		set_global_flag = cp26_diplo_vassalization_enabled
		set_global_flag = cp26_diplomacy_options_set
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.supernova_quest_on
		trigger = { NOT = { has_global_flag = cp26_block_supernova_arc } }
		custom_tooltip = cp26_options.1.supernova_quest_on.tt
		set_global_flag = cp26_block_supernova_arc
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.supernova_quest_off
		trigger = { has_global_flag = cp26_block_supernova_arc }
		custom_tooltip = cp26_options.1.supernova_quest_off.tt
		remove_global_flag = cp26_block_supernova_arc
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.bajor_arc_on
		trigger = { NOT = { has_global_flag = cp26_block_bajor_arc } }
		custom_tooltip = cp26_options.1.bajor_arc_on.tt
		set_global_flag = cp26_block_bajor_arc
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.bajor_arc_off
		trigger = { has_global_flag = cp26_block_bajor_arc }
		custom_tooltip = cp26_options.1.bajor_arc_off.tt
		remove_global_flag = cp26_block_bajor_arc
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.naval_cap_120
		trigger = {
			OR = {
				has_global_flag = cp26_naval_cap_500
				has_global_flag = cp26_naval_cap_100 # Legacy 300-hull option.
			}
		}
		custom_tooltip = cp26_options.1.naval_cap_120.tt
		remove_global_flag = cp26_naval_cap_500
		remove_global_flag = cp26_naval_cap_100
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.naval_cap_500
		trigger = {
			NOR = {
				has_global_flag = cp26_naval_cap_500
				has_global_flag = cp26_naval_cap_100 # Legacy 300-hull option.
			}
		}
		custom_tooltip = cp26_options.1.naval_cap_500.tt
		set_global_flag = cp26_naval_cap_500
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.nausicaan_on
		trigger = { NOT = { has_global_flag = cp26_block_nausicaan_marauders } }
		custom_tooltip = cp26_options.1.nausicaan_on.tt
		set_global_flag = cp26_block_nausicaan_marauders
		set_global_flag = cp26_nausicaan_options_set
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.nausicaan_off
		trigger = { has_global_flag = cp26_block_nausicaan_marauders }
		custom_tooltip = cp26_options.1.nausicaan_off.tt
		remove_global_flag = cp26_block_nausicaan_marauders
		set_global_flag = cp26_nausicaan_options_set
		hidden_effect = {
			country_event = { id = cp26_options.1 }
		}
	}

	option = {
		name = cp26_options.1.confirm
		custom_tooltip = cp26_options.1.confirm.tt
		set_global_flag = cp26_diplomacy_options_set
		set_country_flag = cp26_game_start_options_confirmed
	}
}
