				name = "CP26_A500_SECURITY_RESPONSE"

	}

		name = cp26_a500.1.b
		set_planet_flag = cp26_a500_science_response
		owner = {
			if = {

				begin_event_chain = {
					event_chain = "cp26_a500_rebellion_chain"
					target = THIS
				}
			}

				name = "CP26_A500_TECHNICIAN_RESPONSE"
				location = event_target:et_a500_mars

##########################################################################
# 2 - Insurrection (country event) - crée les armées rebelles sur Mars
##########################################################################
country_event = {
	id = cp26_a500.2
	title = cp26_a500.2.title
	desc = {
		trigger = {
			exists = event_target:et_a500_mars
			event_target:et_a500_mars = { has_planet_flag = cp26_a500_fleet_response }
		}
		desc = cp26_a500.2.desc_fleet
	}
	desc = cp26_a500.2.desc_default
	picture = GFX_evt_red_alert_ufop
	show_sound = event_explosion_minor_ship
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_a500_resolved }
	}

	immediate = {
		if = {
			limit = { NOT = { exists = event_target:et_a500_mars } }
			random_galaxy_planet = {
				limit = { has_planet_flag = planet_mars }
				save_global_event_target_as = et_a500_mars
			}
		}
	}

	option = {
		name = cp26_a500.2.a

		hidden_effect = {
			# Créer un pays événement pour les rebelles A500
			create_country = {
				name = "NAME_A500_Rebels"
				type = global_event
				auto_delete = yes
			}

			last_created_country = {
				save_global_event_target_as = et_a500_rebel_country
			}

			# Spawner les armées rebelles sur Mars.
			event_target:et_a500_mars = {
				set_planet_flag = cp26_a500_rebellion_active
				add_planet_devastation = 10
				if = {
					limit = {
						count_owned_pop = { count > 3 }
						any_owned_pop = {
							species = { NOT = { is_species_class = CP26_A500 } }
						}
					}
					random_owned_pop = {
						limit = {
							species = { NOT = { is_species_class = CP26_A500 } }
						}
						kill_pop = yes
					}
				}
				if = {
					limit = { has_planet_flag = cp26_a500_fleet_response }
					while = {
						count = 3
						create_army = {
							owner = event_target:et_a500_rebel_country
							type = assault_army
							species = event_target:et_a500_species_global
						}
					}
				}
				else = {
					while = {
						count = 5
						create_army = {
							owner = event_target:et_a500_rebel_country
							type = assault_army
							species = event_target:et_a500_species_global
						}
					}
				}
			}

			# Voie sans réponse initiale → situation critique, intervention lourde
				if = {
					limit = { has_country_flag = cp26_a500_no_response }
					enable_special_project = {
						name = "CP26_A500_GROUND_ASSAULT_HARD"
						location = event_target:et_a500_mars
						owner = this
					}
				}
				# Voie science-first → intervention militaire requise
				else_if = {
					limit = { event_target:et_a500_mars = { has_planet_flag = cp26_a500_science_response } }
					enable_special_project = {
						name = "CP26_A500_GROUND_ASSAULT"
						location = event_target:et_a500_mars
						owner = this
					}
				}
				# Voie military-first → analyse scientifique requise
				else_if = {
					limit = { event_target:et_a500_mars = { has_planet_flag = cp26_a500_fleet_response } }
					enable_special_project = {
						name = "CP26_A500_GROUND_ASSAULT_SCIENCE"
						location = event_target:et_a500_mars
						owner = this
					}
				}
				# Voie governor-first → science + militaire requis
				else = {
					enable_special_project = {
						name = "CP26_A500_GROUND_ASSAULT_GOVERNOR"
						location = event_target:et_a500_mars
						owner = this
					}
				}
			}

		country_event = { id = cp26_a500.3 days = 30 }
	}
}

##########################################################################
# 3 - Pulse mensuel : dévastations progressives
##########################################################################
country_event = {
	id = cp26_a500.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_a500_resolved }
		exists = event_target:et_a500_mars
		event_target:et_a500_mars = {
			has_planet_flag = cp26_a500_rebellion_active
			planet_devastation < 95
		}
	}

	immediate = {
		event_target:et_a500_mars = {
			add_planet_devastation = 5
			if = {
				limit = {
					count_owned_pop = { count > 3 }
					any_owned_pop = {
						species = { NOT = { is_species_class = CP26_A500 } }
					}
				}
				random_list = {
					35 = {
						random_owned_pop = {
							limit = {
								species = { NOT = { is_species_class = CP26_A500 } }
							}
							kill_pop = yes
						}
					}
					65 = { }
				}
			}
		}
		country_event = { id = cp26_a500.3 days = 30 }
	}
}

# 10 - Résolution militaire : combat au sol victorieux
##########################################################################
country_event = {
	id = cp26_a500.10
	title = cp26_a500.10.title
	desc = cp26_a500.10.desc
	picture = GFX_evt_empire_fleet_ufop
	show_sound = event_default
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_a500_resolved }
	}

	immediate = {
		if = {
			limit = { NOT = { exists = event_target:et_a500_mars } }
			random_galaxy_planet = {
				limit = { has_planet_flag = planet_mars }
				save_global_event_target_as = et_a500_mars
			}
		}
	}

	option = {
		name = cp26_a500.10.a

		set_country_flag = cp26_a500_resolved
		set_country_flag = cp26_a500_utopia_modernization_unlocked
		end_event_chain = cp26_a500_rebellion_chain

		if = {
			limit = { exists = event_target:et_a500_rebel_country }
			event_target:et_a500_rebel_country = {
				every_owned_army = {
					remove_army = yes
				}
			}
		}

		event_target:et_a500_mars = {
			solar_system = {
				starbase = {
					while = {
						limit = { has_starbase_module = shipyard }
						remove_starbase_module = { module = shipyard }
					}
				}
			}
			remove_modifier = cp26_a500_installed
			remove_planet_flag = cp26_a500_installed
			remove_planet_flag = cp26_a500_rebellion_active
			remove_planet_flag = cp26_a500_fleet_response
			remove_planet_flag = cp26_a500_science_response
			remove_planet_flag = cp26_a500_governor_response
			add_planet_devastation = -25
		}

		add_research_option = tech_neo_constitution_unlock
		add_research_option = tech_sagan_unlock
		add_research_option = tech_duderstadt_unlock
	}
}
