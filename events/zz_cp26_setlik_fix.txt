namespace = cp26_setlik

country_event = {
	id = cp26_setlik.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		is_ufop = yes
	}

	immediate = {
		random_system = {
			limit = { has_star_flag = setlik }
			set_star_flag = outpost
			set_star_flag = ufop_system
			set_star_flag = ufop
			set_star_flag = setlik
			if = {
				limit = { exists = starbase }
				starbase = { set_owner = root }
			}
			else = {
				create_starbase = {
					size = starbase_outpost
					owner = root
				}
			}
		}
	}
}

country_event = {
	id = cp26_setlik.4
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = ferengi
	}

	immediate = {
		save_event_target_as = ferengi_country

		every_country = {
			if = {
				limit = {
					OR = {
						has_country_flag = cardassian
						has_country_flag = cardassia
					}
				}
				save_event_target_as = cardassian_country
			}
			if = {
				limit = { has_country_flag = tholian }
				save_event_target_as = tholian_country
			}
			if = {
				limit = { has_country_flag = bajoran }
				save_event_target_as = bajoran_country
			}
		}

			random_galaxy_planet = {
				limit = {
					has_planet_flag = ferengi_capital
					exists = owner
					owner = { is_same_value = root }
				}
				if = {
					limit = { is_colony = no }
					create_ferenginar_colony = yes
				}
				if = {
					limit = { NOT = { has_building = building_capital } }
					add_building = building_capital
				}
				if = {
					limit = { NOT = { has_building = cp26_ferengi_energy_consortium } }
					add_building = cp26_ferengi_energy_consortium
				}
				if = {
					limit = { NOT = { has_building = cp26_ferengi_food_consortium } }
					add_building = cp26_ferengi_food_consortium
				}
				if = {
					limit = { has_building = cp26_ferengi_mining_consortium }
					remove_building = cp26_ferengi_mining_consortium
				}
				remove_modifier = minor_power_starting_buffs
				root = {
					set_adjective = "Ferengi"
					set_custom_capital_location = prev
				}
				root = {
					give_technology = { tech = tech_destroyer_unlock message = no }
					give_technology = { tech = tech_megastructure_1 message = no }
					give_technology = { tech = tech_cp26_ferengi_trade_platforms message = no }
					remove_modifier = minor_power_starting_buffs_01
					remove_modifier = minor_civic_debuff
				}
					solar_system = {
						save_event_target_as = ferengi_capital_system
						set_surveyed = { surveyed = yes surveyor = root }
						set_star_flag = outpost
					set_star_flag = ferengi_home_system
				set_star_flag = ferengi_system
				set_star_flag = ferengi
				if = {
					limit = { exists = starbase }
					starbase = { set_owner = root }
				}
				else = {
					create_starbase = {
						size = starbase_starport
						owner = root
					}
				}
			}
		}

		add_modifier = {
			modifier = cp26_ferengi_beta_start_bonus
			days = -1
		}

		random_owned_planet = {
			limit = { is_capital = yes }
			remove_modifier = minor_power_starting_buffs
			if = {
				limit = { NOT = { has_building = building_capital } }
				add_building = building_capital
			}
			if = {
				limit = { NOT = { has_building = cp26_ferengi_energy_consortium } }
				add_building = cp26_ferengi_energy_consortium
			}
			if = {
				limit = { NOT = { has_building = cp26_ferengi_food_consortium } }
				add_building = cp26_ferengi_food_consortium
			}
			if = {
				limit = { has_building = cp26_ferengi_mining_consortium }
				remove_building = cp26_ferengi_mining_consortium
			}
		}
		remove_modifier = minor_power_starting_buffs_01
		remove_modifier = minor_civic_debuff

		if = {
			limit = {
				exists = event_target:cardassian_country
			}
			add_opinion_modifier = {
				who = event_target:cardassian_country
				modifier = cp26_ferengi_cardassian_trade_affinity
			}
			establish_contact = { who = event_target:cardassian_country hidden = yes }
			establish_communications_no_message = event_target:cardassian_country
		}
		if = {
			limit = {
				exists = event_target:cardassian_country
			}
			event_target:cardassian_country = {
				establish_contact = { who = root hidden = yes }
				establish_communications_no_message = root
				add_opinion_modifier = {
					who = root
					modifier = cp26_ferengi_cardassian_trade_affinity
				}
			}
		}
		if = {
			limit = {
				is_ai = no
				exists = event_target:tholian_country
			}
			establish_communications_no_message = event_target:tholian_country
		}
		if = {
			limit = {
				exists = event_target:bajoran_country
			}
			establish_contact = { who = event_target:bajoran_country hidden = yes }
			establish_communications_no_message = event_target:bajoran_country
		}
		if = {
			limit = {
				exists = event_target:bajoran_country
			}
			event_target:bajoran_country = {
				establish_contact = { who = root hidden = yes }
				establish_communications_no_message = root
			}
		}

		if = {
			limit = {
				NOT = { any_owned_fleet = { is_ship_class = shipclass_constructor } }
			}
			create_fleet = {
				effect = {
					set_owner = event_target:ferengi_country

					create_ship = {
						name = random
						random_existing_design = science
					}

					set_fleet_stance = evasive
					set_location = event_target:ferengi_capital_system

					owner = {
						create_leader = {
							class = scientist
							sub_type = survey
							name = random
							species = owner_main_species
						}
					}
					set_leader = last_created_leader
				}
			}

			create_fleet = {
				effect = {
					set_owner = event_target:ferengi_country

					create_ship = {
						name = random
						random_existing_design = constructor
					}

					set_fleet_stance = evasive
					set_location = event_target:ferengi_capital_system
				}
			}

			create_fleet = {
				effect = {
					set_owner = event_target:ferengi_country

					create_ship = {
						name = random
						random_existing_design = spy_ship
					}

					set_fleet_stance = passive
					set_location = event_target:ferengi_capital_system

					owner = {
						create_leader = {
							class = spy
							name = random
							species = owner_main_species
						}
					}
					set_leader = last_created_leader
				}
				settings = {
					can_upgrade = yes
					can_change_composition = no
					can_change_leader = no
					spawn_debris = no
				}
			}
		}

		if = {
			limit = {
				NOT = { any_owned_fleet = { is_ship_class = shipclass_military } }
			}
			create_fleet = {
				set_take_point = yes
				effect = {
					set_owner = root
					create_ship = {
						name = random
						random_existing_design = ferengicorvette
						initial_officers_perc = 0.6
					}
							create_ship = {
								name = random
								random_existing_design = ferengi_destroyer
								initial_officers_perc = 0.2
							}
								set_location = event_target:ferengi_capital_system
						}
					}
				}
	}
}
