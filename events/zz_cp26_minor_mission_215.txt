namespace = cp26_minor_mission

country_event = {
	id = cp26_minor_mission.16
	title = "cp26_minor_mission.16.name"
	desc = "cp26_minor_mission.16.desc"
	picture = GFX_evt_space_distress_ship_minor
	show_sound = event_space_travel
	location = event_target:minor_state_event_fleet

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = { exists = event_target:minor_state_event_system }
				event_target:minor_state_event_system = {
					remove_star_flag = fleet_to_rescue_system
				}
			}

			if = {
				limit = { exists = event_target:minor_state_event_country }
				finished_major_minor_mission = {
					minor_country = event_target:minor_state_event_country
					finished_successfully = yes
				}
				event_target:minor_state_event_country = {
					add_opinion_modifier = {
						who = event_target:major_state_country
						modifier = opinion_mipo_mission_success
					}
				}
			}

			rescue_minor_ship_sp_removed_effect = yes
			first_contact_mission_extra_rewards = yes
			mission_successful_gt_decrease = yes
		}
	}

	option = {
		name = cp26_minor_mission.16.a
		custom_tooltip = cp26_minor_mission.16.a.tt
		ai_chance = { factor = 1 }
	}

	after = {
		hidden_effect = {
			if = {
				limit = { exists = event_target:minor_state_event_fleet }
				destroy_fleet = event_target:minor_state_event_fleet
			}
		}
	}
}

	picture = GFX_evt_pc_barren_sky
	show_sound = event_generic_popup
	is_triggered_only = yes

	option = {
		name = cp26_minor_mission.177.a
		custom_tooltip = cp26_minor_mission.177.a.tt
		ai_chance = { factor = 1 }
	}

	after = {
		owner = {
			if = {
				limit = {
					exists = event_target:planetary_inauguration_leader
				}
				event_target:planetary_inauguration_leader = {
					leader = { add_experience = 100 }
					hidden_effect = {
						remove_trait = leader_trait_governor_inauguration_investigator
					}
				}
			}
		}

		hidden_effect = {
			add_modifier = {
				modifier = gov_remove_monthly_unity
				years = 5
			}
		}
	}
}
